% Newton-Raphson: follow the tangent line at x_i down to the x axis.
% Quadratic convergence near a simple root, but needs f' and can diverge.
% Cropped to its own bounding box so it can be dropped onto a Xournal++ page as
% an image.
%
%   latexmk -pdf newton-raphson-pseudocode.tex
\documentclass[border=6pt,varwidth=11cm]{standalone}

\usepackage{algorithm2e}
\usepackage{amsmath}

\RestyleAlgo{plain}          % no float, no ruled box
\DontPrintSemicolon
\SetKwComment{Comment}{// }{}
\SetKwInOut{Input}{input}
\SetKwInOut{Output}{output}

\begin{document}
\begin{minipage}{\linewidth}
\begin{algorithm}[H]
  \textbf{Newton-Raphson}
  \BlankLine
  \Input{function $f$ and its derivative $f'$, initial guess $x_0$,
         tolerance $\varepsilon_s$, maximum iterations $N$}
  \Output{estimate $x_r$ of the root}
  \BlankLine
  $x_r \leftarrow x_0$\;
  \For{$i \leftarrow 1$ \KwTo $N$}{
    $x_{r,\text{old}} \leftarrow x_r$\;
    \If{$f'(x_{r,\text{old}}) = 0$}{
      \Return failure \tcp*{horizontal tangent}
    }
    \tcp{zero of the tangent line at $x_{r,\text{old}}$}
    $x_r \leftarrow x_{r,\text{old}} - \dfrac{f(x_{r,\text{old}})}{f'(x_{r,\text{old}})}$\;
    $\varepsilon_a \leftarrow \left| \dfrac{x_r - x_{r,\text{old}}}{x_r} \right| \cdot 100\%$\;
    \If{$\varepsilon_a < \varepsilon_s$}{
      \Return $x_r$\;
    }
  }
  \Return $x_r$ \tcp*{did not converge in $N$ iterations}
\end{algorithm}
\end{minipage}
\end{document}
