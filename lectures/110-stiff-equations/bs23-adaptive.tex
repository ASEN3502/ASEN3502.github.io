% Bogacki-Shampine 3(2) adaptive step (used by ode23) with ode45-style
% inputs and outputs.
%
%   latexmk -pdf bs23-adaptive.tex
\documentclass[border=6pt]{standalone}

\usepackage{amsmath}
\usepackage{algpseudocode}

\begin{document}
\begin{minipage}{4.6in}
\begin{algorithmic}[1]
\Function{bs23\_adaptive}{$f, [t_0, t_f], x_0, E_s$}
  \State $t \gets [t_0]$,\quad $x \gets [x_0]$,\quad $h \gets (t_f - t_0)/100$
  \State $k_1 \gets f(t_0, x_0)$
  \While{$t_{\text{end}} < t_f$}
    \State $h \gets \min(h,\; t_f - t_{\text{end}})$
    \State $k_2 \gets f(t_{\text{end}} + \tfrac{h}{2},\; x_{\text{end}} + \tfrac{h}{2} k_1)$
    \State $k_3 \gets f(t_{\text{end}} + \tfrac{3h}{4},\; x_{\text{end}} + \tfrac{3h}{4} k_2)$
    \State $x_3 \gets x_{\text{end}} + h\left(\tfrac{2}{9} k_1 + \tfrac{1}{3} k_2 + \tfrac{4}{9} k_3\right)$ \Comment{3rd order}
    \State $k_4 \gets f(t_{\text{end}} + h,\; x_3)$
    \State $x_2 \gets x_{\text{end}} + h\left(\tfrac{7}{24} k_1 + \tfrac{1}{4} k_2 + \tfrac{1}{3} k_3 + \tfrac{1}{8} k_4\right)$ \Comment{2nd order}
    \State $E_a \gets \lvert x_3 - x_2 \rvert$
    \If{$E_a \le E_s$}
      \State append $t_{\text{end}} + h$ to $t$
      \State append $x_3$ to $x$
      \State $k_1 \gets k_4$ \Comment{slope at the new $x_{\text{end}}$}
    \EndIf
    \State $h \gets 0.9\, h \left(E_s / E_a\right)^{1/3}$ \Comment{b/c $E_a$ is $O(h^3)$}
  \EndWhile
  \State \Return $t, x$
\EndFunction
\end{algorithmic}
\end{minipage}
