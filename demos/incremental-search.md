  <p class="is-intro">Incremental search splits the search range
  \([x_\text{min}, x_\text{max}]\) into \(n\) candidate intervals of width
  \(\Delta x = (x_\text{max} - x_\text{min})/n\) and evaluates \(f\) at every
  boundary.  A candidate \([x_l, x_u]\) is kept as a <b>root bracket</b> when
  \(f(x_l)\,f(x_u) &lt; 0\), because a continuous function that changes sign
  must cross zero somewhere in between.</p>

<script>
(function () {
  // Root counts are for the search range shown.  The second function is
  // Chapra's incremental search example: pairs of roots close enough
  // together that a coarse step lands the same sign on both sides.
  var FUNCS = [
    { f: function (x) { return Math.exp(x) - 4 * Math.sqrt(x); },
      xmin: 0, xmax: 2, ylo: -1.6, yhi: 2.0, ystep: 0.5, xstep: 0.5, nroots: 2,
      desc: '<code>f(x) = e<sup>x</sup> − 4√x</code> on [0, 2], which has ' +
            'two roots, near x ≈ 0.07 and x ≈ 1.64.' },
    { f: function (x) { return Math.sin(10 * x) + Math.cos(3 * x); },
      xmin: 3, xmax: 6, ylo: -2, yhi: 2, ystep: 1, xstep: 0.5, nroots: 9,
      desc: '<code>f(x) = sin(10x) + cos(3x)</code> on [3, 6] (Chapra, ' +
            'Example 5.2), which has nine roots, several of them in closely ' +
            'spaced pairs.' }
  ];
  var f, XMIN, XMAX, YLO, YHI, NROOTS, YSTEP, XSTEP;

  var svg  = document.getElementById('is-svg'),
      nIn  = document.getElementById('is-n'),
      fnIn = document.getElementById('is-fn'),
      showIn = document.getElementById('is-show'),
      descOut = document.getElementById('is-desc'),
      legF = document.getElementById('is-leg-f'),
      nOut = document.getElementById('is-nval'),
      dxOut = document.getElementById('is-dx'),
      sumOut = document.getElementById('is-summary'),
      tbl  = document.getElementById('is-table');

  var W = 640, H = 400, ML = 52, MR = 16, MT = 14, MB = 44;
  var PW = W - ML - MR, PH = H - MT - MB;
  function px(x) { return ML + (x - XMIN) / (XMAX - XMIN) * PW; }
  function py(y) { return MT + PH - (y - YLO) / (YHI - YLO) * PH; }

  function el(name, attrs, text) {
    var e = document.createElementNS('http://www.w3.org/2000/svg', name);
    for (var a in attrs) e.setAttribute(a, attrs[a]);
    if (text !== undefined) e.textContent = text;
    return e;
  }
  function fmt(x, d) { return (x < 0 ? '−' : '+') + Math.abs(x).toFixed(d); }

  // The algorithm, exactly as in the pseudocode.
  function incrementalSearch(f, xmin, xmax, n) {
    var dx = (xmax - xmin) / n, brackets = [], xl = xmin;
    for (var k = 1; k <= n; k++) {
      var xu = xl + dx;
      if (f(xl) * f(xu) < 0) brackets.push([xl, xu]);
      xl = xu;
    }
    return brackets;
  }

  function render() {
    var F = FUNCS[+fnIn.value];
    f = F.f; XMIN = F.xmin; XMAX = F.xmax; YLO = F.ylo; YHI = F.yhi;
    NROOTS = F.nroots; YSTEP = F.ystep; XSTEP = F.xstep;
    descOut.innerHTML = 'Here ' + F.desc;
    var showF = showIn.checked;
    legF.style.visibility = showF ? '' : 'hidden';

    var n = +nIn.value, dx = (XMAX - XMIN) / n;
    var brackets = incrementalSearch(f, XMIN, XMAX, n);

    nOut.textContent = n;
    dxOut.textContent = dx.toFixed(3);

    while (svg.firstChild) svg.removeChild(svg.firstChild);

    // Root brackets first so everything else draws on top of the shading.
    brackets.forEach(function (b) {
      svg.appendChild(el('rect', { x: px(b[0]), y: MT, width: px(b[1]) - px(b[0]),
                                   height: PH, class: 'is-bracket' }));
    });

    // Horizontal grid and y ticks.
    for (var t = Math.ceil(YLO / YSTEP) * YSTEP; t <= YHI + 1e-9; t += YSTEP) {
      svg.appendChild(el('line', { x1: ML, y1: py(t), x2: ML + PW, y2: py(t),
                                   class: 'is-tick' }));
      svg.appendChild(el('text', { x: ML - 8, y: py(t) + 4, 'text-anchor': 'end',
                                   class: 'is-ticklbl' }, t.toFixed(1)));
    }
    // x ticks at 0.5 spacing.
    for (var xt = XMIN; xt <= XMAX + 1e-9; xt += XSTEP) {
      svg.appendChild(el('text', { x: px(xt), y: MT + PH + 18, 'text-anchor': 'middle',
                                   class: 'is-ticklbl' }, xt.toFixed(1)));
    }

    // Candidate interval boundaries.
    for (var k = 0; k <= n; k++) {
      var xb = XMIN + k * dx;
      svg.appendChild(el('line', { x1: px(xb), y1: MT, x2: px(xb), y2: MT + PH,
                                   class: 'is-cand' }));
    }

    // Frame, zero line, axis labels.
    svg.appendChild(el('rect', { x: ML, y: MT, width: PW, height: PH,
                                 fill: 'none', class: 'is-axis' }));
    svg.appendChild(el('line', { x1: ML, y1: py(0), x2: ML + PW, y2: py(0),
                                 class: 'is-zero' }));
    svg.appendChild(el('text', { x: ML + PW / 2, y: H - 6, 'text-anchor': 'middle',
                                 class: 'is-axlbl' }, 'x'));
    svg.appendChild(el('text', { x: 14, y: MT + PH / 2 + 5, 'text-anchor': 'middle',
                                 class: 'is-axlbl' }, 'f(x)'));

    // The curve, clipped to the frame.  Hidden, the plot shows only what the
    // algorithm actually gets to see: the sampled values.
    if (showF) {
      var d = '', N = 800, pen = false;
      for (var i = 0; i <= N; i++) {
        var x = XMIN + (XMAX - XMIN) * i / N, y = f(x);
        if (y < YLO || y > YHI) { pen = false; continue; }
        d += (pen ? ' L' : ' M') + px(x).toFixed(2) + ',' + py(y).toFixed(2);
        pen = true;
      }
      svg.appendChild(el('path', { d: d, class: 'is-curve' }));
    }

    // Sample points at the boundaries.
    for (k = 0; k <= n; k++) {
      xb = XMIN + k * dx;
      var yb = f(xb);
      if (yb >= YLO && yb <= YHI) {
        svg.appendChild(el('circle', { cx: px(xb), cy: py(yb), r: 4.5,
                                       class: 'is-pt' }));
      }
    }

    // Summary and table.
    var missed = NROOTS - brackets.length;
    var s = 'Found <b>' + brackets.length + '</b> root bracket' +
            (brackets.length === 1 ? '' : 's') + ' from ' + n + ' candidate' +
            (n === 1 ? '' : 's') + '.';
    if (missed > 0) {
      s += ' <span class="is-missed">' + missed + ' of the ' + NROOTS +
           ' roots ' + (missed === 1 ? 'was' : 'were') + ' missed:</span> ' +
           'the sign is the same at both ends of every candidate that contains ' +
           (missed === 1 ? 'it' : 'them') + ', so the algorithm cannot tell ' +
           (missed === 1 ? 'it is' : 'they are') + ' there.';
    }
    sumOut.innerHTML = s;

    var h = '<tr><th></th><th>x<sub>l</sub></th><th>x<sub>u</sub></th>' +
            '<th>f(x<sub>l</sub>)</th><th>f(x<sub>u</sub>)</th></tr>';
    brackets.forEach(function (b, i) {
      var fl = f(b[0]), fu = f(b[1]);
      h += '<tr><th>' + (i + 1) + '</th><td>' + b[0].toFixed(4) + '</td><td>' +
           b[1].toFixed(4) + '</td><td class="' + (fl < 0 ? 'is-neg' : 'is-pos') +
           '">' + fmt(fl, 4) + '</td><td class="' + (fu < 0 ? 'is-neg' : 'is-pos') +
           '">' + fmt(fu, 4) + '</td></tr>';
    });
    tbl.innerHTML = h;
  }

  nIn.addEventListener('input', render);
  fnIn.addEventListener('change', render);
  showIn.addEventListener('change', render);
  render();
})();
</script>
