  <div class="cn-def">
    <p><b>Definition</b>: the condition number of an invertible matrix is
    \(\text{cond}(A) = \|A\| \, \|A^{-1}\|\).  It bounds how much a relative error in \(A\) can be
    amplified in the solution:
    \[ \frac{\|\Delta x\|}{\|x\|} \;\le\; \text{cond}(A)\,\frac{\|\Delta A\|}{\|A\|} \]</p>
  </div>

  <p class="cn-intro">Consider the matrix
  \[ A = \begin{bmatrix} 1 & 1 \\ 1 & 1 + \varepsilon \end{bmatrix}, \qquad
     b = \begin{bmatrix} 2 \\ 2 + \varepsilon \end{bmatrix}, \]
  where \(b\) is chosen so that the solution of \(Ax = b\) is always
  \(x = (1, 1)\), whatever the value of \(\varepsilon\).  As
  \(\varepsilon \to 0\) the second row becomes a copy of the first and
  \(A\) becomes singular.</p>

  <p class="cn-intro">Now perturb the bottom-right entry of \(A\) by a small
  amount \(\Delta\), keeping \(b\) the same:
  \[ A + \Delta A = \begin{bmatrix} 1 & 1 \\ 1 & 1 + \varepsilon + \Delta \end{bmatrix},
     \qquad
     \Delta A = \begin{bmatrix} 0 & 0 \\ 0 & \Delta \end{bmatrix}. \]
  The solution moves to \(x + \Delta x\), the intersection of row 1 with
  the <span class="cn-pertword">perturbed row 2</span> on the plot.</p>

<script>
(function () {
  // Row 1 is fixed at [1, 1]; row 2 is [1, 1 + eps].  b = A x*, with
  // x* = (1, 1), so the crossing point never moves.
  var X1 = 1, X2 = 1;
  var LO = -2, HI = 4;                 // plot spans [LO, HI] in both axes,
                                       // centred on the solution
  var DIGITS = 8;                      // decimals shown in every matrix entry

  // Log-scaled slider: eps from 1 down to 1e-8, reversed so that dragging
  // right means "closer to singular".
  var EMAX = 1, EMIN = 1e-8, STEPS = 1000;
  function epsFromSlider(v) {
    var lo = Math.log(EMAX), hi = Math.log(EMIN);
    return Math.exp(lo + (v / STEPS) * (hi - lo));
  }
  // Delta is signed, zero at the centre stop, and log-scaled in magnitude on
  // each side from 1e-6 out to 1.  Negative Delta pushes row 2 toward and
  // then past row 1, so the perturbed system passes through singular.
  var DMAX = 1, DMIN = 1e-6;
  function deltaFromSlider(v) {
    if (v === 0) return 0;
    var lo = Math.log(DMIN), hi = Math.log(DMAX);
    var mag = Math.exp(lo + ((Math.abs(v) - 1) / (STEPS - 1)) * (hi - lo));
    return v < 0 ? -mag : mag;
  }

  function sig(x, n) {
    if (x === 0) return '0';
    var a = Math.abs(x);
    if (a >= 1e5 || a < 1e-4) {
      var parts = x.toExponential(n - 1).split('e');
      return parts[0] + ' × 10' + sup(+parts[1]);
    }
    return String(+x.toPrecision(n));
  }
  function fmtBig(x) {
    if (!isFinite(x)) return '∞';
    if (x < 1000) return x.toFixed(x < 10 ? 2 : 0);
    var e = Math.floor(Math.log(x) / Math.LN10);
    return (x / Math.pow(10, e)).toFixed(1) + ' × 10' + sup(e);
  }
  var SUPS = '⁰¹²³⁴⁵⁶⁷⁸⁹';
  function sup(e) {
    return String(e).split('').map(function (c) {
      return c === '-' ? '⁻' : SUPS[+c];
    }).join('');
  }

  // 2-norm condition number of a 2x2 matrix.  sigma_max comes from the larger
  // eigenvalue of A'A; sigma_min is then |det A| / sigma_max, which avoids the
  // catastrophic cancellation of computing the smaller eigenvalue directly.
  function cond2(a, b, c, d) {
    var t = a * a + b * b + c * c + d * d;      // trace(A'A)
    var det = Math.abs(a * d - b * c);
    var disc = Math.sqrt(Math.max(t * t - 4 * det * det, 0));
    var l1 = (t + disc) / 2;                   // sigma_max^2
    if (det === 0) return Infinity;
    return l1 / det;                           // sigma_max / sigma_min
  }

  var svg    = document.getElementById('cn-svg'),
      epsIn  = document.getElementById('cn-eps'),
      epsOut = document.getElementById('cn-eval'),
      Aout   = document.getElementById('cn-A'),
      bout   = document.getElementById('cn-b'),
      kOut   = document.getElementById('cn-cond'),
      dIn    = document.getElementById('cn-delta'),
      dOut   = document.getElementById('cn-dval'),
      errOut = document.getElementById('cn-err'),
      cells  = { x1: 'cn-x1', x2: 'cn-x2', t1: 'cn-t1', t2: 'cn-t2' };
  for (var k in cells) cells[k] = document.getElementById(cells[k]);

  var M = 34, S = 460;                 // margin, viewBox side
  var PW = S - 2 * M;
  function px(x) { return M + (x - LO) / (HI - LO) * PW; }
  function py(y) { return M + PW - (y - LO) / (HI - LO) * PW; }

  function el(name, attrs, text) {
    var e = document.createElementNS('http://www.w3.org/2000/svg', name);
    for (var a in attrs) e.setAttribute(a, attrs[a]);
    if (text !== undefined) e.textContent = text;
    return e;
  }

  // The segment of a1*x1 + a2*x2 = rhs inside the visible square, found by
  // clipping against the four edges.  Handles vertical and horizontal lines
  // without special cases.
  function segment(a1, a2, rhs) {
    var pts = [];
    if (Math.abs(a2) > 1e-12) {
      pts.push([LO, (rhs - a1 * LO) / a2]);
      pts.push([HI, (rhs - a1 * HI) / a2]);
    }
    if (Math.abs(a1) > 1e-12) {
      pts.push([(rhs - a2 * LO) / a1, LO]);
      pts.push([(rhs - a2 * HI) / a1, HI]);
    }
    var inside = pts.filter(function (p) {
      return p[0] >= LO - 1e-9 && p[0] <= HI + 1e-9 &&
             p[1] >= LO - 1e-9 && p[1] <= HI + 1e-9;
    });
    if (inside.length < 2) return null;
    // Take the two furthest apart, so duplicate corner hits do no harm.
    var best = null;
    for (var i = 0; i < inside.length; i++) {
      for (var j = i + 1; j < inside.length; j++) {
        var dx = inside[i][0] - inside[j][0], dy = inside[i][1] - inside[j][1];
        var d2 = dx * dx + dy * dy;
        if (!best || d2 > best.d2) best = { d2: d2, p: inside[i], q: inside[j] };
      }
    }
    return best && best.d2 > 1e-12 ? best : null;
  }

  function drawLine(a1, a2, rhs, cls, label) {
    var s = segment(a1, a2, rhs);
    if (!s) return;
    svg.appendChild(el('line', { x1: px(s.p[0]), y1: py(s.p[1]),
                                 x2: px(s.q[0]), y2: py(s.q[1]), class: cls }));
    if (!label) return;
    // Labels sit a little way in from one end of the segment and are pushed
    // off the line perpendicularly: row 1 near its left end on one side,
    // row 2 near its right end on the other, so the two never collide
    // however close the lines get.
    var left  = s.p[0] < s.q[0] ? s.p : s.q;
    var right = left === s.p ? s.q : s.p;
    var a = label.left ? left : right, o = label.left ? right : left;
    var ax = px(a[0]), ay = py(a[1]), ox = px(o[0]), oy = py(o[1]);
    var dx = ox - ax, dy = oy - ay, len = Math.hypot(dx, dy) || 1;
    dx /= len; dy /= len;
    var nx = -dy, ny = dx;                       // unit normal
    if (ny < 0) { nx = -nx; ny = -ny; }          // point "down" on screen
    var side = label.left ? 1 : -1;
    var tx = ax + 60 * dx + side * 22 * nx;
    var ty = ay + 60 * dy + side * 22 * ny + 5;
    svg.appendChild(el('text', { x: tx, y: ty, 'text-anchor': 'middle',
                                 class: 'cn-linelbl ' + label.cls }, label.text));
  }

  function matHTML(rows, cls) {
    var g = '<span class="cn-brk"><span class="cn-grid" style="' +
            'grid-template-columns: repeat(' + rows[0].length + ', auto)">';
    for (var i = 0; i < rows.length; i++) {
      for (var j = 0; j < rows[i].length; j++) {
        g += '<span class="' + cls[i] + '">' + rows[i][j] + '</span>';
      }
    }
    return g + '</span></span>';
  }

  function render() {
    var eps = epsFromSlider(+epsIn.value);
    var del = deltaFromSlider(+dIn.value);

    // A = [1 1; 1 1+eps], b = A x*, so x* = (1, 1) exactly.
    var a11 = 1, a12 = 1, a21 = 1, a22 = 1 + eps;
    var b1 = a11 * X1 + a12 * X2;
    var b2 = a21 * X1 + a22 * X2;

    var cond = cond2(a11, a12, a21, a22);

    epsOut.textContent = sig(eps, 3);

    var f = function (x) { return x.toFixed(DIGITS); };
    Aout.innerHTML = matHTML([[f(a11), f(a12)], [f(a21), f(a22)]],
                             ['cn-a1', 'cn-a2']);
    bout.innerHTML = matHTML([[f(b1)], [f(b2)]], ['cn-a1', 'cn-a2']);

    kOut.innerHTML = 'cond(A) = <span class="cn-num">' + fmtBig(cond) + '</span>';

    // Perturbed system: a22 -> a22 + del with b unchanged.  Solving the
    // 2x2 system directly: x2 = eps / (eps + del), x1 = b1 - x2.
    var pivot = a22 + del - a12;
    var singular = Math.abs(pivot) < 1e-15;
    var t2 = (b2 - b1) / pivot;
    var t1 = (b1 - a12 * t2) / a11;

    dOut.textContent = del === 0 ? '0' : (del > 0 ? '+' : '−') + sig(Math.abs(del), 2);
    cells.x1.textContent = sig(X1, 4);
    cells.x2.textContent = sig(X2, 4);
    cells.t1.textContent = singular ? '—' : sig(t1, 4);
    cells.t2.textContent = singular ? '—' : sig(t2, 4);

    // Relative changes in the 2-norm (Frobenius for the matrices, which for
    // this rank-one Delta A equals its 2-norm), and the observed amplification.
    var nA  = Math.hypot(a11, a12, a21, a22), ndA = Math.abs(del);
    var nx  = Math.hypot(X1, X2), ndx = Math.hypot(t1 - X1, t2 - X2);
    if (del === 0) {
      errOut.innerHTML = 'Move the Δ slider to perturb <i>A</i> and watch ' +
        'how far the solution moves.';
    } else if (singular) {
      errOut.innerHTML = '<i>A</i> + Δ<i>A</i> is singular: the two rows ' +
        'coincide and there is no unique solution.';
    } else {
      var rA = ndA / nA, rx = ndx / nx;
      errOut.innerHTML =
        '‖Δ<i>A</i>‖ / ‖<i>A</i>‖ = <b>' + sig(rA, 3) + '</b><br>' +
        '‖Δ<i>x</i>‖ / ‖<i>x</i>‖ = <b>' + sig(rx, 3) + '</b><br>' +
        'ratio <b>' + fmtBig(rx / rA) + '</b>' +
        (rx / rA <= cond
          ? ' ≤ cond(A) = ' + fmtBig(cond)
          : ' > cond(A) = ' + fmtBig(cond) + '<br><small>The bound only ' +
            'holds to first order in Δ<i>A</i>; here Δ<i>A</i> is large ' +
            'enough to bring <i>A</i> + Δ<i>A</i> nearly to singular.</small>');
    }

    while (svg.firstChild) svg.removeChild(svg.firstChild);

    // Grid and ticks on whole numbers, labelled along the axes.
    for (var t = LO; t <= HI; t++) {
      svg.appendChild(el('line', { x1: px(t), y1: py(LO), x2: px(t),
                                   y2: py(HI), class: 'cn-tick' }));
      svg.appendChild(el('line', { x1: px(LO), y1: py(t), x2: px(HI),
                                   y2: py(t), class: 'cn-tick' }));
      if (t !== 0) {
        svg.appendChild(el('text', { x: px(t), y: py(0) + 15,
                                     'text-anchor': 'middle',
                                     class: 'cn-ticklbl' }, t));
        svg.appendChild(el('text', { x: px(0) - 6, y: py(t) + 4,
                                     'text-anchor': 'end',
                                     class: 'cn-ticklbl' }, t));
      }
    }

    // Axes through the origin, plus a frame so the clipping reads as a window.
    svg.appendChild(el('rect', { x: M, y: M, width: PW, height: PW,
                                 fill: 'none', stroke: '#bbb' }));
    svg.appendChild(el('line', { x1: px(LO), y1: py(0), x2: px(HI),
                                 y2: py(0), class: 'cn-axis' }));
    svg.appendChild(el('line', { x1: px(0), y1: py(LO), x2: px(0),
                                 y2: py(HI), class: 'cn-axis' }));
    svg.appendChild(el('text', { x: M + PW - 4, y: py(0) - 8, 'text-anchor': 'end',
                                 class: 'cn-axlbl' }, 'x₁'));
    svg.appendChild(el('text', { x: px(0) + 8, y: M + 14,
                                 class: 'cn-axlbl' }, 'x₂'));

    // The two rows of the system.
    drawLine(a11, a12, b1, 'cn-line1',
             { text: 'row 1', cls: 'cn-lbl1', left: true });
    drawLine(a21, a22, b2, 'cn-line2',
             { text: 'row 2', cls: 'cn-lbl2', left: false });

    if (del !== 0) drawLine(a21, a22 + del, b2, 'cn-line2p', null);

    // Solutions.  The perturbed one runs off the plot long before the exact
    // one does, which is the whole point; say so instead of drawing nothing.
    if (del !== 0 && !singular) {
      if (t1 >= LO && t1 <= HI && t2 >= LO && t2 <= HI) {
        svg.appendChild(el('circle', { cx: px(t1), cy: py(t2), r: 6,
                                       class: 'cn-dot-pert' }));
      } else {
        svg.appendChild(el('text', { x: M + PW / 2, y: M + PW - 12,
                                     'text-anchor': 'middle',
                                     class: 'cn-offnote' },
          'perturbed solution (' + sig(t1, 3) + ', ' + sig(t2, 3) +
          ') is off the plot'));
      }
    }
    svg.appendChild(el('circle', { cx: px(X1), cy: py(X2), r: 6,
                                   class: 'cn-dot-exact' }));
  }

  epsIn.addEventListener('input', render);
  dIn.addEventListener('input', render);
  render();
})();
</script>
