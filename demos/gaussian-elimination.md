<script>
(function () {
  var n = 6;
  var A = document.getElementById('ge-A'),
      bEl = document.getElementById('ge-b'),
      cIn = document.getElementById('ge-c'),
      rIn = document.getElementById('ge-r'),
      sentence = document.getElementById('ge-sentence'),
      code = document.getElementById('ge-code');

  var c = 1, r = 2;

  function build() {
    A.style.gridTemplateColumns = 'repeat(' + n + ', 2.2rem)';
    bEl.style.gridTemplateColumns = '2.2rem';
    for (var row = 1; row <= n; row++) {
      for (var col = 1; col <= n; col++) {
        var d = document.createElement('div');
        d.className = 'ge-cell';
        d.dataset.row = row;
        d.dataset.col = col;
        A.appendChild(d);
      }
      var v = document.createElement('div');
      v.className = 'ge-cell';
      v.dataset.row = row;
      bEl.appendChild(v);
    }
  }

  // Entries in column `col` below the diagonal that the algorithm has already
  // zeroed by the time we are at step (c, r).
  function isZeroed(row, col) {
    if (row <= col) return false;
    if (col < c) return true;
    return col === c && row < r;
  }

  function render() {
    rIn.min = c + 1;
    if (r < c + 1) r = c + 1;
    cIn.value = c; rIn.value = r;
    document.getElementById('ge-cval').textContent = c;
    document.getElementById('ge-rval').textContent = r;

    A.querySelectorAll('.ge-cell').forEach(function (d) {
      var row = +d.dataset.row, col = +d.dataset.col, cls = 'ge-cell', txt = '×';
      if (isZeroed(row, col)) { cls += ' zero'; txt = '0'; }
      else if (row === r && col === c) { cls += ' purple'; }
      else if (row === c && col >= c) { cls += ' green'; }
      else if (row === r && col > c) { cls += ' blue'; }
      d.className = cls;
      d.textContent = txt;
    });
    bEl.querySelectorAll('.ge-cell').forEach(function (d) {
      var row = +d.dataset.row, cls = 'ge-cell';
      if (row === c) cls += ' green';
      else if (row === r) cls += ' blue';
      d.className = cls;
      d.textContent = '×';
    });

    sentence.innerHTML =
      'Subtract <i>f</i> &times; <span class="g">row ' + c + '</span> from ' +
      '<span class="b">row ' + r + '</span> to eliminate ' +
      '<span class="p">a<sub>' + r + c + '</sub></span>, ' +
      'where <i>f</i> = a<sub>' + r + c + '</sub> / a<sub>' + c + c + '</sub>.';

    code.innerHTML =
      'for c = 1 &hellip; n-1          <span class="hl">c = ' + c + '</span>\n' +
      '    for r = c+1 &hellip; n      <span class="hl">r = ' + r + '</span>\n' +
      '        f = A[r,c] / A[c,c]\n' +
      '        A[r,c:n] -= f * A[c,c:n]\n' +
      '        b[r]     -= f * b[c]';
  }

  function stepBy(d) {
    var flat = [];
    for (var cc = 1; cc <= n - 1; cc++)
      for (var rr = cc + 1; rr <= n; rr++) flat.push([cc, rr]);
    var at = flat.findIndex(function (p) { return p[0] === c && p[1] === r; });
    at = Math.min(flat.length - 1, Math.max(0, at + d));
    c = flat[at][0]; r = flat[at][1];
    render();
  }

  cIn.addEventListener('input', function () { c = +cIn.value; if (r < c + 1) r = c + 1; render(); });
  rIn.addEventListener('input', function () { r = +rIn.value; render(); });
  document.getElementById('ge-prev').addEventListener('click', function () { stepBy(-1); });
  document.getElementById('ge-next').addEventListener('click', function () { stepBy(1); });

  build();
  render();
})();
</script>
{:/}

Every cell marked `0` was cleared by an earlier pass of the inner loop; the
grey cells are untouched so far. Note that the row operation only touches
columns `c` through `n` --- everything to the left is already zero, so there is
no reason to compute on it.

<script>
(function () {
  var n = 6;
  var A = document.getElementById('bs-A'),
      xEl = document.getElementById('bs-x'),
      bEl = document.getElementById('bs-b'),
      rIn = document.getElementById('bs-r'),
      sentence = document.getElementById('bs-sentence'),
      code = document.getElementById('bs-code');

  // The slider counts steps left-to-right; the loop index runs the other way.
  function rOf(step) { return n + 1 - step; }

  function build() {
    A.style.gridTemplateColumns = 'repeat(' + n + ', 2.2rem)';
    xEl.style.gridTemplateColumns = '2.2rem';
    bEl.style.gridTemplateColumns = '2.2rem';
    for (var row = 1; row <= n; row++) {
      for (var col = 1; col <= n; col++) {
        var d = document.createElement('div');
        d.className = 'ge-cell';
        d.dataset.row = row;
        d.dataset.col = col;
        A.appendChild(d);
      }
      [xEl, bEl].forEach(function (host) {
        var v = document.createElement('div');
        v.className = 'ge-cell';
        v.dataset.row = row;
        host.appendChild(v);
      });
    }
  }

  function render() {
    var r = rOf(+rIn.value);
    document.getElementById('bs-rval').textContent = r;

    A.querySelectorAll('.ge-cell').forEach(function (d) {
      var row = +d.dataset.row, col = +d.dataset.col, cls = 'ge-cell', txt = '×';
      if (row > col) { cls += ' zero'; txt = '0'; }
      else if (row === r && col === r) { cls += ' pivot'; }
      else if (row === r && col > r) { cls += ' blue'; }
      d.className = cls;
      d.textContent = txt;
    });
    xEl.querySelectorAll('.ge-cell').forEach(function (d) {
      var row = +d.dataset.row, cls = 'ge-cell', txt = '×';
      if (row === r) { cls += ' purple'; txt = '?'; }
      else if (row > r) { cls += ' green'; }        // already solved for
      else { cls += ' unknown'; txt = '?'; }        // still unknown
      d.className = cls;
      d.textContent = txt;
    });
    bEl.querySelectorAll('.ge-cell').forEach(function (d) {
      var row = +d.dataset.row, cls = 'ge-cell';
      if (row === r) cls += ' orange';
      d.className = cls;
      d.textContent = '×';
    });

    var tail = r < n
      ? '<span class="b">A<sub>' + r + ',' + (r + 1) + ':' + n + '</sub></span> &middot; ' +
        '<span class="g">x<sub>' + (r + 1) + ':' + n + '</sub></span> from '
      : 'nothing (there is no <span class="g">x</span> below it yet) from ';
    sentence.innerHTML =
      'Calculate <span class="p">x<sub>' + r + '</sub></span> by subtracting ' +
      tail + '<span class="o">b<sub>' + r + '</sub></span>, ' +
      'then dividing by a<sub>' + r + r + '</sub>.';

    code.innerHTML =
      'for r = n &hellip; 1                      <span class="hl">r = ' + r + '</span>\n' +
      '    x[r] = (b[r] - A[r,r+1:n] &middot; x[r+1:n]) / A[r,r]';
  }

  rIn.addEventListener('input', render);
  document.getElementById('bs-prev').addEventListener('click', function () {
    rIn.value = Math.max(+rIn.min, +rIn.value - 1); render();
  });
  document.getElementById('bs-next').addEventListener('click', function () {
    rIn.value = Math.min(+rIn.max, +rIn.value + 1); render();
  });

  build();
  render();
})();
</script>
{:/}

Each solved entry of <i>x</i> turns green and is used by every row above it,
so the work per row grows as the loop climbs.
