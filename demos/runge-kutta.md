  <p>Test problem: a damped pendulum,
  \[ \ddot{x} = -\sin x - 0.1\,\dot{x}, \qquad x(0) = 2.5,\quad \dot{x}(0) = 0, \]
  written as a first-order system in the state vector
  \(\vec{y} = [x,\ \dot{x}]^T\):
  \[ \dot{\vec{y}} = \vec{f}(t, \vec{y}) =
     \begin{bmatrix} y_2 \\ -\sin y_1 - 0.1\,y_2 \end{bmatrix}. \]
  Every method below applies the same update to the whole vector at once.
  There is no closed-form solution, so the "exact" curve is RK4 with
  \(h = 2.5\times10^{-4}\).</p>

  <div class="rk-formula" data-m="euler">
    \[ \vec{y}_{i+1} = \vec{y}_i + h\,\vec{k}_1, \qquad
       \vec{k}_1 = \vec{f}(t_i, \vec{y}_i) \]
    <p>One slope per step, taken at the start. First order: global error \(O(h)\).</p>
  </div>
  <div class="rk-formula" data-m="midpoint" hidden>
    \[ \vec{k}_1 = \vec{f}(t_i, \vec{y}_i), \qquad
       \vec{k}_2 = \vec{f}\!\left(t_i + \tfrac{h}{2},\ \vec{y}_i + \tfrac{h}{2}\vec{k}_1\right), \qquad
       \vec{y}_{i+1} = \vec{y}_i + h\,\vec{k}_2 \]
    <p>Use the Euler slope to reach the midpoint, then step with the slope there. Second order: global error \(O(h^2)\).</p>
  </div>
  <div class="rk-formula" data-m="ralston" hidden>
    \[ \vec{k}_1 = \vec{f}(t_i, \vec{y}_i), \qquad
       \vec{k}_2 = \vec{f}\!\left(t_i + \tfrac{2h}{3},\ \vec{y}_i + \tfrac{2h}{3}\vec{k}_1\right), \qquad
       \vec{y}_{i+1} = \vec{y}_i + h\left(\tfrac{1}{4}\vec{k}_1 + \tfrac{3}{4}\vec{k}_2\right) \]
    <p>Same cost as midpoint with a different sample point and weights chosen to minimize the truncation error bound. Second order: global error \(O(h^2)\).</p>
  </div>
  <div class="rk-formula" data-m="rk4" hidden>
    \[ \begin{aligned}
       \vec{k}_1 &= \vec{f}(t_i, \vec{y}_i) &
       \vec{k}_2 &= \vec{f}\!\left(t_i + \tfrac{h}{2},\ \vec{y}_i + \tfrac{h}{2}\vec{k}_1\right) \\
       \vec{k}_3 &= \vec{f}\!\left(t_i + \tfrac{h}{2},\ \vec{y}_i + \tfrac{h}{2}\vec{k}_2\right) &
       \vec{k}_4 &= \vec{f}\!\left(t_i + h,\ \vec{y}_i + h\,\vec{k}_3\right)
       \end{aligned} \]
    \[ \vec{y}_{i+1} = \vec{y}_i + \tfrac{h}{6}\left(\vec{k}_1 + 2\vec{k}_2 + 2\vec{k}_3 + \vec{k}_4\right) \]
    <p>Four slope evaluations per step. Fourth order: global error \(O(h^4)\).</p>
  </div>

<script>
(function () {
  'use strict';

  // ---- problem --------------------------------------------------------
  var C = 0.1, Y0 = [2.5, 0], TF = 10;
  function f(t, y) { return [y[1], -Math.sin(y[0]) - C * y[1]]; }
  function add(y, h, k) { return [y[0] + h * k[0], y[1] + h * k[1]]; }

  var METHODS = {
    euler:    { name: 'Euler',          order: 1, color: '#1e90ff',
      step: function (t, y, h) { return add(y, h, f(t, y)); } },
    midpoint: { name: 'Midpoint',       order: 2, color: '#e07b00',
      step: function (t, y, h) {
        var k1 = f(t, y), k2 = f(t + h / 2, add(y, h / 2, k1));
        return add(y, h, k2); } },
    ralston:  { name: 'Ralston',        order: 2, color: '#2e8b57',
      step: function (t, y, h) {
        var k1 = f(t, y), k2 = f(t + 2 * h / 3, add(y, 2 * h / 3, k1));
        return [y[0] + h * (k1[0] / 4 + 3 * k2[0] / 4), y[1] + h * (k1[1] / 4 + 3 * k2[1] / 4)]; } },
    rk4:      { name: 'Classical RK4',  order: 4, color: '#8a2be2',
      step: function (t, y, h) {
        var k1 = f(t, y), k2 = f(t + h / 2, add(y, h / 2, k1)),
            k3 = f(t + h / 2, add(y, h / 2, k2)), k4 = f(t + h, add(y, h, k3));
        return [y[0] + h / 6 * (k1[0] + 2 * k2[0] + 2 * k3[0] + k4[0]),
                y[1] + h / 6 * (k1[1] + 2 * k2[1] + 2 * k3[1] + k4[1])]; } }
  };
  var ORDER = ['euler', 'midpoint', 'ralston', 'rk4'];
  var HS = [1, 0.5, 0.25, 0.1, 0.05, 0.025, 0.01, 0.005];

  // integrate, returning arrays of t, x, v
  function integrate(m, h) {
    var N = Math.round(TF / h), t = new Float64Array(N + 1),
        x = new Float64Array(N + 1), v = new Float64Array(N + 1), y = Y0;
    x[0] = y[0]; v[0] = y[1];
    for (var i = 0; i < N; i++) {
      y = METHODS[m].step(t[i], y, h);
      t[i + 1] = t[i] + h; x[i + 1] = y[0]; v[i + 1] = y[1];
    }
    return { t: t, x: x, v: v };
  }

  // reference solution and per-method global errors, computed once
  var REF = integrate('rk4', 2.5e-4);
  var REFEND = [REF.x[REF.x.length - 1], REF.v[REF.v.length - 1]];
  var SOL = {}, ERR = {};
  ORDER.forEach(function (m) {
    SOL[m] = HS.map(function (h) { return integrate(m, h); });
    ERR[m] = SOL[m].map(function (s) {
      return Math.hypot(s.x[s.x.length - 1] - REFEND[0], s.v[s.v.length - 1] - REFEND[1]);
    });
  });
  function fitSlope(m) {          // least-squares slope on log-log over h <= 0.1
    var xs = [], ys = [];
    HS.forEach(function (h, i) { if (h <= 0.1) { xs.push(Math.log10(h)); ys.push(Math.log10(ERR[m][i])); } });
    var n = xs.length, mx = 0, my = 0, num = 0, den = 0, i;
    for (i = 0; i < n; i++) { mx += xs[i] / n; my += ys[i] / n; }
    for (i = 0; i < n; i++) { num += (xs[i] - mx) * (ys[i] - my); den += (xs[i] - mx) * (xs[i] - mx); }
    return num / den;
  }

  // ---- svg helpers ----------------------------------------------------
  var NS = 'http://www.w3.org/2000/svg';
  function el(name, attrs, text) {
    var e = document.createElementNS(NS, name);
    for (var k in attrs) e.setAttribute(k, attrs[k]);
    if (text != null) e.textContent = text;
    return e;
  }
  function clear(svg) { while (svg.firstChild) svg.removeChild(svg.firstChild); }
  function fmtH(h) { return h >= 0.1 ? h.toFixed(2) : h.toFixed(3); }
  // append "m × 10^k" to an svg text element, with a real superscript
  function appendE(t, e) {
    var parts = e.toExponential(2).split('e');
    t.appendChild(el('tspan', {}, parts[0] + ' × 10'));
    t.appendChild(el('tspan', { 'baseline-shift': 'super', 'font-size': '9px' }, String(+parts[1])));
  }
  function pow10Text(x, y, k, anchor) {
    var t = el('text', { x: x, y: y, 'text-anchor': anchor || 'end' }, '10');
    t.appendChild(el('tspan', { 'baseline-shift': 'super', 'font-size': '9px' }, String(k)));
    return t;
  }

  // ---- time-series panel ----------------------------------------------
  function drawSeries(svg, m, hi, comp, label) {
    clear(svg);
    var W = 480, H = 300, ML = 42, MR = 12, MT = 28, MB = 36, PW = W - ML - MR, PH = H - MT - MB;
    var ymin = -3, ymax = 3;
    var px = function (t) { return ML + PW * t / TF; };
    var py = function (y) { return MT + PH * (ymax - y) / (ymax - ymin); };
    var s = SOL[m][hi], color = METHODS[m].color;
    var id = svg.id + '-clip';
    var defs = el('defs'); var cp = el('clipPath', { id: id });
    cp.appendChild(el('rect', { x: ML, y: MT, width: PW, height: PH })); defs.appendChild(cp);
    svg.appendChild(defs);

    var yt;
    for (yt = ymin; yt <= ymax; yt += 1) {
      svg.appendChild(el('line', { x1: ML, y1: py(yt), x2: ML + PW, y2: py(yt), 'class': yt === 0 ? 'axis' : 'grid' }));
      svg.appendChild(el('text', { x: ML - 6, y: py(yt) + 4, 'text-anchor': 'end' }, yt));
    }
    for (var tt = 0; tt <= TF; tt += 2) {
      svg.appendChild(el('line', { x1: px(tt), y1: MT, x2: px(tt), y2: MT + PH, 'class': 'grid' }));
      svg.appendChild(el('text', { x: px(tt), y: MT + PH + 16, 'text-anchor': 'middle' }, tt));
    }
    svg.appendChild(el('rect', { x: ML, y: MT, width: PW, height: PH, 'class': 'axis' }));
    svg.appendChild(el('text', { x: ML + PW / 2, y: H - 6, 'text-anchor': 'middle' }, 't'));
    svg.appendChild(el('text', { x: ML + 6, y: MT + 16, 'class': 'ttl' }, label));

    var g = el('g', { 'clip-path': 'url(#' + id + ')' });
    // reference
    var d = '', n = REF.t.length, stride = 20;
    for (var i = 0; i < n; i += stride) d += (i ? 'L' : 'M') + px(REF.t[i]).toFixed(1) + ',' + py(REF[comp][i]).toFixed(1);
    g.appendChild(el('path', { d: d, fill: 'none', stroke: '#000', 'stroke-width': 2 }));
    // numerical
    d = '';
    for (i = 0; i < s.t.length; i++) d += (i ? 'L' : 'M') + px(s.t[i]).toFixed(1) + ',' + py(s[comp][i]).toFixed(1);
    g.appendChild(el('path', { d: d, fill: 'none', stroke: color, 'stroke-width': 1.5 }));
    if (s.t.length <= 201) for (i = 0; i < s.t.length; i++)
      g.appendChild(el('circle', { cx: px(s.t[i]), cy: py(s[comp][i]), r: 2.6, fill: color }));
    // error bar at t_f
    var yN = s[comp][s.t.length - 1], yE = REF[comp][n - 1];
    g.appendChild(el('line', { x1: px(TF), y1: py(yE), x2: px(TF), y2: py(yN), stroke: 'red', 'stroke-width': 4 }));
    svg.appendChild(g);
    svg.appendChild(el('text', { x: ML + PW - 6, y: MT + 16, 'text-anchor': 'end' },
      'error = ' + Math.abs(yN - yE).toPrecision(3)));
  }

  // ---- log-log error panel --------------------------------------------
  function drawErr(svg, m, hi) {
    clear(svg);
    var W = 640, H = 400, ML = 60, MR = 16, MT = 28, MB = 40, PW = W - ML - MR, PH = H - MT - MB;
    var lxmin = Math.log10(HS[HS.length - 1]) - 0.15, lxmax = 0.15;
    var all = []; ORDER.forEach(function (k) { all = all.concat(ERR[k]); });
    var lymin = Math.floor(Math.log10(Math.min.apply(null, all))), lymax = Math.ceil(Math.log10(Math.max.apply(null, all)));
    var px = function (h) { return ML + PW * (Math.log10(h) - lxmin) / (lxmax - lxmin); };
    var py = function (e) { return MT + PH * (lymax - Math.log10(e)) / (lymax - lymin); };

    var k, j;
    for (k = lymin; k <= lymax; k++) {
      svg.appendChild(el('line', { x1: ML, y1: py(Math.pow(10, k)), x2: ML + PW, y2: py(Math.pow(10, k)), 'class': 'grid' }));
      svg.appendChild(pow10Text(ML - 8, py(Math.pow(10, k)) + 4, k));
      if (k < lymax) for (j = 2; j < 10; j++)
        svg.appendChild(el('line', { x1: ML, y1: py(j * Math.pow(10, k)), x2: ML + PW, y2: py(j * Math.pow(10, k)), 'class': 'minor' }));
    }
    for (k = -3; k <= 0; k++) {
      var h0 = Math.pow(10, k);
      if (Math.log10(h0) >= lxmin) {
        svg.appendChild(el('line', { x1: px(h0), y1: MT, x2: px(h0), y2: MT + PH, 'class': 'grid' }));
        svg.appendChild(pow10Text(px(h0) + 8, MT + PH + 17, k));
      }
      for (j = 2; j < 10; j++) if (Math.log10(j * h0) > lxmin && Math.log10(j * h0) < lxmax)
        svg.appendChild(el('line', { x1: px(j * h0), y1: MT, x2: px(j * h0), y2: MT + PH, 'class': 'minor' }));
    }
    svg.appendChild(el('rect', { x: ML, y: MT, width: PW, height: PH, 'class': 'axis' }));
    svg.appendChild(el('text', { x: ML + PW / 2, y: H - 8, 'text-anchor': 'middle' }, 'h'));
    var yl = el('text', { x: 14, y: MT + PH / 2, 'text-anchor': 'middle', transform: 'rotate(-90 14 ' + (MT + PH / 2) + ')' }, 'global error');
    svg.appendChild(yl);

    function line(key, bold) {
      var d = '';
      HS.forEach(function (h, i) { d += (i ? 'L' : 'M') + px(h).toFixed(1) + ',' + py(ERR[key][i]).toFixed(1); });
      var color = bold ? METHODS[key].color : '#bbb';
      svg.appendChild(el('path', { d: d, fill: 'none', stroke: color, 'stroke-width': bold ? 2.5 : 1.5 }));
      HS.forEach(function (h, i) {
        svg.appendChild(el('circle', { cx: px(h), cy: py(ERR[key][i]), r: bold ? 3.5 : 2.5, fill: color }));
      });
    }
    ORDER.forEach(function (key) { if (key !== m) line(key, false); });
    line(m, true);
    svg.appendChild(el('circle', { cx: px(HS[hi]), cy: py(ERR[m][hi]), r: 7, fill: 'none', stroke: 'red', 'stroke-width': 2.5 }));

    // legend (bottom right, the empty corner since every line rises to the right)
    var lx = ML + PW - 160, ly = MT + PH - 16 * ORDER.length - 10;
    ORDER.forEach(function (key, i) {
      var bold = key === m, color = bold ? METHODS[key].color : '#bbb', yy = ly + 16 * i + 6;
      svg.appendChild(el('line', { x1: lx, y1: yy, x2: lx + 22, y2: yy, stroke: color, 'stroke-width': bold ? 2.5 : 1.5 }));
      svg.appendChild(el('circle', { cx: lx + 11, cy: yy, r: 3, fill: color }));
      var t = el('text', { x: lx + 28, y: yy + 4, fill: bold ? '#000' : '#999', 'font-weight': bold ? 'bold' : 'normal' },
        METHODS[key].name + ' (slope ' + METHODS[key].order + ')');
      svg.appendChild(t);
    });
    var ttl = el('text', { x: ML + PW / 2, y: MT - 10, 'text-anchor': 'middle', 'class': 'ttl', fill: METHODS[m].color });
    ttl.appendChild(el('tspan', {}, METHODS[m].name + ':  h = ' + fmtH(HS[hi]) + ',  error = '));
    appendE(ttl, ERR[m][hi]);
    ttl.appendChild(el('tspan', {}, ',  fitted slope = ' + fitSlope(m).toFixed(2)));
    svg.appendChild(ttl);
  }

  // ---- wiring ---------------------------------------------------------
  var sel = document.getElementById('rk-method'), rng = document.getElementById('rk-h'),
      hval = document.getElementById('rk-hval');
  function update() {
    var m = sel.value, hi = +rng.value;
    hval.textContent = fmtH(HS[hi]);
    document.querySelectorAll('.rk-formula').forEach(function (d) { d.hidden = d.dataset.m !== m; });
    drawSeries(document.getElementById('rk-x'), m, hi, 'x', 'x(t)');
    drawSeries(document.getElementById('rk-v'), m, hi, 'v', 'ẋ(t)');
    drawErr(document.getElementById('rk-err'), m, hi);
  }
  sel.addEventListener('change', update);
  rng.addEventListener('input', update);
  update();
})();
</script>
