  <p>The function from Chapra's chapter on one-dimensional optimization,
  negated so that we are minimizing:
  \[ f(x) = \frac{x^2}{10} - 2\sin x, \qquad
     f'(x) = \frac{x}{5} - 2\cos x, \qquad
     f''(x) = \frac{1}{5} + 2\sin x. \]
  It has a minimum near \(x \approx 1.43\) and a maximum near
  \(x \approx 5.27\). At either one, \(f'(x) = 0\). So one way to find the
  minimum is to hand \(f'\) to a root finder from earlier in the course.
  Bisection and Newton-Raphson below do exactly that: they only ever look at
  the <em>bottom</em> plot. Golden-section search is different: it only
  evaluates \(f\) itself, never the derivative.</p>

  <p>Bisection and Newton converge to <em>a</em> point where \(f' = 0\), but
  nothing in them cares whether that point is a maximum or a minimum. Drag
  the Newton start to the right and it lands on the maximum instead. The
  sign of \(f''\) tells the two apart. Golden-section search keeps the
  bracket around the smaller of its two interior values, so it only ever
  closes in on a minimum (negate \(f\) to find a maximum).</p>

(function () {
  var NS = 'http://www.w3.org/2000/svg';
  function el(name, attrs, text) {
    var e = document.createElementNS(NS, name);
    for (var a in attrs) e.setAttribute(a, attrs[a]);
    if (text !== undefined) e.textContent = text;
    return e;
  }
  function $(id) { return document.getElementById(id); }
  function clear(svg) { while (svg.firstChild) svg.removeChild(svg.firstChild); }
  function fmt(v, d) { return isFinite(v) ? v.toFixed(d) : 'NaN'; }

  // ---- the 1-D problem -----------------------------------------------------
  function f(x)   { return x * x / 10 - 2 * Math.sin(x); }
  function fp(x)  { return x / 5 - 2 * Math.cos(x); }
  function fpp(x) { return 0.2 + 2 * Math.sin(x); }
  var XMIN = 0, XMAX = 6, XL0 = 0, XU0 = 4, NIT = 10;

  // ---- generic axes (same scheme as the other demos) -----------------------
  function axes(svg, o) {
    var W = +svg.viewBox.baseVal.width, H = +svg.viewBox.baseVal.height;
    var ML = 50, MR = 14, MT = o.title ? 34 : 12, MB = 36;
    var PW = W - ML - MR, PH = H - MT - MB;
    function px(x) { return ML + (x - o.xmin) / (o.xmax - o.xmin) * PW; }
    function py(y) { return MT + PH - (y - o.ymin) / (o.ymax - o.ymin) * PH; }
    function ticks(step, lo, hi) {
      var d = Math.max(0, -Math.floor(Math.log10(step) + 1e-9)), t = [];
      for (var v = Math.ceil(lo / step - 1e-9) * step; v <= hi + 1e-9; v += step)
        t.push([v, (Math.abs(v) < 1e-12 ? 0 : v).toFixed(d)]);
      return t;
    }
    clear(svg);
    ticks(o.ystep, o.ymin, o.ymax).forEach(function (t) {
      svg.appendChild(el('line', { x1: ML, y1: py(t[0]), x2: ML + PW, y2: py(t[0]),
                                   class: t[0] === 0 ? 'op-zero' : 'op-tick' }));
      svg.appendChild(el('text', { x: ML - 8, y: py(t[0]) + 4, 'text-anchor': 'end', class: 'op-ticklbl' }, t[1]));
    });
    ticks(o.xstep, o.xmin, o.xmax).forEach(function (t) {
      svg.appendChild(el('line', { x1: px(t[0]), y1: MT, x2: px(t[0]), y2: MT + PH, class: 'op-tick' }));
      svg.appendChild(el('text', { x: px(t[0]), y: MT + PH + 16, 'text-anchor': 'middle', class: 'op-ticklbl' }, t[1]));
    });
    var id = svg.id + '-clip', defs = el('defs'), cp = el('clipPath', { id: id });
    cp.appendChild(el('rect', { x: ML, y: MT, width: PW, height: PH }));
    defs.appendChild(cp); svg.appendChild(defs);
    var g = el('g', { 'clip-path': 'url(#' + id + ')' });
    svg.appendChild(g);
    svg.appendChild(el('rect', { x: ML, y: MT, width: PW, height: PH, class: 'op-axis' }));
    svg.appendChild(el('text', { x: ML + PW / 2, y: H - 4, 'text-anchor': 'middle', class: 'op-axlbl' }, 'x'));
    svg.appendChild(el('text', { x: 16, y: MT + PH / 2 + 7, 'text-anchor': 'middle', class: 'op-axlbl ' + (o.cls || '') }, o.ylabel));
    if (o.title) svg.appendChild(el('text', { x: ML, y: 22, class: 'op-title ' + (o.cls || '') }, o.title));
    return { px: px, py: py, g: g, ML: ML, MT: MT, PW: PW, PH: PH };
  }
  function pathD(fn, a, x0, x1, n) {
    var d = '';
    for (var i = 0; i <= n; i++) {
      var x = x0 + (x1 - x0) * i / n, y = fn(x);
      d += (i ? ' L' : 'M') + a.px(x).toFixed(2) + ',' + a.py(y).toFixed(2);
    }
    return d;
  }
  function dot(a, x, y, cls, r) { return el('circle', { cx: a.px(x), cy: a.py(y), r: r || 5, class: cls }); }
  function vline(a, x, cls) {
    return el('line', { x1: a.px(x), y1: a.MT, x2: a.px(x), y2: a.MT + a.PH, class: cls });
  }
  function band(a, x0, x1, cls) {
    return el('rect', { x: a.px(Math.min(x0, x1)), y: a.MT,
                        width: Math.abs(a.px(x1) - a.px(x0)), height: a.PH, class: cls });
  }
  function label(a, x, y, txt, cls, dx, dy) {
    return el('text', { x: a.px(x) + (dx || 0), y: a.py(y) + (dy || 0),
                        'text-anchor': 'middle', class: 'op-lbl ' + (cls || '') }, txt);
  }

  // ---- iteration histories -------------------------------------------------
  // Each returns an array of states; state k is what the k-th iteration looks
  // at, and `est` is the estimate it produces.
  function bisection() {
    var xl = XL0, xu = XU0, prev = NaN, s = [];
    for (var k = 0; k <= NIT; k++) {
      var xr = (xl + xu) / 2, st = { xl: xl, xu: xu, xr: xr, est: xr };
      st.ea = k ? Math.abs((xr - prev) / xr) * 100 : NaN;
      st.keep = fp(xl) * fp(xr) < 0 ? 'left' : 'right';
      s.push(st);
      if (st.keep === 'left') xu = xr; else xl = xr;
      prev = xr;
    }
    return s;
  }
  function newton(x0) {
    var x = x0, s = [];
    for (var k = 0; k <= NIT; k++) {
      var xn = x - fp(x) / fpp(x);
      s.push({ x: x, xn: xn, est: x, ea: k ? Math.abs((x - s[k - 1].x) / x) * 100 : NaN });
      x = xn;
      if (!isFinite(x)) break;
    }
    return s;
  }
  function golden() {
    var R = (Math.sqrt(5) - 1) / 2, xl = XL0, xu = XU0, s = [];
    var d = R * (xu - xl), x1 = xl + d, x2 = xu - d, f1 = f(x1), f2 = f(x2);
    for (var k = 0; k <= NIT; k++) {
      var st = { xl: xl, xu: xu, x1: x1, x2: x2, f1: f1, f2: f2, keep: f1 < f2 ? 'right' : 'left' };
      st.est = f1 < f2 ? x1 : x2;
      st.ea = (1 - R) * Math.abs((xu - xl) / st.est) * 100;
      s.push(st);
      d = R * d;
      if (f1 < f2) { xl = x2; x2 = x1; f2 = f1; x1 = xl + d; f1 = f(x1); }
      else         { xu = x1; x1 = x2; f1 = f2; x2 = xu - d; f2 = f(x2); }
    }
    return s;
  }

  // ---- drawing -------------------------------------------------------------
  var fsvg = $('op-f-svg'), fpsvg = $('op-fp-svg');
  function method() { return document.querySelector('input[name=op-method]:checked').value; }

  function draw() {
    var m = method(), k = +$('op-it').value, x0 = +$('op-x0').value;
    $('op-itv').textContent = k;
    $('op-x0v').textContent = fmt(x0, 2);
    $('op-x0-lbl').classList.toggle('op-off', m !== 'newton');
    var hist = m === 'bisection' ? bisection() : m === 'newton' ? newton(x0) : golden();
    if (k >= hist.length) k = hist.length - 1;
    var st = hist[k];

    var A = axes(fsvg,  { xmin: XMIN, xmax: XMAX, ymin: -2.5, ymax: 5, xstep: 1, ystep: 1, ylabel: 'f', cls: 'op-col-f',
                               title: 'f(x) = x²/10 − 2 sin x' });
    var B = axes(fpsvg, { xmin: XMIN, xmax: XMAX, ymin: -2.5, ymax: 3, xstep: 1, ystep: 1, ylabel: 'f′', cls: 'op-col-fp',
                               title: 'f′(x) = x/5 − 2 cos x' });

    if (m === 'bisection' || m === 'golden') {
      [A, B].forEach(function (a) {
        var lo = st.keep === 'left' ? st.xl : (m === 'bisection' ? st.xr : st.x2);
        var hi = st.keep === 'left' ? (m === 'bisection' ? st.xr : st.x1) : st.xu;
        a.g.appendChild(band(a, st.xl, st.xu, 'op-bracket'));
        a.g.appendChild(band(a, lo, hi, 'op-keep'));
        a.g.appendChild(vline(a, st.xl, 'op-bracket-edge'));
        a.g.appendChild(vline(a, st.xu, 'op-bracket-edge'));
      });
      A.g.appendChild(label(A, st.xl, 5, 'xₗ', 'op-lbl-amb', 10, 16));
      A.g.appendChild(label(A, st.xu, 5, 'xᵤ', 'op-lbl-amb', -10, 16));
    }
    A.g.appendChild(el('path', { d: pathD(f, A, XMIN, XMAX, 240), class: 'op-curve' }));
    B.g.appendChild(el('path', { d: pathD(fp, B, XMIN, XMAX, 240),
                                 class: 'op-curve op-curve-fp' + (m === 'golden' ? ' op-curve-dim' : '') }));

    var note = '';
    if (m === 'bisection') {
      var xr = st.xr;
      [st.xl, st.xu].forEach(function (x) {
        A.g.appendChild(dot(A, x, f(x), 'op-pt'));
        B.g.appendChild(dot(B, x, fp(x), 'op-pt'));
      });
      A.g.appendChild(vline(A, xr, 'op-drop-line'));
      B.g.appendChild(vline(B, xr, 'op-drop-line'));
      A.g.appendChild(dot(A, xr, f(xr), 'op-pt-new', 6));
      B.g.appendChild(dot(B, xr, fp(xr), 'op-pt-new', 6));
      B.g.appendChild(label(B, xr, fp(xr), 'xᵣ', 'op-lbl-pur', 0, fp(xr) > 0 ? -12 : 22));
      note = 'Iteration ' + k + ': midpoint <b>x<sub>r</sub> = ' + fmt(xr, 4) + '</b>, ' +
             'f′(x<sub>l</sub>)·f′(x<sub>r</sub>) ' + (st.keep === 'left' ? '&lt; 0' : '&gt; 0') +
             ', so the root of f′ is in the <b>' + st.keep + '</b> half (green). ' +
             (k ? 'ε<sub>a</sub> = ' + fmt(st.ea, 3) + '%.' : '');
    } else if (m === 'newton') {
      var x = st.x, xn = st.xn;
      // tangent line of f' at x, on the f' plot
      var t0 = XMIN - 1, t1 = XMAX + 1;
      B.g.appendChild(el('line', { x1: B.px(t0), y1: B.py(fp(x) + fpp(x) * (t0 - x)),
                                   x2: B.px(t1), y2: B.py(fp(x) + fpp(x) * (t1 - x)), class: 'op-tangent' }));
      // the matching quadratic model of f, on the f plot
      A.g.appendChild(el('path', { class: 'op-model',
        d: pathD(function (t) { return f(x) + fp(x) * (t - x) + 0.5 * fpp(x) * (t - x) * (t - x); }, A, XMIN, XMAX, 160) }));
      A.g.appendChild(vline(A, x, 'op-drop-line'));
      B.g.appendChild(vline(B, x, 'op-drop-line'));
      A.g.appendChild(dot(A, x, f(x), 'op-pt', 6));
      B.g.appendChild(dot(B, x, fp(x), 'op-pt', 6));
      B.g.appendChild(label(B, x, fp(x), 'xᵢ', 'op-lbl-amb', 0, fp(x) > 0 ? -12 : 22));
      if (isFinite(xn)) {
        B.g.appendChild(dot(B, xn, 0, 'op-pt-open', 6));
        B.g.appendChild(label(B, xn, 0, 'xᵢ₊₁', 'op-lbl-pur', 0, fp(x) > 0 ? 24 : -14));
        A.g.appendChild(dot(A, xn, f(x) + fp(x) * (xn - x) + 0.5 * fpp(x) * (xn - x) * (xn - x), 'op-pt-open', 6));
      }
      var kind = fpp(x) > 0 ? '<span class="op-good">f″ &gt; 0, heading for a minimum</span>'
                            : '<span class="op-bad">f″ &lt; 0, heading for a maximum</span>';
      note = 'Iteration ' + k + ': at <b>x<sub>i</sub> = ' + fmt(x, 4) + '</b>, ' +
             'f′ = ' + fmt(fp(x), 3) + ', f″ = ' + fmt(fpp(x), 3) + '. ' +
             'The tangent to f′ hits zero at <b>x<sub>i+1</sub> = ' + fmt(xn, 4) + '</b>; ' +
             'on the top plot that is the vertex of the dashed parabola. ' + kind + '. ' +
             (k ? 'ε<sub>a</sub> = ' + fmt(st.ea, 3) + '%.' : '');
    } else {
      [st.x1, st.x2].forEach(function (x, i) {
        A.g.appendChild(vline(A, x, 'op-drop-line'));
        A.g.appendChild(dot(A, x, f(x), 'op-pt-new', 6));
        A.g.appendChild(label(A, x, f(x), i ? 'x₂' : 'x₁', 'op-lbl-pur', 0, -12));
      });
      [st.xl, st.xu].forEach(function (x) { A.g.appendChild(dot(A, x, f(x), 'op-pt')); });
      fpsvg.appendChild(el('text', { x: B.ML + B.PW / 2, y: B.MT + 24, 'text-anchor': 'middle', class: 'op-banner' },
                           'golden-section search never evaluates f′'));
      note = 'Iteration ' + k + ': bracket [' + fmt(st.xl, 3) + ', ' + fmt(st.xu, 3) + '], ' +
             'f(x<sub>1</sub>) = ' + fmt(st.f1, 4) + ', f(x<sub>2</sub>) = ' + fmt(st.f2, 4) + '. ' +
             (st.keep === 'right' ? 'f(x<sub>1</sub>) is smaller, so drop the left end: x<sub>l</sub> ← x<sub>2</sub>'
                                  : 'f(x<sub>2</sub>) is smaller, so drop the right end: x<sub>u</sub> ← x<sub>1</sub>') +
             ' (green is kept). Estimate <b>' + fmt(st.est, 4) + '</b>, ε<sub>a</sub> = ' + fmt(st.ea, 3) + '%.';
    }
    $('op-note').innerHTML = note;

    // table
    var tbl = $('op-table'); tbl.innerHTML = '';
    var hdr = m === 'bisection' ? ['k', 'xl', 'xu', 'xr', 'f′(xr)', 'εa %']
            : m === 'newton'    ? ['k', 'xi', 'f′(xi)', 'f″(xi)', 'xi+1', 'εa %']
                                : ['k', 'xl', 'xu', 'x1', 'x2', 'εa %'];
