  <h2>1. Finding a minimum in one dimension</h2>

  <p>The function from Chapra's chapter on one-dimensional optimization,
  negated so that we are minimizing:
  \[ f(x) = \frac{x^2}{10} - 2\sin x, \qquad
     f'(x) = \frac{x}{5} - 2\cos x, \qquad
     f''(x) = \frac{1}{5} + 2\sin x. \]
  It has a minimum near \(x \approx 1.43\) and a maximum near
  \(x \approx 5.27\). At either one, \(f'(x) = 0\). So one way to find the
  minimum is to hand \(f'\) to a root finder from earlier in the course.
  Bisection and Newton-Raphson below do exactly that: they only ever look at
  the <em>bottom</em> plot. Golden-section search is different: it only
  evaluates \(f\) itself, never the derivative.</p>

  <p>Bisection and Newton converge to <em>a</em> point where \(f' = 0\), but
  nothing in them cares whether that point is a maximum or a minimum. Drag
  the Newton start to the right and it lands on the maximum instead. The
  sign of \(f''\) tells the two apart. Golden-section search keeps the
  bracket around the smaller of its two interior values, so it only ever
  closes in on a minimum (negate \(f\) to find a maximum).</p>

  <h2>2. Two dimensions: what the Hessian looks like</h2>

  <p>Near a point where the gradient is zero, every smooth function looks
  like a quadratic built from its Hessian \(H\), the matrix of second
  derivatives:
  \[ f(x, y) \approx \tfrac{1}{2}
     \begin{bmatrix} x & y \end{bmatrix} H
     \begin{bmatrix} x \\ y \end{bmatrix}, \qquad
     H = \begin{bmatrix} \dfrac{\partial^2 f}{\partial x^2} & \dfrac{\partial^2 f}{\partial x\,\partial y} \\[1.1em]
         \dfrac{\partial^2 f}{\partial x\,\partial y} & \dfrac{\partial^2 f}{\partial y^2} \end{bmatrix}. \]
  The quadratic below is set up through the eigenvalues of \(H\), which are
  the curvatures along its two principal directions, and the angle of the
  first principal direction. The signs of the eigenvalues decide what kind
  of point the origin is.</p>

  <p>The dashed lines on the contour plot are the eigenvectors of \(H\).
  Along each one the function is a parabola with curvature equal to that
  eigenvalue. Both positive: a bowl, the origin is a minimum, \(H\) is
  positive definite. Both negative: a dome, a maximum, negative definite.
  One of each: a saddle, which is a maximum along one line and a minimum
  along the other, and \(H\) is indefinite. Without computing eigenvalues,
  the same test is
  \[ \det H = \frac{\partial^2 f}{\partial x^2}\,\frac{\partial^2 f}{\partial y^2}
     - \left(\frac{\partial^2 f}{\partial x\,\partial y}\right)^2 : \]
  a saddle has \(\det H < 0\), and when \(\det H > 0\) the sign of
  \(\partial^2 f / \partial x^2\) says
  whether it is a minimum or a maximum. The surface can be dragged to
  rotate it.</p>

(function () {
  var NS = 'http://www.w3.org/2000/svg';
  function el(name, attrs, text) {
    var e = document.createElementNS(NS, name);
    for (var a in attrs) e.setAttribute(a, attrs[a]);
    if (text !== undefined) e.textContent = text;
    return e;
  }
  function $(id) { return document.getElementById(id); }
  function clear(svg) { while (svg.firstChild) svg.removeChild(svg.firstChild); }
  function fmt(v, d) { return isFinite(v) ? v.toFixed(d) : 'NaN'; }

  // ---- the 1-D problem -----------------------------------------------------
  function f(x)   { return x * x / 10 - 2 * Math.sin(x); }
  function fp(x)  { return x / 5 - 2 * Math.cos(x); }
  function fpp(x) { return 0.2 + 2 * Math.sin(x); }
  var XMIN = 0, XMAX = 6, XL0 = 0, XU0 = 4, NIT = 10;

  // ---- generic axes (same scheme as the other demos) -----------------------
  function axes(svg, o) {
    var W = +svg.viewBox.baseVal.width, H = +svg.viewBox.baseVal.height;
    var ML = 50, MR = 14, MT = 12, MB = 36;
    var PW = W - ML - MR, PH = H - MT - MB;
    function px(x) { return ML + (x - o.xmin) / (o.xmax - o.xmin) * PW; }
    function py(y) { return MT + PH - (y - o.ymin) / (o.ymax - o.ymin) * PH; }
    function ticks(step, lo, hi) {
      var d = Math.max(0, -Math.floor(Math.log10(step) + 1e-9)), t = [];
      for (var v = Math.ceil(lo / step - 1e-9) * step; v <= hi + 1e-9; v += step)
        t.push([v, (Math.abs(v) < 1e-12 ? 0 : v).toFixed(d)]);
      return t;
    }
    clear(svg);
    ticks(o.ystep, o.ymin, o.ymax).forEach(function (t) {
      svg.appendChild(el('line', { x1: ML, y1: py(t[0]), x2: ML + PW, y2: py(t[0]),
                                   class: t[0] === 0 ? 'op-zero' : 'op-tick' }));
      svg.appendChild(el('text', { x: ML - 8, y: py(t[0]) + 4, 'text-anchor': 'end', class: 'op-ticklbl' }, t[1]));
    });
    ticks(o.xstep, o.xmin, o.xmax).forEach(function (t) {
      svg.appendChild(el('line', { x1: px(t[0]), y1: MT, x2: px(t[0]), y2: MT + PH, class: 'op-tick' }));
      svg.appendChild(el('text', { x: px(t[0]), y: MT + PH + 16, 'text-anchor': 'middle', class: 'op-ticklbl' }, t[1]));
    });
    var id = svg.id + '-clip', defs = el('defs'), cp = el('clipPath', { id: id });
    cp.appendChild(el('rect', { x: ML, y: MT, width: PW, height: PH }));
    defs.appendChild(cp); svg.appendChild(defs);
    var g = el('g', { 'clip-path': 'url(#' + id + ')' });
    svg.appendChild(g);
    svg.appendChild(el('rect', { x: ML, y: MT, width: PW, height: PH, class: 'op-axis' }));
    svg.appendChild(el('text', { x: ML + PW / 2, y: H - 4, 'text-anchor': 'middle', class: 'op-axlbl' }, 'x'));
    svg.appendChild(el('text', { x: 16, y: MT + PH / 2 + 5, 'text-anchor': 'middle', class: 'op-axlbl' }, o.ylabel));
    return { px: px, py: py, g: g, ML: ML, MT: MT, PW: PW, PH: PH };
  }
  function pathD(fn, a, x0, x1, n) {
    var d = '';
    for (var i = 0; i <= n; i++) {
      var x = x0 + (x1 - x0) * i / n, y = fn(x);
      d += (i ? ' L' : 'M') + a.px(x).toFixed(2) + ',' + a.py(y).toFixed(2);
    }
    return d;
  }
  function dot(a, x, y, cls, r) { return el('circle', { cx: a.px(x), cy: a.py(y), r: r || 5, class: cls }); }
  function vline(a, x, cls) {
    return el('line', { x1: a.px(x), y1: a.MT, x2: a.px(x), y2: a.MT + a.PH, class: cls });
  }
  function band(a, x0, x1, cls) {
    return el('rect', { x: a.px(Math.min(x0, x1)), y: a.MT,
                        width: Math.abs(a.px(x1) - a.px(x0)), height: a.PH, class: cls });
  }
  function label(a, x, y, txt, cls, dx, dy) {
    return el('text', { x: a.px(x) + (dx || 0), y: a.py(y) + (dy || 0),
                        'text-anchor': 'middle', class: 'op-lbl ' + (cls || '') }, txt);
  }

  // ---- iteration histories -------------------------------------------------
  // Each returns an array of states; state k is what the k-th iteration looks
  // at, and `est` is the estimate it produces.
  function bisection() {
    var xl = XL0, xu = XU0, prev = NaN, s = [];
    for (var k = 0; k <= NIT; k++) {
      var xr = (xl + xu) / 2, st = { xl: xl, xu: xu, xr: xr, est: xr };
      st.ea = k ? Math.abs((xr - prev) / xr) * 100 : NaN;
      st.keep = fp(xl) * fp(xr) < 0 ? 'left' : 'right';
      s.push(st);
      if (st.keep === 'left') xu = xr; else xl = xr;
      prev = xr;
    }
    return s;
  }
  function newton(x0) {
    var x = x0, s = [];
    for (var k = 0; k <= NIT; k++) {
      var xn = x - fp(x) / fpp(x);
      s.push({ x: x, xn: xn, est: x, ea: k ? Math.abs((x - s[k - 1].x) / x) * 100 : NaN });
      x = xn;
      if (!isFinite(x)) break;
    }
    return s;
  }
  function golden() {
    var R = (Math.sqrt(5) - 1) / 2, xl = XL0, xu = XU0, s = [];
    var d = R * (xu - xl), x1 = xl + d, x2 = xu - d, f1 = f(x1), f2 = f(x2);
    for (var k = 0; k <= NIT; k++) {
      var st = { xl: xl, xu: xu, x1: x1, x2: x2, f1: f1, f2: f2, keep: f1 < f2 ? 'right' : 'left' };
      st.est = f1 < f2 ? x1 : x2;
      st.ea = (1 - R) * Math.abs((xu - xl) / st.est) * 100;
      s.push(st);
      d = R * d;
      if (f1 < f2) { xl = x2; x2 = x1; f2 = f1; x1 = xl + d; f1 = f(x1); }
      else         { xu = x1; x1 = x2; f1 = f2; x2 = xu - d; f2 = f(x2); }
    }
    return s;
  }

        a.g.appendChild(vline(a, st.xu, 'op-bracket-edge'));
      });
      A.g.appendChild(label(A, st.xl, 5, 'xₗ', 'op-lbl-amb', 10, 16));
      A.g.appendChild(label(A, st.xu, 5, 'xᵤ', 'op-lbl-amb', -10, 16));
    }
    A.g.appendChild(el('path', { d: pathD(f, A, XMIN, XMAX, 240), class: 'op-curve' }));
    B.g.appendChild(el('path', { d: pathD(fp, B, XMIN, XMAX, 240),
                                 class: 'op-curve' + (m === 'golden' ? ' op-curve-dim' : '') }));

    var note = '';
    if (m === 'bisection') {
      var xr = st.xr;
      [st.xl, st.xu].forEach(function (x) {
        A.g.appendChild(dot(A, x, f(x), 'op-pt'));
        B.g.appendChild(dot(B, x, fp(x), 'op-pt'));
      });
      A.g.appendChild(vline(A, xr, 'op-drop-line'));
      B.g.appendChild(vline(B, xr, 'op-drop-line'));
      A.g.appendChild(dot(A, xr, f(xr), 'op-pt-new', 6));
      B.g.appendChild(dot(B, xr, fp(xr), 'op-pt-new', 6));
      B.g.appendChild(label(B, xr, fp(xr), 'xᵣ', 'op-lbl-pur', 0, fp(xr) > 0 ? -12 : 22));

  function el2(name, text) { var e = document.createElement(name); if (text !== undefined) e.textContent = text; return e; }

  for (var i = 0; i < N; i++) { xs.push(-LIM + 2 * LIM * i / (N - 1)); ys.push(-LIM + 2 * LIM * i / (N - 1)); }

  function hessian() {
    var l1 = +$('op-l1').value, l2 = +$('op-l2').value, th = +$('op-th').value * Math.PI / 180;
    var c = Math.cos(th), s = Math.sin(th);
    // H = R diag(l1, l2) R^T with R = [c -s; s c]
    return { l1: l1, l2: l2, th: th,
             a: l1 * c * c + l2 * s * s, b: (l1 - l2) * c * s, d: l1 * s * s + l2 * c * c };
  }
  var surfaceMade = false;
  function drawQuad() {
    var H = hessian();
    $('op-l1v').textContent = fmt(H.l1, 1);
    $('op-l2v').textContent = fmt(H.l2, 1);
    $('op-thv').textContent = Math.round(H.th * 180 / Math.PI);
    $('op-h11').textContent = fmt(H.a, 2); $('op-h12').textContent = fmt(H.b, 2);
    $('op-h21').textContent = fmt(H.b, 2); $('op-h22').textContent = fmt(H.d, 2);
    var det = H.a * H.d - H.b * H.b, eps = 1e-9, cls, txt;
    if (H.l1 > eps && H.l2 > eps)        { cls = 'op-min'; txt = 'positive definite: the origin is a minimum (a bowl)'; }
    else if (H.l1 < -eps && H.l2 < -eps) { cls = 'op-max'; txt = 'negative definite: the origin is a maximum (a dome)'; }
    else if (H.l1 * H.l2 < -eps)         { cls = 'op-sad'; txt = 'indefinite: the origin is a saddle point'; }
    else                                 { cls = 'op-deg'; txt = 'singular (an eigenvalue is zero): a flat direction, the test is inconclusive'; }
    $('op-defn').innerHTML = 'λ₁ = ' + fmt(H.l1, 1) + ', λ₂ = ' + fmt(H.l2, 1) + ' — <span class="' + cls + '">' + txt + '</span>';
    $('op-dettr').textContent = 'det H = ' + fmt(det, 2) + ' = λ₁λ₂,   H₁₁ = ' + fmt(H.a, 2);

    var z = ys.map(function (y) {
      return xs.map(function (x) { return 0.5 * (H.a * x * x + 2 * H.b * x * y + H.d * y * y); });
    });
    var zmax = 0.5 * Math.max(Math.abs(H.l1), Math.abs(H.l2), 0.5) * 2 * LIM * LIM;
    var zr = [-zmax, zmax];

    // eigenvector lines on the contour plot
    var c = Math.cos(H.th), s = Math.sin(H.th), L = LIM * 1.5;
    var eig = [[[-L * c, L * c], [-L * s, L * s]], [[L * s, -L * s], [-L * c, L * c]]];
    var contour = [{
      type: 'contour', x: xs, y: ys, z: z, colorscale: 'RdBu', reversescale: true,
      zmin: zr[0], zmax: zr[1], ncontours: 24, line: { width: 1 },
      contours: { showlabels: false }, showscale: false, hovertemplate: 'x %{x:.2f}, y %{y:.2f}<br>f %{z:.2f}<extra></extra>'
    }];
    eig.forEach(function (e, i) {
      contour.push({ type: 'scatter', mode: 'lines', x: e[0], y: e[1], showlegend: false, hoverinfo: 'skip',
                     line: { color: '#333', width: 1.5, dash: 'dash' } });
      contour.push({ type: 'scatter', mode: 'text', x: [e[0][1] * 0.6], y: [e[1][1] * 0.6],
                     text: ['λ' + (i ? '₂' : '₁') + ' = ' + fmt(i ? H.l2 : H.l1, 1)], textposition: 'top center',
                     textfont: { size: 14, color: '#222' }, showlegend: false, hoverinfo: 'skip' });
    });
    var clayout = { margin: { l: 40, r: 10, t: 10, b: 40 }, xaxis: { title: 'x', range: [-LIM, LIM], scaleanchor: 'y' },
                    yaxis: { title: 'y', range: [-LIM, LIM] }, paper_bgcolor: 'rgba(0,0,0,0)' };
    Plotly.react('op-contour', contour, clayout, { displayModeBar: false, responsive: true });

    var surface = [{ type: 'surface', x: xs, y: ys, z: z, colorscale: 'RdBu', reversescale: true,
                     cmin: zr[0], cmax: zr[1], showscale: false, opacity: 0.95,
                     contours: { z: { show: true, usecolormap: true, width: 1, project: { z: false } } },
                     hovertemplate: 'x %{x:.2f}, y %{y:.2f}<br>f %{z:.2f}<extra></extra>' }];
    var slayout = { margin: { l: 0, r: 0, t: 0, b: 0 }, paper_bgcolor: 'rgba(0,0,0,0)',
                    scene: { xaxis: { title: 'x', range: [-LIM, LIM] }, yaxis: { title: 'y', range: [-LIM, LIM] },
                             zaxis: { title: 'f', range: [-8.5, 8.5] }, aspectmode: 'cube' } };
    if (surfaceMade) {
      // keep the user's camera while the sliders move
      slayout.scene.camera = $('op-surface').layout.scene.camera;
      Plotly.react('op-surface', surface, slayout, { displayModeBar: false, responsive: true });
    } else {
      slayout.scene.camera = { eye: { x: 1.6, y: -1.6, z: 0.9 } };
      Plotly.newPlot('op-surface', surface, slayout, { displayModeBar: false, responsive: true });
      surfaceMade = true;
    }
  }
  ['op-l1', 'op-l2', 'op-th'].forEach(function (id) { $(id).addEventListener('input', drawQuad); });
  drawQuad();
})();
