  <p>The model is a weighted sum of basis functions \(z_j = \phi_j(x)\) with
  parameters \(\theta = \vec{a} = (a_1, \dots, a_m)\),
  \[ \hat{y} = m(x; \theta) = \sum_{j=1}^m a_j \phi_j(x) = \vec{\phi}(x)^T \theta . \]
  It is linear in \(\theta\), so with \(Z_{ij} = \phi_j(x_i)\) the least-squares
  fit \(\theta^* = \operatorname{argmin}_\theta J(\theta)\) solves the normal
  equations \(Z^T Z \, \theta = Z^T \vec{y}\).</p>

  <p><b>Pick a family</b> of basis functions and <b>check</b> the ones to
  include in the model.</p>

<script>
(function () {
  function $(id) { return document.getElementById(id); }
  var PLOTOPTS = { displayModeBar: false, responsive: true };
  var BLUE = '#123f63', RED = '#c02a2a', AMB = '#b06a00';
  var XMAX = 6, XBOX = [-0.2, XMAX + 0.2], N = 20, SIGMA = 0.25;

  // ---- families of basis functions ---------------------------------------------
  var COLORS = ['#555555', '#1f77b4', '#2a7a3b', '#c02a2a', '#8e44ad', '#d6589f', '#0e8a8a'];
  var ONE = { name: '1', tex: '1', one: true, f: function (x) { return 1; } };
  function pow(k) {
    return { name: k === 1 ? 'x' : 'x<sup>' + k + '</sup>', tex: k === 1 ? 'x' : 'x^' + k, f: function (x) { return Math.pow(x, k); } };
  }
  function trig(fn, k) {
    var arg = (k === 1 ? '' : k) + 'x';
    return { name: fn + ' ' + arg, tex: '\\' + fn + ' ' + arg, f: function (x) { return Math[fn](k * x); } };
  }
  function bump(c) {
    return { name: 'e<sup>−(x−' + c + ')²</sup>', tex: 'e^{-(x-' + c + ')^2}', f: function (x) { return Math.exp(-(x - c) * (x - c)); } };
  }
  function hinge(c) {
    return { name: 'max(0, x−' + c + ')', tex: '\\max(0,\\, x-' + c + ')', f: function (x) { return Math.max(0, x - c); } };
  }
  var FAMILIES = {
    mix:     [ONE, pow(1), pow(2), trig('sin', 1),
              { name: 'e<sup>−x</sup>', tex: 'e^{-x}', f: function (x) { return Math.exp(-x); } }, bump(4), hinge(3)],
    poly:    [ONE, pow(1), pow(2), pow(3), pow(4), pow(5)],
    fourier: [ONE, trig('sin', 1), trig('cos', 1), trig('sin', 2), trig('cos', 2), trig('sin', 3), trig('cos', 3)],
    rbf:     [ONE, bump(0.5), bump(1.5), bump(2.5), bump(3.5), bump(4.5), bump(5.5)],
    hinge:   [ONE, pow(1), hinge(1), hinge(2), hinge(3), hinge(4), hinge(5)]
  };
  var BASES = [];
  function setFamily() {                               // rebuild the checkboxes; the first two start checked
    BASES = FAMILIES[$('gl-fam').value].map(function (b, j) {
      var s = 0;                                       // typical size on [0, XMAX], used only to generate data
      for (var i = 0; i <= 60; i++) s = Math.max(s, Math.abs(b.f(XMAX * i / 60)));
      return { name: b.name, tex: b.tex, one: b.one, f: b.f, s: s, color: COLORS[j] };
    });
    var box = $('gl-boxes');
    box.innerHTML = '<span class="gl-grp">\\(\\phi_j(x)\\):</span>' + BASES.map(function (b, j) {
      return '<label style="color:' + b.color + '"><input type="checkbox" id="gl-b' + j + '"' + (j < 2 ? ' checked' : '') +
             '> \\(' + b.tex + '\\)</label>';
    }).join('');
    for (var j = 0; j < BASES.length; j++) $('gl-b' + j).addEventListener('change', draw);
    if (window.MathJax && MathJax.typesetPromise) MathJax.typesetPromise([box]);
  }
  function active() {
    var a = [];
    for (var j = 0; j < BASES.length; j++) if ($('gl-b' + j).checked) a.push(j);
    return a;
  }

  // ---- data: a constant plus two randomly chosen functions from the current family, plus noise ----
  var X = [], Y = [], YBOX = [0, 1];
  function randn() { var u = 1 - Math.random(), v = Math.random(); return Math.sqrt(-2 * Math.log(u)) * Math.cos(2 * Math.PI * v); }
  function newData() {
    var pool = [];
    BASES.forEach(function (b, j) { if (!b.one) pool.push(j); });
    var j1 = pool[Math.floor(Math.random() * pool.length)], j2 = j1;
    while (j2 === j1) j2 = pool[Math.floor(Math.random() * pool.length)];
    function coef(j) { return (Math.random() < 0.5 ? -1 : 1) * (1.5 + 1.5 * Math.random()) / BASES[j].s; }
    var c0 = 2 * (Math.random() - 0.5), c1 = coef(j1), c2 = coef(j2);
    X = []; Y = [];
    for (var i = 0; i < N; i++) {
      var x = XMAX * (i + 0.5 + 0.7 * (Math.random() - 0.5)) / N;
      X.push(x); Y.push(c0 + c1 * BASES[j1].f(x) + c2 * BASES[j2].f(x) + SIGMA * randn());
    }
    var lo = Math.min.apply(null, Y), hi = Math.max.apply(null, Y), pad = 0.25 * (hi - lo) + 0.5;
    YBOX = [lo - pad, hi + pad];
  }

  // ---- least squares: Householder QR on Z (same answer as the normal equations) ----
  function lstsq(Z, y) {
    var A = Z.map(function (r) { return r.slice(); }), b = y.slice();
    var rows = A.length, p = rows ? A[0].length : 0, i, j, k;
    if (p === 0) return [];
    for (k = 0; k < p; k++) {
      var nrm = 0; for (i = k; i < rows; i++) nrm += A[i][k] * A[i][k];
      nrm = Math.sqrt(nrm); if (nrm === 0) continue;
      var alpha = A[k][k] > 0 ? -nrm : nrm, v = [];
      for (i = k; i < rows; i++) v.push(A[i][k]);
      v[0] -= alpha;
      var vv = 0; for (i = 0; i < v.length; i++) vv += v[i] * v[i];
      if (vv === 0) continue;
      for (j = k; j < p; j++) {
        var d = 0; for (i = k; i < rows; i++) d += v[i - k] * A[i][j];
        d = 2 * d / vv; for (i = k; i < rows; i++) A[i][j] -= d * v[i - k];
      }
      var db = 0; for (i = k; i < rows; i++) db += v[i - k] * b[i];
      db = 2 * db / vv; for (i = k; i < rows; i++) b[i] -= db * v[i - k];
    }
    var c = new Array(p);
    for (k = p - 1; k >= 0; k--) {
      var sum = b[k]; for (j = k + 1; j < p; j++) sum -= A[k][j] * c[j];
      c[k] = sum / A[k][k];
    }
    return c;
  }


  // ---- the feature matrix as a grid of numbers; hovering a data point highlights its row ----
  function zfmt(v) { var a = Math.abs(v); return (a >= 100 ? v.toFixed(0) : a >= 10 ? v.toFixed(1) : v.toFixed(2)).replace('-', '−'); }
  function showZ(heads, Z) {
    var el = $('gl-z'), n = Z.length, m = heads.length, h = '';
    heads.forEach(function (hd, j) {
      h += '<div class="gl-zh" style="grid-row:1;grid-column:' + (j + 1) + ';color:' + hd.color + '">' + hd.html + '</div>';
    });
    if (m) h += '<div class="gl-zbr" style="grid-row:2 / ' + (n + 2) + ';grid-column:1 / ' + (m + 1) + '"></div>';
    Z.forEach(function (r, i) { r.forEach(function (v, j) {
      h += '<div class="gl-zc" data-r="' + i + '" style="grid-row:' + (i + 2) + ';grid-column:' + (j + 1) + '">' + zfmt(v) + '</div>';
    }); });
    el.style.gridTemplateColumns = 'repeat(' + Math.max(m, 1) + ', auto)';
    el.innerHTML = h;
  }
  function highlightRow(i) {
    document.querySelectorAll('#gl-z .gl-zc').forEach(function (c) { c.classList.toggle('gl-zhi', +c.dataset.r === i); });
  }
  function setupHover(gd, curve) {
    gd.on('plotly_hover', function (e) { var p = e.points[0]; if (p.curveNumber === curve) highlightRow(p.pointNumber); });
    gd.on('plotly_unhover', function () { highlightRow(-1); });
  }

  // ---- drawing ---------------------------------------------------------------
  var hoverMade = false;
  function draw() {
    var act = active(), m = act.length, scaled = $('gl-scale').checked;
    var Z = X.map(function (x) { return act.map(function (j) { return BASES[j].f(x); }); });
    var a = lstsq(Z, Y);
    function model(x) { var s = 0; for (var k = 0; k < m; k++) s += a[k] * BASES[act[k]].f(x); return s; }
    var yhat = X.map(model), Jv = 0;
    for (var i = 0; i < N; i++) Jv += (Y[i] - yhat[i]) * (Y[i] - yhat[i]);
    Jv /= N;

    var gx = [], gy = [];
    for (i = 0; i <= 300; i++) { var x = XMAX * i / 300; gx.push(x); gy.push(model(x)); }
    var rx = [], ry = [];
    for (i = 0; i < N; i++) { rx.push(X[i], X[i], null); ry.push(Y[i], yhat[i], null); }
    var dt = [
      { type: 'scatter', mode: 'lines', x: rx, y: ry, visible: $('gl-res').checked,
        line: { color: RED, width: 2 }, hoverinfo: 'skip', showlegend: false },
      { type: 'scatter', mode: 'lines', x: gx, y: gy, line: { color: AMB, width: 3 }, hoverinfo: 'skip', showlegend: false },
      { type: 'scatter', mode: 'markers', x: X, y: Y, marker: { size: 9, color: BLUE }, showlegend: false,
        hovertemplate: 'x = %{x:.2f}, y = %{y:.2f}<extra></extra>' }
    ];
    var dl = { margin: { l: 40, r: 10, t: 10, b: 40 }, paper_bgcolor: 'rgba(0,0,0,0)',
               xaxis: { title: 'x', range: XBOX, fixedrange: true },
               yaxis: { title: 'y', range: YBOX, fixedrange: true } };
    Plotly.react('gl-data', dt, dl, PLOTOPTS);
    if (!hoverMade) { hoverMade = true; setupHover($('gl-data'), 2); }
    showZ(act.map(function (j, k) { return { html: 'z<sub>' + (k + 1) + '</sub> = ' + BASES[j].name, color: BASES[j].color }; }), Z);

    $('gl-t-scaled').hidden = !scaled; $('gl-t-raw').hidden = scaled;
    var bt = act.map(function (j, k) {
      return { type: 'scatter', mode: 'lines', x: gx, y: gx.map(function (x) { return (scaled ? a[k] : 1) * BASES[j].f(x); }),
               name: (scaled ? 'a<sub>' + (k + 1) + '</sub>z<sub>' + (k + 1) + '</sub>' : 'z<sub>' + (k + 1) + '</sub> = ' + BASES[j].name),
               line: { color: BASES[j].color, width: 2.5 }, hoverinfo: 'skip' };
    });
    var bl = { margin: { l: 40, r: 10, t: 30, b: 40 }, paper_bgcolor: 'rgba(0,0,0,0)', showlegend: true,
               legend: { orientation: 'h', x: 0, y: 1.14, itemclick: false, itemdoubleclick: false },
               xaxis: { title: 'x', range: XBOX, fixedrange: true },
               yaxis: scaled ? { autorange: true, fixedrange: true, zeroline: true }
                             : { range: [-1.5, 6.5], fixedrange: true, zeroline: true } };
    Plotly.react('gl-basis', bt, bl, PLOTOPTS);

    var eq = 'ŷ = ';
    if (m === 0) eq += '0';
    act.forEach(function (j, k) {
      var neg = a[k] < 0, mag = Math.abs(a[k]).toFixed(Math.abs(a[k]) < 0.1 ? 3 : 2);
      eq += (k === 0 ? (neg ? '−' : '') : (neg ? ' − ' : ' + ')) +
            '<span style="color:' + BASES[j].color + ';font-weight:600">' + mag + (BASES[j].one ? '' : ' ' + BASES[j].name) + '</span>';
    });
    $('gl-note').innerHTML = '<div>' + eq + '</div>' +
      '<div>J(θ*) = ' + Jv.toFixed(4) + ',  Z is ' + N + ' × ' + m + '</div>';
  }

  // ---- wiring ------------------------------------------------------------------
  $('gl-fam').addEventListener('change', function () { setFamily(); draw(); });
  ['gl-scale', 'gl-res'].forEach(function (id) { $(id).addEventListener('change', draw); });
  $('gl-new').addEventListener('click', function () { newData(); draw(); });
  setFamily(); newData(); draw();
})();
</script>
