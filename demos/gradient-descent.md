  <p>In more than one dimension there is no bracket to shrink around a
  minimum. Gradient descent instead takes a step downhill, against the gradient:
  \[ \vec{x}_{k+1} = \vec{x}_k - \alpha\, \nabla f(\vec{x}_k). \]
  The test function is a quadratic bowl,
  \[ f(x, y) = \tfrac{1}{2}\, \vec{x}^\mathsf{T} A\, \vec{x}
             = \tfrac{1}{2}\left(x^2 + r\,y^2\right),
     \qquad A = \begin{bmatrix} 1 & 0 \\ 0 & r \end{bmatrix}, \]
  whose minimum is at the origin. Its Hessian is \(A\) everywhere, with
  eigenvalues \(\lambda_1 = r\) (steep, in \(y\)) and \(\lambda_2 = 1\)
  (shallow, in \(x\)), so the ratio \(\lambda_1 / \lambda_2 = r\) sets how
  stretched the contours are. Click on the contour plot to move the starting
  point.</p>

  <p>With a ratio of 1 the contours are circles, the gradient points straight
  at the minimum, and one well-chosen step lands on it. As the ratio grows the
  gradient points across the narrow valley rather than along it: a step size
  small enough to be stable in the steep direction (\(\alpha < 2/\lambda_1\)
) crawls along the shallow one, and a larger one zig-zags across the
  valley or diverges. No single \(\alpha\) is good in both directions at
  once, which is the motivation for choosing the step size adaptively, the
  subject of the next lecture.</p>

<script>
(function () {
  function $(id) { return document.getElementById(id); }
  function fmt(v, d) { return isFinite(v) ? v.toFixed(d) : 'NaN'; }

  // ---- part 2: gradient descent on a quadratic bowl ------------------
  var GL = 3, GN = 61, gxs = [], gys = [];
  for (var gi = 0; gi < GN; gi++) { gxs.push(-GL + 2 * GL * gi / (GN - 1)); gys.push(-GL + 2 * GL * gi / (GN - 1)); }
  var PLOTOPTS = { displayModeBar: false, responsive: true };

  function gmat() {
    // A = diag(1, r): eigenvalue 1 along x (the shallow, long axis of the contours), r along y
    var r = +$('op-gr').value;
    return { r: r, l1: r, a: 1, d: r };
  }
  function G(A, x, y)     { return 0.5 * (A.a * x * x + A.d * y * y); }
  function gradG(A, x, y) { return [A.a * x, A.d * y]; }

  function gdPath(A, x, y, alpha, n) {
    var pts = [[x, y]];
    for (var k = 0; k < n; k++) {
      var g = gradG(A, x, y);
      if (Math.hypot(g[0], g[1]) < 1e-12) break;
      x -= alpha * g[0]; y -= alpha * g[1];
      pts.push([x, y]);
    }
    return pts;
  }

  var gcontourMade = false;
  function drawGD() {
    var A = gmat();
    var alpha = +$('op-ga').value, K = +$('op-gk').value, x0 = +$('op-gx0').value, y0 = +$('op-gy0').value;
    $('op-grv').textContent = A.r;
    $('op-gav').textContent = fmt(alpha, 2); $('op-gkv').textContent = K;
    $('op-gx0v').textContent = fmt(x0, 1); $('op-gy0v').textContent = fmt(y0, 1);

    var path = gdPath(A, x0, y0, alpha, +$('op-gk').max);
    var shown = path.slice(0, K + 1), cur = shown[shown.length - 1];
    var g = gradG(A, cur[0], cur[1]), gn = Math.hypot(g[0], g[1]);
    var note = 'iteration ' + (shown.length - 1) + ':  x = ' + fmt(cur[0], 3) + ',  y = ' + fmt(cur[1], 3) +
               ',  f = ' + fmt(G(A, cur[0], cur[1]), 4) + ',  |∇f| = ' + fmt(gn, 4);
    note += '.  Stable when α < 2/λ₁ = ' + fmt(2 / A.l1, 2);
    $('op-gnote').textContent = note;

    var gz = gys.map(function (y) { return gxs.map(function (x) { return G(A, x, y); }); });
    var px = shown.map(function (p) { return p[0]; }), py = shown.map(function (p) { return p[1]; });
    var hover = 'x %{x:.2f}, y %{y:.2f}<br>f %{z:.2f}<extra></extra>';
    var contour = [
      { type: 'contour', x: gxs, y: gys, z: gz, colorscale: 'Blues', zmin: 0, zmax: 0.5 * GL * GL,
        contours: { start: 0.5, end: 0.5 * GL * GL, size: 0.5 }, line: { width: 1 }, showscale: false, hovertemplate: hover },
      { type: 'scatter', mode: 'lines+markers', x: px, y: py, line: { color: '#b06a00', width: 2 },
        marker: { size: 6, color: '#b06a00' }, hoverinfo: 'skip', showlegend: false },
      { type: 'scatter', mode: 'markers', x: [cur[0]], y: [cur[1]],
        marker: { size: 12, color: '#7b3fa0', line: { color: '#fff', width: 1.5 } }, hoverinfo: 'skip', showlegend: false }
    ];
    var clayout = { margin: { l: 40, r: 10, t: 10, b: 40 }, paper_bgcolor: 'rgba(0,0,0,0)',
                    xaxis: { title: 'x', range: [-GL, GL], scaleanchor: 'y', dtick: 1 },
                    yaxis: { title: 'y', range: [-GL, GL], dtick: 1 } };
    Plotly.react('op-gcontour', contour, clayout, PLOTOPTS);
    if (!gcontourMade) {
      gcontourMade = true;
      $('op-gcontour').on('plotly_click', function (ev) {
        var pt = ev.points && ev.points[0]; if (!pt) return;
        $('op-gx0').value = Math.round(pt.x * 10) / 10; $('op-gy0').value = Math.round(pt.y * 10) / 10;
        drawGD();
      });
    }

  }
  ['op-gr', 'op-ga', 'op-gk', 'op-gx0', 'op-gy0'].forEach(function (id) { $(id).addEventListener('input', drawGD); });
  drawGD();
})();
</script>
