  <p>The model is nonlinear in its parameters,
  <span id="nr-eq-exp">\(\theta = (a_0, a_1)\),
  \[ \hat{y} = m(x; \theta) = a_0 \left( 1 - e^{-a_1 x} \right) , \]</span>
  <span id="nr-eq-sin" hidden>\(\theta = (A, \omega)\),
  \[ \hat{y} = m(x; \theta) = A \sin(\omega x) , \]</span>
  so the cost \(J(\theta) = \frac{1}{n} \sum_{i=1}^n (y_i - \hat{y}_i)^2\) is
  not quadratic and \(\theta^* = \operatorname{argmin}_\theta J(\theta)\) has
  to be found iteratively from a starting guess \(\theta_0\).
  <span id="nr-up-gn">\[ \theta_{k+1} = \theta_k + \alpha_k (Z^T Z)^{-1} Z^T \vec{r} , \qquad
      Z_{ij} = \frac{\partial m(x_i; \theta_k)}{\partial \theta_j} , \qquad r_i = y_i - m(x_i; \theta_k) . \]</span>
  <span id="nr-up-newton" hidden>\[ \theta_{k+1} = \theta_k - \alpha_k H^{-1} \nabla J(\theta_k) , \qquad H = \nabla^2 J(\theta_k) . \]</span>
  <span id="nr-up-gd" hidden>\[ \theta_{k+1} = \theta_k - \alpha \nabla J(\theta_k) . \]</span>
  <span id="nr-up-adam" hidden>\[ \theta_{k+1} = \theta_k - \frac{\alpha}{\epsilon + \sqrt{\hat{\vec{s}}_{k+1}}} \odot \hat{\vec{v}}_{k+1} , \qquad
      \gamma_v = 0.9, \quad \gamma_s = 0.999 . \]</span>
  <span id="nr-up-ls" hidden>The step size \(\alpha_k\) is found by backtracking from \(\alpha_k = 1\), and
  \(-\nabla J(\theta_k)\) replaces the step if it is not a descent direction.</span></p>

  <p><b>Drag the point</b> on the cost plot to move \(\theta\). With an
  optimization method selected, the point is the starting guess \(\theta_0\)
  and the slider steps through the iterations.</p>

<script>
(function () {
  function $(id) { return document.getElementById(id); }
  function fmt(v, d) { return isFinite(v) ? (Math.abs(v) >= 1e4 ? v.toExponential(1) : v.toFixed(d)) : 'NaN'; }
  function clamp(v, lo, hi) { return Math.min(hi, Math.max(lo, v)); }
  var PLOTOPTS = { displayModeBar: false, responsive: true };
  var BLUE = '#123f63', RED = '#c02a2a', AMB = '#b06a00', GRN = '#2a7a3b';
  var N = 20, XMAX = 6, KMAX = 100;

  // ---- models: value f, gradient g = df/dtheta, and second derivatives h = [f00, f01, f11] ----
  var MODELS = {
    exp: { names: ['a₀', 'a₁'], box: [[0, 6], [0, 3]], ybox: [-0.5, 5], sigma: 0.2, start: [1.5, 0.3],
           truth: function () { return [3 + 0.8 * (Math.random() - 0.5), 0.8 + 0.3 * (Math.random() - 0.5)]; },
           f: function (x, t) { return t[0] * (1 - Math.exp(-t[1] * x)); },
           g: function (x, t) { var e = Math.exp(-t[1] * x); return [1 - e, t[0] * x * e]; },
           h: function (x, t) { var e = Math.exp(-t[1] * x); return [0, x * e, -t[0] * x * x * e]; } },
    sin: { names: ['A', 'ω'], box: [[0, 4], [0.5, 5.5]], ybox: [-4.2, 4.2], sigma: 0.3, start: [0.8, 2.5],
           truth: function () { return [2 + 0.6 * (Math.random() - 0.5), 2.5 + 0.6 * (Math.random() - 0.5)]; },
           f: function (x, t) { return t[0] * Math.sin(t[1] * x); },
           g: function (x, t) { return [Math.sin(t[1] * x), t[0] * x * Math.cos(t[1] * x)]; },
           h: function (x, t) { return [0, x * Math.cos(t[1] * x), -t[0] * x * x * Math.sin(t[1] * x)]; } }
  };
  var M = null, X = [], Y = [], th0 = [0, 0], cost = null, star = null;
  function method() { return $('nr-method').value; }
  function alpha() { return Math.pow(10, +$('nr-a').value); }
  function randn() { var u = 1 - Math.random(), v = Math.random(); return Math.sqrt(-2 * Math.log(u)) * Math.cos(2 * Math.PI * v); }

  function setModel() {
    var key = $('nr-model').value;
    M = MODELS[key]; th0 = M.start.slice();
    $('nr-eq-exp').hidden = key !== 'exp'; $('nr-eq-sin').hidden = key !== 'sin';
    newData();
  }
  function newData() {
    var t = M.truth();
    X = []; Y = [];
    for (var i = 0; i < N; i++) {
      var x = XMAX * (i + 0.5 + 0.7 * (Math.random() - 0.5)) / N;
      X.push(x); Y.push(M.f(x, t) + M.sigma * randn());
    }
    costGrid();
  }

  // ---- cost, gradient, Hessian ----------------------------------------------
  function J(t) {
    var s = 0;
    for (var i = 0; i < N; i++) { var r = Y[i] - M.f(X[i], t); s += r * r; }
    return s / N;
  }
  function gradJ(t) {
    var g0 = 0, g1 = 0;
    for (var i = 0; i < N; i++) { var r = Y[i] - M.f(X[i], t), g = M.g(X[i], t); g0 -= 2 * r * g[0]; g1 -= 2 * r * g[1]; }
    return [g0 / N, g1 / N];
  }
  function hessJ(t, full) {                            // Gauss-Newton keeps only the Z'Z term
    var h00 = 0, h01 = 0, h11 = 0;
    for (var i = 0; i < N; i++) {
      var g = M.g(X[i], t);
      h00 += 2 * g[0] * g[0]; h01 += 2 * g[0] * g[1]; h11 += 2 * g[1] * g[1];
      if (full) { var r = Y[i] - M.f(X[i], t), h = M.h(X[i], t); h00 -= 2 * r * h[0]; h01 -= 2 * r * h[1]; h11 -= 2 * r * h[2]; }
    }
    return [h00 / N, h01 / N, h11 / N];
  }
  function newtonDir(t, g, full) {                    // -H^{-1} g, or null if H is singular
    var H = hessJ(t, full), det = H[0] * H[2] - H[1] * H[1];
    if (Math.abs(det) < 1e-12) return null;
    return [-(H[2] * g[0] - H[1] * g[1]) / det, -(H[0] * g[1] - H[1] * g[0]) / det];
  }
  var MU1 = 1e-4, RHO = 0.5;                           // Armijo backtracking (mdobook Alg. 4.2 notation)
  var GV = 0.9, GS = 0.999, EPS = 1e-8;                // Adam (Algorithms for Optimization notation)
  function ok(t) { return isFinite(t[0]) && isFinite(t[1]) && Math.abs(t[0]) < 1e6 && Math.abs(t[1]) < 1e6; }
  function path() {                                    // iterates [{t, a, fb}]: point, step size used, gradient fallback?
    var P = [{ t: th0.slice() }], meth = method(), v = [0, 0], sq = [0, 0];
    for (var k = 0; k < KMAX && meth !== 'none'; k++) {
      var t = P[k].t, g = gradJ(t), nt, a = 1, fb = false;
      if (meth === 'gd') { a = alpha(); nt = [t[0] - a * g[0], t[1] - a * g[1]]; }
      else if (meth === 'adam') {
        a = alpha(); nt = [0, 0];
        for (var c = 0; c < 2; c++) {
          v[c] = GV * v[c] + (1 - GV) * g[c]; sq[c] = GS * sq[c] + (1 - GS) * g[c] * g[c];
          var vh = v[c] / (1 - Math.pow(GV, k + 1)), sh = sq[c] / (1 - Math.pow(GS, k + 1));
          nt[c] = t[c] - a * vh / (EPS + Math.sqrt(sh));
        }
      } else {
        var d = newtonDir(t, g, meth === 'newton');
        if (!d || !(g[0] * d[0] + g[1] * d[1] < 0)) { d = [-g[0], -g[1]]; fb = true; }   // not a descent direction
        var J0 = J(t), slope = g[0] * d[0] + g[1] * d[1];
        for (var b = 0; b < 40; b++) {                 // Armijo backtracking from a = 1
          nt = [t[0] + a * d[0], t[1] + a * d[1]];
          if (J(nt) <= J0 + MU1 * a * slope) break;
          a *= RHO;
        }
      }
      if (!ok(nt)) break;
      P.push({ t: nt, a: a, fb: fb });
    }
    return P;
  }

  // ---- contours and the global minimum ---------------------------------------
  var GN = 91;
  function costGrid() {
    var as = [], bs = [], i, best = Infinity;
    for (i = 0; i < GN; i++) {
      as.push(M.box[0][0] + (M.box[0][1] - M.box[0][0]) * i / (GN - 1));
      bs.push(M.box[1][0] + (M.box[1][1] - M.box[1][0]) * i / (GN - 1));
    }
    var zmin = Infinity, zmax = -Infinity;
    var z = bs.map(function (b) { return as.map(function (a) {
      var v = J([a, b]); if (v < best) { best = v; star = [a, b]; }
      v = Math.sqrt(v); if (v < zmin) zmin = v; if (v > zmax) zmax = v; return v;
    }); });
    for (i = 0; i < 30; i++) {                         // polish the best grid point with damped Gauss-Newton
      var d = newtonDir(star, gradJ(star), false); if (!d) break;
      var nt = [star[0] + d[0], star[1] + d[1]];
      for (var c = 0; c < 12 && J(nt) > J(star); c++) nt = [(nt[0] + star[0]) / 2, (nt[1] + star[1]) / 2];
      if (J(nt) > J(star)) break;
      star = nt;
    }
    cost = { as: as, bs: bs, z: z, zmin: zmin, zmax: zmax };
  }

  // ---- drawing ---------------------------------------------------------------
  var made = false, dragging = false;
  function draw() {
    var meth = method(), P = path(), k = +$('nr-k').value, kk = Math.min(k, P.length - 1), t = P[kk].t;
    $('nr-kv').textContent = k; $('nr-av').textContent = alpha().toFixed(3);
    ['gn', 'newton', 'gd', 'adam'].forEach(function (m) { $('nr-up-' + m).hidden = m !== meth; });
    var fixed = meth === 'gd' || meth === 'adam', none = meth === 'none', ls = meth === 'gn' || meth === 'newton';
    $('nr-up-ls').hidden = !ls; $('nr-alab').style.opacity = fixed ? 1 : 0.4;
    $('nr-klab').style.opacity = none ? 0.4 : 1; $('nr-k').disabled = none;

    // -- top pane: data, model at theta_k, residual segments --
    var gx = [], gy = [], rx = [], ry = [], i;
    for (i = 0; i <= 400; i++) { var x = XMAX * i / 400; gx.push(x); gy.push(M.f(x, t)); }
    for (i = 0; i < N; i++) { rx.push(X[i], X[i], null); ry.push(Y[i], M.f(X[i], t), null); }
    var dt = [
      { type: 'scatter', mode: 'lines', x: rx, y: ry, line: { color: RED, width: 2 }, hoverinfo: 'skip', showlegend: false },
      { type: 'scatter', mode: 'lines', x: gx, y: gy, line: { color: AMB, width: 3 }, hoverinfo: 'skip', showlegend: false },
      { type: 'scatter', mode: 'markers', x: X, y: Y, marker: { size: 9, color: BLUE }, showlegend: false,
        hovertemplate: 'x = %{x:.2f}, y = %{y:.2f}<extra></extra>' }
    ];
    var dl = { margin: { l: 40, r: 10, t: 10, b: 40 }, paper_bgcolor: 'rgba(0,0,0,0)',
               xaxis: { title: 'x', range: [-0.2, XMAX + 0.2], fixedrange: true },
               yaxis: { title: 'y', range: M.ybox, fixedrange: true } };
    Plotly.react('nr-data', dt, dl, PLOTOPTS);

    // -- bottom pane: contours of sqrt(J), the iterates, theta_0 (draggable), and theta_k --
    var ct = [
      { type: 'contour', x: cost.as, y: cost.bs, z: cost.z, zmin: cost.zmin, zmax: cost.zmax, showscale: false,
        colorscale: [[0, '#a9cbe6'], [1, '#f4f8fc']], line: { width: 1, color: '#5b86b0' },
        contours: { start: cost.zmin, end: cost.zmax, size: (cost.zmax - cost.zmin) / 22, coloring: 'fill' }, hoverinfo: 'skip' },
      { type: 'scatter', mode: 'markers', x: [star[0]], y: [star[1]], hoverinfo: 'skip', showlegend: false,
        marker: { size: 12, symbol: 'star', color: GRN, line: { color: '#fff', width: 1 } } },
      { type: 'scatter', mode: 'lines+markers', x: P.map(function (p) { return p.t[0]; }), y: P.map(function (p) { return p.t[1]; }),
        line: { color: AMB, width: 1.5 }, marker: { size: 5, color: AMB }, hoverinfo: 'skip', showlegend: false },
      { type: 'scatter', mode: 'markers', x: [th0[0]], y: [th0[1]], hoverinfo: 'skip', showlegend: false, visible: !none,
        marker: { size: 16, color: '#fff', line: { color: AMB, width: 3 } } },
      { type: 'scatter', mode: 'markers', x: [t[0]], y: [t[1]], hoverinfo: 'skip', showlegend: false,
        marker: { size: none ? 16 : 12, color: AMB, line: { color: '#fff', width: none ? 2 : 1.5 } } }
    ];
    var cl = { margin: { l: 45, r: 10, t: 10, b: 40 }, paper_bgcolor: 'rgba(0,0,0,0)', dragmode: false,
               xaxis: { title: M.names[0], range: M.box[0], fixedrange: true },
               yaxis: { title: M.names[1], range: M.box[1], fixedrange: true } };
    Plotly.react('nr-cost', ct, cl, PLOTOPTS);

    var g = gradJ(t), stopped = !none && P.length - 1 < KMAX;
    $('nr-note').innerHTML =
      (none ? '' : '<div>θ<sub>0</sub> = (' + fmt(th0[0], 2) + ', ' + fmt(th0[1], 2) + '),  J(θ<sub>0</sub>) = ' + fmt(J(th0), 3) + '</div>') +
      '<div><span style="color:' + AMB + ';font-weight:600">θ' + (none ? '' : '<sub>' + kk + '</sub>') + '</span> = (' + fmt(t[0], 3) + ', ' + fmt(t[1], 3) + '),  ' +
      'J = ' + fmt(J(t), 4) + ',  ‖∇J‖ = ' + fmt(Math.sqrt(g[0] * g[0] + g[1] * g[1]), 4) +
      (!fixed && kk > 0 ? ',  α<sub>' + (kk - 1) + '</sub> = ' + (+P[kk].a.toPrecision(3)) + (P[kk].fb ? ' along −∇J' : '') : '') +
      (stopped && k > kk ? '  (the iteration broke down after k = ' + kk + ')' : '') + '</div>' +
      '<div><span style="color:' + GRN + ';font-weight:600">θ*</span> = (' + fmt(star[0], 3) + ', ' + fmt(star[1], 3) + '),  ' +
      'J(θ*) = ' + fmt(J(star), 4) + '</div>';

    if (!made) { made = true; setupDrag(); }
  }

  // ---- dragging theta_0 on the cost plot --------------------------------------
  function setStart(ev) {
    var gd = $('nr-cost'), fl = gd._fullLayout, bb = gd.getBoundingClientRect();
    th0 = [clamp(fl.xaxis.p2d(ev.clientX - bb.left - fl.xaxis._offset), M.box[0][0], M.box[0][1]),
           clamp(fl.yaxis.p2d(ev.clientY - bb.top - fl.yaxis._offset), M.box[1][0], M.box[1][1])];
    draw();
  }
  function inPlot(gd, ev) {
    var fl = gd._fullLayout, bb = gd.getBoundingClientRect();
    var px = ev.clientX - bb.left, py = ev.clientY - bb.top;
    return px >= fl.xaxis._offset && px <= fl.xaxis._offset + fl.xaxis._length &&
           py >= fl.yaxis._offset && py <= fl.yaxis._offset + fl.yaxis._length;
  }
  function setupDrag() {
    var gd = $('nr-cost');
    gd.addEventListener('pointerdown', function (ev) {
      if (!inPlot(gd, ev)) return;
      dragging = true; ev.preventDefault(); setStart(ev);
    });
    window.addEventListener('pointermove', function (ev) { if (!dragging) return; ev.preventDefault(); setStart(ev); });
    window.addEventListener('pointerup', function () { dragging = false; });
    window.addEventListener('pointercancel', function () { dragging = false; });
  }

  // ---- wiring ------------------------------------------------------------------
  ['nr-k', 'nr-a'].forEach(function (id) { $(id).addEventListener('input', draw); });
  $('nr-method').addEventListener('change', draw);
  $('nr-model').addEventListener('change', function () { $('nr-k').value = 0; setModel(); draw(); });
  $('nr-new').addEventListener('click', function () { newData(); draw(); });
  setModel(); draw();
})();
</script>
