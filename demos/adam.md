  <p>This reproduces Figures 4.57 and 4.58 from Example 4.18 of
  <a href="https://mdobook.github.io/html/unconstrained/#sec-lbfgs"><em>Engineering
  Design Optimization</em></a> by Martins and Ning (CC BY-NC-SA), with Adam
  added. Rosenbrock's function \(f = (1 - x)^2 + 100\,(y - x^2)^2\) minimized from
  \(\vec{x}_0 = (-1.2,\, 1.0)\) until \(\|\nabla f\|_\infty \le 10^{-6}\),
  by four methods that share the same Armijo backtracking line search
  (\(\mu_1 = 10^{-4}\), \(\rho = 0.5\)), plus Adam, which takes one gradient
  per iteration and no line search. In the notation of
  <a href="https://algorithmsbook.com/optimization/">Algorithms for Optimization</a>
  (Eqs. 5.30&ndash;5.34, with subscripts for iterates), with \(\vec{g}_k = \nabla f(\vec{x}_k)\),
  \(\gamma_v = 0.9\), \(\gamma_s = 0.999\), \(\epsilon = 10^{-8}\), and
  elementwise vector operations:</p>
  \[ \begin{aligned}
     \text{biased decaying momentum:} \quad & \vec{v}_{k+1} = \gamma_v \vec{v}_k + (1 - \gamma_v)\, \vec{g}_k \\
     \text{biased decaying sq. gradient:} \quad & \vec{s}_{k+1} = \gamma_s \vec{s}_k + (1 - \gamma_s)\, \big(\vec{g}_k \odot \vec{g}_k\big) \\
     \text{corrected decaying momentum:} \quad & \hat{\vec{v}}_{k+1} = \vec{v}_{k+1} / (1 - \gamma_v^{\,k+1}) \\
     \text{corrected decaying sq. gradient:} \quad & \hat{\vec{s}}_{k+1} = \vec{s}_{k+1} / (1 - \gamma_s^{\,k+1}) \\
     \text{next iterate:} \quad & \vec{x}_{k+1} = \vec{x}_k - \alpha\, \hat{\vec{v}}_{k+1} \big/ \big(\epsilon + \sqrt{\hat{\vec{s}}_{k+1}}\big)
     \end{aligned} \]

  <p>Click on the contour plot to move the starting point. On the quadratic,
  Newton needs one step and BFGS a handful, while Adam still takes hundreds,
  since its step is about \(\alpha\) per coordinate no matter how simple the
  function. Each line-search
  iteration hides several function and gradient evaluations, so the
  plot against evaluations is the fairer comparison; Adam's step in each
  coordinate is about \(\alpha\) whatever the gradient's size, which is what
  makes it robust to the scaling and noise of neural network training, but
  its convergence is at best linear.</p>

<script>
(function () {
  function $(id) { return document.getElementById(id); }
  function fmt(v, d) { return isFinite(v) ? v.toFixed(d) : 'NaN'; }
  function sci(v) { return isFinite(v) ? v.toExponential(1) : 'NaN'; }
  var PLOTOPTS = { displayModeBar: false, responsive: true };
  var METHODS = [
    { id: 'sd',   name: 'steepest descent',   col: '#7b3fa0', ml: false },
    { id: 'cg',   name: 'conjugate gradient', col: '#2a7a3b', ml: false },
    { id: 'bfgs', name: 'BFGS',               col: '#1a5fb4', ml: false },
    { id: 'nt',   name: 'Newton',             col: '#b06a00', ml: false },
    { id: 'adam', name: 'Adam',               col: '#c02a2a', ml: true }
  ];

  // ---- core: Rosenbrock, line search, the five methods ---------------------
  var QR = 8, QTH = Math.PI / 6, qc = Math.cos(QTH), qs = Math.sin(QTH);           // quadratic from the Newton demo
  var QA = qc * qc + QR * qs * qs, QB = qc * qs * (1 - QR), QD = qs * qs + QR * qc * qc;
  var FUNCS = {                                                                      // hess returns [h11, h12, h22]
    rosen: { box: [-2, 2, -1, 3], x0: [-1.2, 1.0], tf: function (v) { return Math.log10(v + 0.1); },
             f: function (x) { var u = 1 - x[0], v = x[1] - x[0] * x[0]; return u * u + 100 * v * v; },
             grad: function (x) { var v = x[1] - x[0] * x[0]; return [-2 * (1 - x[0]) - 400 * x[0] * v, 200 * v]; },
             hess: function (x) { return [2 - 400 * x[1] + 1200 * x[0] * x[0], -400 * x[0], 200]; } },
    quad:  { box: [-3, 3, -3, 3], x0: [2.0, -1.5], tf: function (v) { return Math.sqrt(v); },   // levels evenly spaced in radius
             f: function (x) { return 0.5 * (QA * x[0] * x[0] + 2 * QB * x[0] * x[1] + QD * x[1] * x[1]); },
             grad: function (x) { return [QA * x[0] + QB * x[1], QB * x[0] + QD * x[1]]; },
             hess: function (x) { return [QA, QB, QD]; } }
  };
  function dot(a, b) { return a[0] * b[0] + a[1] * b[1]; }
  function ginf(g) { return Math.max(Math.abs(g[0]), Math.abs(g[1])); }

  var TOL = 1e-6, KMAX = 20000, MU1 = 1e-4, RHO = 0.5, EPS = 1e-8;

  // Armijo backtracking (mdobook Alg. 4.2): shrink the step from a0 by rho until
  // sufficient decrease holds. Returns the step and how many f evaluations it used.
  function armijo(F, x, g, d, a0) {
    var f0 = F.f(x), s0 = dot(g, d), a = a0, n = 0;
    if (!(s0 < 0)) return { a: 0, n: n };
    for (var i = 0; i < 60; i++) {
      n++;
      if (F.f([x[0] + a * d[0], x[1] + a * d[1]]) <= f0 + MU1 * a * s0) return { a: a, n: n };
      a *= RHO;
    }
    return { a: 0, n: n };
  }

  // 2x2 helpers for BFGS: matrices as [[a, b], [c, d]]
  function mv(A, v) { return [A[0][0] * v[0] + A[0][1] * v[1], A[1][0] * v[0] + A[1][1] * v[1]]; }
  function mm(A, B) { return [[A[0][0] * B[0][0] + A[0][1] * B[1][0], A[0][0] * B[0][1] + A[0][1] * B[1][1]],
                              [A[1][0] * B[0][0] + A[1][1] * B[1][0], A[1][0] * B[0][1] + A[1][1] * B[1][1]]]; }
  function tr(A) { return [[A[0][0], A[1][0]], [A[0][1], A[1][1]]]; }
  function bfgsUpdate(V, s, y) {                                         // mdobook Eq. 4.86
    var sig = 1 / dot(s, y);
    var A = [[1 - sig * s[0] * y[0], -sig * s[0] * y[1]], [-sig * s[1] * y[0], 1 - sig * s[1] * y[1]]];
    var W = mm(mm(A, V), tr(A));
    return [[W[0][0] + sig * s[0] * s[0], W[0][1] + sig * s[0] * s[1]], [W[1][0] + sig * s[1] * s[0], W[1][1] + sig * s[1] * s[1]]];
  }

  // Runs one method from x0; P = {a2, b1, b2}. Returns the iterates with
  // ||grad||_inf and the cumulative evaluation count at each one.
  function run(F, m, x0, P) {
    var ml = m === 'adam';
    var grad = F.grad, hess = F.hess, x = x0.slice(), g = grad(x), ne = ml ? 1 : 2, status = '';   // f and grad at x0
    var pts = [{ x: x[0], y: x[1], gn: ginf(g), ne: ne }];
    var dprev = null, gprev = null, aprev = null, V = null;
    var mA = [0, 0], vA = [0, 0];
    for (var k = 0; k < KMAX; k++) {
      var gn = ginf(g);
      if (gn <= TOL) { status = 'converged'; break; }
      var xn, a = null, d = null, note = '';
      if (m === 'adam') {
        mA = [P.b1 * mA[0] + (1 - P.b1) * g[0], P.b1 * mA[1] + (1 - P.b1) * g[1]];
        vA = [P.b2 * vA[0] + (1 - P.b2) * g[0] * g[0], P.b2 * vA[1] + (1 - P.b2) * g[1] * g[1]];
        var c1 = 1 - Math.pow(P.b1, k + 1), c2 = 1 - Math.pow(P.b2, k + 1);
        xn = [x[0] - P.a2 * (mA[0] / c1) / (Math.sqrt(vA[0] / c2) + EPS),
              x[1] - P.a2 * (mA[1] / c1) / (Math.sqrt(vA[1] / c2) + EPS)];
      } else {
        if (m === 'sd') d = [-g[0], -g[1]];
        else if (m === 'cg') {
          d = [-g[0], -g[1]];
          if (gprev) {                                                   // Polak-Ribiere+ with restart
            var beta = Math.max(0, dot(g, [g[0] - gprev[0], g[1] - gprev[1]]) / dot(gprev, gprev));
            d = [-g[0] + beta * dprev[0], -g[1] + beta * dprev[1]];
            if (dot(d, g) >= 0) { d = [-g[0], -g[1]]; note = 'restart'; }
          }
        } else if (m === 'bfgs') {
          if (!V) { var gg = Math.hypot(g[0], g[1]); V = [[1 / gg, 0], [0, 1 / gg]]; }   // mdobook Eq. 4.85
          var Vg = mv(V, g); d = [-Vg[0], -Vg[1]];
          if (dot(d, g) >= 0) { V = null; d = [-g[0], -g[1]]; note = 'reset'; }
        } else {                                                         // Newton
          var H = hess(x), det = H[0] * H[2] - H[1] * H[1];
          d = Math.abs(det) > 1e-14 ? [-(H[2] * g[0] - H[1] * g[1]) / det, -(-H[1] * g[0] + H[0] * g[1]) / det] : [-g[0], -g[1]];
          if (dot(d, g) >= 0) { d = [-g[0], -g[1]]; note = 'not descent: steepest descent step'; }
        }
        var a0;
        if (m === 'nt' || m === 'bfgs') a0 = 1;
        else if (aprev === null) a0 = 1 / Math.hypot(g[0], g[1]);
        else a0 = Math.min(1e3, Math.max(1e-10, 2 * aprev * dot(gprev, dprev) / dot(g, d)));   // backtracking only shrinks, so start above the last step
        var w = armijo(F, x, g, d, a0); ne += w.n; a = w.a;
        if (a === 0) { status = 'stuck: line search failed'; break; }
        xn = [x[0] + a * d[0], x[1] + a * d[1]];
      }
      var gnew = grad(xn); ne++;                                             // one gradient per accepted point
      if (m === 'bfgs' && V) {
        var s = [xn[0] - x[0], xn[1] - x[1]], y = [gnew[0] - g[0], gnew[1] - g[1]];
        if (dot(s, y) > 1e-14) V = bfgsUpdate(V, s, y); else V = null;
      }
      var last = pts[pts.length - 1]; last.alpha = a; last.note = note;
      x = xn; gprev = g; dprev = d; aprev = a; g = gnew;
      if (!isFinite(x[0]) || !isFinite(x[1]) || !isFinite(g[0]) || !isFinite(g[1]) || Math.abs(x[0]) > 1e4 || Math.abs(x[1]) > 1e4) {
        status = 'diverged'; break;
      }
      pts.push({ x: x[0], y: x[1], gn: ginf(g), ne: ne });
    }
    if (!status) status = 'not converged after ' + KMAX + ' iterations';
    return { pts: pts, status: status };
  }
  // ---- end core -------------------------------------------------------------

  if (typeof module !== 'undefined') { module.exports = { run: run, FUNCS: FUNCS }; return; }   // for the node test harness

  // ---- drawing --------------------------------------------------------------
  var GN = 121, made = false, cache = {}, X0 = {};
  Object.keys(FUNCS).forEach(function (k) {                                        // contour grids, computed once
    var F = FUNCS[k], B = F.box, xs = [], ys = [];
    X0[k] = F.x0.slice();
    for (var gi = 0; gi < GN; gi++) { xs.push(B[0] + (B[1] - B[0]) * gi / (GN - 1)); ys.push(B[2] + (B[3] - B[2]) * gi / (GN - 1)); }
    var z = ys.map(function (y) { return xs.map(function (x) { return F.tf(F.f([x, y])); }); }), zmin = Infinity, zmax = -Infinity;
    z.forEach(function (r) { r.forEach(function (v) { if (v > zmax) zmax = v; if (v < zmin) zmin = v; }); });
    F.grid = { xs: xs, ys: ys, z: z, zmin: zmin, zmax: zmax };
  });
  function curFn() { return document.querySelector('input[name=op-fn]:checked').value; }

  function iterOf(v) { return v <= 30 ? v : Math.round(30 * Math.pow(KMAX / 30, (v - 30) / 100)); }
  function params() {
    return { a2: Math.pow(10, +$('op-a2').value), b1: 0.9, b2: 0.999 };   // b1 = γ_v, b2 = γ_s
  }
  function path(fn, m, x0, P) {
    var key = [fn, m, x0[0], x0[1], P.a2].join('|');
    if (!cache[key]) { if (Object.keys(cache).length > 40) cache = {}; cache[key] = run(FUNCS[fn], m, x0, P); }
    return cache[key];
  }
  function fmtA(a) { return a >= 0.1 ? fmt(a, 2) : a >= 0.01 ? fmt(a, 3) : fmt(a, 4); }

  function draw() {
    var fn = curFn(), F = FUNCS[fn], G = F.grid, BOX = F.box, P = params(), K = iterOf(+$('op-k').value), x0 = X0[fn], logx = $('op-logx').checked;
    $('op-a2v').textContent = fmtA(P.a2); $('op-kv').textContent = K;

    var active = METHODS.filter(function (m) { return $('op-m' + m.id).checked; });
    var ctr = [{ type: 'contour', x: G.xs, y: G.ys, z: G.z, zmin: G.zmin, zmax: G.zmax, showscale: false,
                 colorscale: [[0, '#a9cbe6'], [1, '#f4f8fc']],
                 contours: { start: G.zmin, end: G.zmax, size: (G.zmax - G.zmin) / 24, coloring: 'fill' },
                 line: { width: 1, color: '#5b86b0' }, hovertemplate: 'x %{x:.2f}, y %{y:.2f}<extra></extra>' }];
    var conv = [], evs = [], notes = [];
    active.forEach(function (m) {
      var r = path(fn, m.id, x0, P), shown = r.pts.slice(0, K + 1), cur = shown[shown.length - 1], n = r.pts.length - 1;
      var done = shown.length === r.pts.length;
      var line = '<span style="color:' + m.col + ';font-weight:600">' + m.name + '</span>: k = ' + (shown.length - 1) +
                 ', ' + cur.ne + ' evaluations,  x = ' + fmt(cur.x, 4) + ',  y = ' + fmt(cur.y, 4) +
                 ',  f = ' + sci(F.f([cur.x, cur.y])) + ',  ‖∇f‖<sub>∞</sub> = ' + sci(cur.gn);
      if (cur.alpha !== null && cur.alpha !== undefined && !m.ml) line += ',  α = ' + fmt(cur.alpha, 4);
      if (cur.note) line += '  <span class="op-bad">' + cur.note + '</span>';
      if (done && r.status === 'converged') line += '  <span class="op-good">converged after ' + n + ' iterations</span>';
      else if (done) line += '  <span class="op-bad">' + r.status + '</span>';
      notes.push('<div>' + line + '</div>');

      var few = shown.length <= 150;
      ctr.push({ type: 'scatter', mode: few ? 'lines+markers' : 'lines', name: m.name, showlegend: false,
                 x: shown.map(function (p) { return p.x; }), y: shown.map(function (p) { return p.y; }),
                 line: { color: m.col, width: 2.5 }, marker: { size: 6, color: m.col }, hoverinfo: 'skip' });
      ctr.push({ type: 'scatter', mode: 'markers', x: [cur.x], y: [cur.y], hoverinfo: 'skip', showlegend: false,
                 marker: { size: 13, color: m.col, line: { color: '#fff', width: 2 } } });
      var ks = r.pts.map(function (p, i) { return i; });
      conv.push({ type: 'scatter', mode: 'lines', name: m.name, x: ks, y: r.pts.map(function (p) { return p.gn; }),
                  line: { color: m.col, width: 2 }, hovertemplate: m.name + '<br>k = %{x}, ‖∇f‖∞ = %{y:.2e}<extra></extra>' });
      conv.push({ type: 'scatter', mode: 'markers', x: [shown.length - 1], y: [cur.gn], hoverinfo: 'skip', showlegend: false,
                  marker: { size: 11, color: m.col, line: { color: '#fff', width: 2 } } });
      evs.push({ type: 'scatter', mode: 'lines', name: m.name, x: r.pts.map(function (p) { return p.ne; }), y: r.pts.map(function (p) { return p.gn; }),
                 line: { color: m.col, width: 2 }, hovertemplate: m.name + '<br>%{x} evaluations, ‖∇f‖∞ = %{y:.2e}<extra></extra>' });
      evs.push({ type: 'scatter', mode: 'markers', x: [cur.ne], y: [cur.gn], hoverinfo: 'skip', showlegend: false,
                 marker: { size: 11, color: m.col, line: { color: '#fff', width: 2 } } });
    });
    $('op-note').innerHTML = notes.join('');

    Plotly.react('op-contour', ctr, { margin: { l: 40, r: 10, t: 10, b: 40 }, paper_bgcolor: 'rgba(0,0,0,0)',
                  xaxis: { title: 'x', range: [BOX[0], BOX[1]], scaleanchor: 'y', dtick: 1 },
                  yaxis: { title: 'y', range: [BOX[2], BOX[3]], dtick: 1 } }, PLOTOPTS);
    function convLayout(xtitle, showLegend) {
      return { margin: { l: 55, r: 10, t: showLegend ? 40 : 10, b: 40 }, paper_bgcolor: 'rgba(0,0,0,0)', showlegend: showLegend,
               xaxis: { title: xtitle, type: logx ? 'log' : 'linear', rangemode: 'tozero' }, yaxis: { title: '‖∇f‖∞', type: 'log', exponentformat: 'power', dtick: 2 },
               legend: { orientation: 'h', x: 0, y: 1.02, xanchor: 'left', yanchor: 'bottom' } };
    }
    Plotly.react('op-conv', conv, convLayout('iteration k', false), PLOTOPTS);
    Plotly.react('op-evals', evs, convLayout('function and gradient evaluations', false), PLOTOPTS);
    if (!made) {
      made = true;
      $('op-contour').on('plotly_click', function (ev) {
        var pt = ev.points && ev.points[0]; if (!pt) return;
        X0[curFn()] = [Math.round(pt.x * 10) / 10, Math.round(pt.y * 10) / 10];
        draw();
      });
    }
  }
  var ids = ['op-a2', 'op-k', 'op-logx'];
  METHODS.forEach(function (m) { ids.push('op-m' + m.id); });
  ids.forEach(function (id) { $(id).addEventListener('input', draw); });
