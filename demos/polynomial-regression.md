  <p>The model is a polynomial of degree \(m\) with parameters
  \(\theta = (a_0, \dots, a_m)\),
  \[ \hat{y} = m(x; \theta) = a_0 + a_1 x + \cdots + a_m x^m , \]
  fit to the training data only,
  \(\theta^* = \operatorname{argmin}_\theta J_\text{train}(\theta)\).
  The error on each set is \(\text{RMSE} = \sqrt{J(\theta^*)}\).</p>

<script>
(function () {
  function $(id) { return document.getElementById(id); }
  function fmt(v, d) { return isFinite(v) ? (v >= 1000 ? v.toExponential(1) : v.toFixed(d)) : 'NaN'; }
  var PLOTOPTS = { displayModeBar: false, responsive: true };
  var BLUE = '#123f63', RED = '#c02a2a', AMB = '#b06a00', GREY = '#888';
  var MMAX = 15, YBOX = [-1.9, 1.9], EMAX = 1.2;

  // ---- data ------------------------------------------------------------------
  function ftrue(x) { return Math.sin(2 * Math.PI * x); }
  function randn() { var u = 1 - Math.random(), v = Math.random(); return Math.sqrt(-2 * Math.log(u)) * Math.cos(2 * Math.PI * v); }
  var TR = null, TE = null, XS = [], EPS = [], Y = [], tr = [], te = [], fits = [];
  function gen(k) {                                    // k points on a jittered grid, with unit-variance noise draws
    var d = { x: [], e: [] };
    for (var i = 0; i < k; i++) { d.x.push((i + 0.5 + 0.8 * (Math.random() - 0.5)) / k); d.e.push(randn()); }
    return d;
  }
  function newTrain() { TR = gen(+$('pr-ntr').value); assemble(); }
  function newTest() { TE = gen(+$('pr-nte').value); assemble(); }
  function newData() { TR = gen(+$('pr-ntr').value); TE = gen(+$('pr-nte').value); assemble(); }
  function newSplit() {                                // pool all the points and reassign them at random
    var x = TR.x.concat(TE.x), e = TR.e.concat(TE.e), k = TR.x.length;
    for (var i = x.length - 1; i > 0; i--) {
      var j = Math.floor(Math.random() * (i + 1)), t = x[i]; x[i] = x[j]; x[j] = t; t = e[i]; e[i] = e[j]; e[j] = t;
    }
    TR = { x: x.slice(0, k), e: e.slice(0, k) }; TE = { x: x.slice(k), e: e.slice(k) };
    assemble();
  }
  function assemble() {                                // training points first, then test points
    XS = TR.x.concat(TE.x); EPS = TR.e.concat(TE.e); tr = []; te = [];
    for (var i = 0; i < XS.length; i++) (i < TR.x.length ? tr : te).push(i);
    refit();
  }
  function refit() {
    var s = +$('pr-s').value;
    Y = XS.map(function (x, i) { return ftrue(x) + s * EPS[i]; });
    var mmax = Math.min(MMAX, tr.length - 1);          // degree n_train - 1 interpolates the training data
    fits = [];
    for (var m = 0; m <= mmax; m++) {
      var c = polyfit(tr, m);
      fits.push({ c: c, tr: rmse(c, tr), te: rmse(c, te) });
    }
    var sl = $('pr-m'); sl.max = mmax; if (+sl.value > mmax) sl.value = mmax;
  }

  // ---- least squares ----------------------------------------------------------
  // The fit is done in a Chebyshev basis on t = 2x - 1 with Householder QR. This
  // gives the same polynomial as the monomial normal equations, without the
  // ill-conditioning of Z'Z at high degree.
  function basis(x, m) {
    var t = 2 * x - 1, T = [1];
    if (m >= 1) T.push(t);
    for (var j = 2; j <= m; j++) T.push(2 * t * T[j - 1] - T[j - 2]);
    return T;
  }
  function peval(c, x) { var T = basis(x, c.length - 1), s = 0; for (var j = 0; j < c.length; j++) s += c[j] * T[j]; return s; }
  function rmse(c, idx) {
    var s = 0;
    idx.forEach(function (i) { var r = Y[i] - peval(c, XS[i]); s += r * r; });
    return Math.sqrt(s / idx.length);
  }
  function polyfit(idx, m) {
    var A = idx.map(function (i) { return basis(XS[i], m); }), b = idx.map(function (i) { return Y[i]; });
    var rows = A.length, p = m + 1, i, j, k;
    for (k = 0; k < p; k++) {
      var nrm = 0; for (i = k; i < rows; i++) nrm += A[i][k] * A[i][k];
      nrm = Math.sqrt(nrm); if (nrm === 0) continue;
      var alpha = A[k][k] > 0 ? -nrm : nrm, v = [];
      for (i = k; i < rows; i++) v.push(A[i][k]);
      v[0] -= alpha;
      var vv = 0; for (i = 0; i < v.length; i++) vv += v[i] * v[i];
      if (vv === 0) continue;
      for (j = k; j < p; j++) {
        var d = 0; for (i = k; i < rows; i++) d += v[i - k] * A[i][j];
        d = 2 * d / vv; for (i = k; i < rows; i++) A[i][j] -= d * v[i - k];
      }
      var db = 0; for (i = k; i < rows; i++) db += v[i - k] * b[i];
      db = 2 * db / vv; for (i = k; i < rows; i++) b[i] -= db * v[i - k];
    }
    var c = new Array(p);
    for (k = p - 1; k >= 0; k--) {
      var sum = b[k]; for (j = k + 1; j < p; j++) sum -= A[k][j] * c[j];
      c[k] = sum / A[k][k];
    }
    return c;
  }

  // ---- drawing ---------------------------------------------------------------
  var made = false;
  function draw() {
    var m = +$('pr-m').value, fit = fits[m], showT = $('pr-test').checked, showF = $('pr-true').checked, s = +$('pr-s').value;
    $('pr-mv').textContent = m; $('pr-ntrv').textContent = tr.length; $('pr-ntev').textContent = te.length; $('pr-sv').textContent = s.toFixed(2);

    var gx = [], gy = [], gf = [];
    for (var i = 0; i <= 400; i++) { var x = i / 400; gx.push(x); gy.push(peval(fit.c, x)); gf.push(ftrue(x)); }
    function pick(idx, a) { return idx.map(function (i) { return a[i]; }); }
    var dt = [
      { type: 'scatter', mode: 'lines', x: gx, y: gf, name: 'true function', visible: showF,
        line: { color: GREY, width: 2, dash: 'dash' }, hoverinfo: 'skip' },
      { type: 'scatter', mode: 'lines', x: gx, y: gy, name: 'fit, m = ' + m,
        line: { color: AMB, width: 3 }, hoverinfo: 'skip' },
      { type: 'scatter', mode: 'markers', x: pick(tr, XS), y: pick(tr, Y), name: 'training data (' + tr.length + ')',
        marker: { size: 9, color: BLUE }, hovertemplate: 'x = %{x:.2f}, y = %{y:.2f}<extra>training</extra>' },
      { type: 'scatter', mode: 'markers', x: pick(te, XS), y: pick(te, Y), name: 'test data (' + te.length + ')', visible: showT,
        marker: { size: 12, symbol: 'star', color: RED },
        hovertemplate: 'x = %{x:.2f}, y = %{y:.2f}<extra>test</extra>' }
    ];
    var dl = { margin: { l: 40, r: 10, t: 30, b: 40 }, paper_bgcolor: 'rgba(0,0,0,0)',
               legend: { orientation: 'h', x: 0, y: 1.12, itemclick: false, itemdoubleclick: false },
               xaxis: { title: 'x', range: [-0.03, 1.03], fixedrange: true },
               yaxis: { title: 'y', range: YBOX, fixedrange: true } };
    Plotly.react('pr-data', dt, dl, PLOTOPTS);

    var ms = fits.map(function (_, j) { return j; });
    var et = [
      { type: 'scatter', mode: 'lines+markers', x: ms, y: fits.map(function (f) { return f.tr; }), name: 'training',
        line: { color: BLUE, width: 2 }, marker: { size: 7 }, hovertemplate: 'm = %{x}: %{y:.3g}<extra>training</extra>' },
      { type: 'scatter', mode: 'lines+markers', x: ms, y: fits.map(function (f) { return f.te; }), name: 'test', visible: showT,
        line: { color: RED, width: 2 }, marker: { size: 10, symbol: 'star' },
        hovertemplate: 'm = %{x}: %{y:.3g}<extra>test</extra>' }
    ];
    var shapes = [{ type: 'line', x0: m, x1: m, y0: 0, y1: 1, yref: 'paper', line: { color: AMB, width: 3 }, layer: 'below' }];
    var annos = [];
    if (showF) {
      shapes.push({ type: 'line', x0: 0, x1: 1, xref: 'paper', y0: s, y1: s, line: { color: GREY, width: 1.5, dash: 'dash' } });
      annos.push({ x: 1, xref: 'paper', xanchor: 'right', y: s, yanchor: 'bottom', text: 'noise σ', showarrow: false, font: { color: GREY } });
    }
    var el = { margin: { l: 50, r: 10, t: 30, b: 40 }, paper_bgcolor: 'rgba(0,0,0,0)', shapes: shapes, annotations: annos,
               legend: { orientation: 'h', x: 0, y: 1.15, itemclick: false, itemdoubleclick: false },
               xaxis: { title: 'degree m', range: [-0.5, fits.length - 0.5], dtick: 1, fixedrange: true },
               yaxis: { title: 'RMSE', range: [0, EMAX], fixedrange: true } };
    Plotly.react('pr-err', et, el, PLOTOPTS);

    $('pr-note').innerHTML =
      '<span style="color:' + AMB + ';font-weight:600">m = ' + m + '</span>,  ' +
      '<span style="color:' + BLUE + '">RMSE<sub>train</sub> = ' + fmt(fit.tr, 3) + '</span>' +
      (showT ? ',  <span style="color:' + RED + '">RMSE<sub>test</sub> = ' + fmt(fit.te, 3) + '</span>' : '');

    if (!made) { made = true; setupClick(); }
  }

  // ---- click on the error plot to pick a degree ---------------------------------
  function setupClick() {
    var gd = $('pr-err');
    gd.addEventListener('click', function (ev) {
      var fl = gd._fullLayout, bb = gd.getBoundingClientRect();
      var px = ev.clientX - bb.left - fl.xaxis._offset, py = ev.clientY - bb.top - fl.yaxis._offset;
      if (px < 0 || px > fl.xaxis._length || py < 0 || py > fl.yaxis._length) return;
      var m = Math.round(fl.xaxis.p2d(px));
      $('pr-m').value = Math.min(fits.length - 1, Math.max(0, m)); draw();
    });
  }

  // ---- wiring ------------------------------------------------------------------
  $('pr-m').addEventListener('input', draw);
  $('pr-ntr').addEventListener('input', function () { newTrain(); draw(); });
  $('pr-nte').addEventListener('input', function () { newTest(); draw(); });
  $('pr-s').addEventListener('input', function () { refit(); draw(); });
  ['pr-test', 'pr-true'].forEach(function (id) { $(id).addEventListener('change', draw); });
  $('pr-split').addEventListener('click', function () { newSplit(); draw(); });
  $('pr-new').addEventListener('click', function () { newData(); draw(); });
  newData(); draw();
})();
</script>
