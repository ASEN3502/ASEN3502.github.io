# Optimization: KKT Stationarity

    <blockquote>
      <p>Let's add a demo showing the stationarity KKT condition. 2D quadratic
      objective. 1 equality constraint. 1 inequality constraint. Let me move a
      point. From that point have a grad f vector, a lambda * grad h vector,
      and a sigma * grad g vector, so we can see that they cancel out at
      stationary points. We may have to add "snapping" to get active
      constraints.</p>
    </blockquote>

  \[ \begin{aligned}
     \underset{\vec{x}}{\text{minimize}} &\quad f(\vec{x}) = (x_1 - 2)^2 + 2\,(x_2 - 2)^2 \\
     \text{subject to} &\quad h(\vec{x}) = \tfrac{1}{4} x_1^2 - x_2 - \tfrac{1}{2} = 0 \\
                       &\quad g(\vec{x}) = x_1^2 + x_2^2 - r^2 \le 0
     \end{aligned} \]
  \[ \mathcal{L}(\vec{x}, \lambda, \sigma) = f(\vec{x}) + \lambda\, h(\vec{x}) + \sigma\, g(\vec{x}) \]
  \[ \nabla \mathcal{L} = \nabla f + \lambda \nabla h + \sigma \nabla g = \vec{0} \]

  <p>Drag the point. It snaps onto \(h = 0\) and \(g = 0\) and onto nearby
  stationary points. With "solve" checked, a multiplier on its constraint is
  the value that cancels as much of \(\nabla f\) as possible, and off the
  constraint it is zero. Uncheck "solve" to set the multiplier with its slider.</p>

<script>
(function () {
  function $(id) { return document.getElementById(id); }
  function fmt(v, d) { v = Math.abs(v) < 0.5 * Math.pow(10, -d) ? 0 : v; return isFinite(v) ? v.toFixed(d).replace('-', '−') : 'NaN'; }
  var PLOTOPTS = { displayModeBar: false, responsive: true };
  var CF = '#123f63', CH = '#7b3fa0', CG = '#b06a00', CL = '#0d8bff';
  var X1L = -3.2, X1R = 4.4, X2L = -2.6, X2R = 3.8;
  var SCALE = 0.2;      // plot length of a gradient = SCALE * its norm
  var SNAP = 0.15;      // snapping distance, in x units
  var MSNAP = 0.03;     // a hand-set multiplier this close to the solved value snaps to it
  var TOL = 1e-8;

  // each function: value, gradient, (constant or x-dependent) Hessian
  var F = { v: function (x) { return Math.pow(x[0] - 2, 2) + 2 * Math.pow(x[1] - 2, 2); },
            g: function (x) { return [2 * (x[0] - 2), 4 * (x[1] - 2)]; },
            H: [[2, 0], [0, 4]] };
  var Hc = { v: function (x) { return x[0] * x[0] / 4 - x[1] - 0.5; },
             g: function (x) { return [x[0] / 2, -1]; },
             H: [[0.5, 0], [0, 0]] };
  var R = 2;
  var Gc = { v: function (x) { return x[0] * x[0] + x[1] * x[1] - R * R; },
             g: function (x) { return [2 * x[0], 2 * x[1]]; },
             H: [[2, 0], [0, 2]] };

  function dist(a, b) { return Math.hypot(a[0] - b[0], a[1] - b[1]); }

  // Gaussian elimination with partial pivoting; returns null if singular
  function solve(A, b) {
    var n = b.length, i, j, k, M = A.map(function (r, i) { return r.concat([b[i]]); });
    for (k = 0; k < n; k++) {
      var p = k;
      for (i = k + 1; i < n; i++) if (Math.abs(M[i][k]) > Math.abs(M[p][k])) p = i;
      if (Math.abs(M[p][k]) < 1e-12) return null;
      var t = M[k]; M[k] = M[p]; M[p] = t;
      for (i = k + 1; i < n; i++) {
        var m = M[i][k] / M[k][k];
        for (j = k; j <= n; j++) M[i][j] -= m * M[k][j];
      }
    }
    var x = new Array(n);
    for (i = n - 1; i >= 0; i--) {
      var s = M[i][n];
      for (j = i + 1; j < n; j++) s -= M[i][j] * x[j];
      x[i] = s / M[i][i];
    }
    return x;
  }

  // closest-ish point on c = 0: repeated steps along the gradient
  function project(c, x) {
    x = x.slice();
    for (var i = 0; i < 30; i++) {
      var v = c.v(x), g = c.g(x), n2 = g[0] * g[0] + g[1] * g[1];
      if (Math.abs(v) < 1e-13 || n2 < 1e-14) break;
      x = [x[0] - v * g[0] / n2, x[1] - v * g[1] / n2];
    }
    return Math.abs(c.v(x)) < 1e-9 ? x : null;
  }

  // multipliers that minimize || grad f + sum mu_i grad c_i || (least squares)
  // `base` replaces grad f when some multipliers are already fixed
  function fit(x, cons, base) {
    var gf = base || F.g(x), G = cons.map(function (c) { return c.g(x); }), m = cons.length;
    if (!m) return [];
    var A = G.map(function (gi) { return G.map(function (gj) { return gi[0] * gj[0] + gi[1] * gj[1]; }); });
    var b = G.map(function (gi) { return -(gi[0] * gf[0] + gi[1] * gf[1]); });
    return solve(A, b) || (m === 2 ? [b[0] / A[0][0], 0] : [0]);
  }

  // Newton's method on the KKT system for the constraints in `act` (all treated as equalities)
  // `fixed`: [{c, mu}] constraints whose multipliers are held at mu and are not enforced
  function kktNewton(x, act, fixed) {
    fixed = fixed || [];
    function base(x) {
      var b = F.g(x);
      fixed.forEach(function (q) { var g = q.c.g(x); b[0] += q.mu * g[0]; b[1] += q.mu * g[1]; });
      return b;
    }
    var m = act.length, mu = fit(x, act, base(x)), it, i, j;
    x = x.slice();
    for (it = 0; it < 30; it++) {
      var gf = base(x), G = act.map(function (c) { return c.g(x); });
      var r = [gf[0], gf[1]].concat(act.map(function (c) { return c.v(x); }));
      var A = [];
      for (i = 0; i < 2 + m; i++) { A.push([]); for (j = 0; j < 2 + m; j++) A[i].push(0); }
      for (i = 0; i < 2; i++) for (j = 0; j < 2; j++) {
        A[i][j] = F.H[i][j];
        fixed.forEach(function (q) { A[i][j] += q.mu * q.c.H[i][j]; });
      }
      act.forEach(function (c, q) {
        for (i = 0; i < 2; i++) {
          r[i] += mu[q] * G[q][i];
          A[i][2 + q] = G[q][i]; A[2 + q][i] = G[q][i];
          for (j = 0; j < 2; j++) A[i][j] += mu[q] * c.H[i][j];
        }
      });
      if (Math.sqrt(r.reduce(function (s, v) { return s + v * v; }, 0)) < 1e-12) return x;
      var d = solve(A, r.map(function (v) { return -v; }));
      if (!d) return null;
      x = [x[0] + d[0], x[1] + d[1]];
      for (i = 0; i < m; i++) mu[i] += d[2 + i];
    }
    return null;
  }

  var useH = true, useG = true, snap = true, solveL = true, solveS = true;
  function manL() { return useH && !solveL; }
  function manS() { return useG && !solveS; }
  var pt = [-0.9, 1.4];

  // where the pointer position p ends up
  function place(p) {
    if (!snap) return p;
    var ph = useH ? project(Hc, p) : null, pg = useG ? project(Gc, p) : null;
    if (ph && dist(ph, p) > SNAP) ph = null;
    if (pg && dist(pg, p) > SNAP) pg = null;
    var act = [], x = p;
    if (ph && pg) {
      var c = kktNewton(p, [Hc, Gc]);                       // the corner h = 0, g = 0
      if (c && dist(c, p) < 2 * SNAP) return c;
      if (dist(ph, p) <= dist(pg, p)) pg = null; else ph = null;
    }
    if (ph) { act = [Hc]; x = ph; } else if (pg) { act = [Gc]; x = pg; }
    var s = kktNewton(x, act);                              // a stationary point for this active set
    if (s && dist(s, p) < 1.5 * SNAP) return s;
    // with hand-set multipliers, also the stationary point of L for those values
    var fixed = (manL() ? [{ c: Hc, mu: +$('kk-lam').value }] : []).concat(manS() ? [{ c: Gc, mu: +$('kk-sig').value }] : []);
    if (fixed.length) {
      s = kktNewton(x, act.filter(function (c) { return !(c === Hc ? manL() : manS()); }), fixed);
      if (s && dist(s, p) < 1.5 * SNAP) return s;
    }
    return x;
  }

  function state(x) {
    var onG = useG && Math.abs(Gc.v(x)) < TOL;
    var onH = useH && Math.abs(Hc.v(x)) < TOL;
    var cons = (onH ? [Hc] : []).concat(onG ? [Gc] : []);
    var mu = fit(x, cons), gf = F.g(x), gh = Hc.g(x), gg = Gc.g(x);
    var lam = onH ? mu[0] : 0, sig = onG ? mu[cons.length - 1] : 0;   // solved together
    if (manL() || manS()) {
      // hand-set values (snapped to the solved value when close), then re-solve the rest around them
      var base = gf.slice(), rest = [];
      if (manL()) { var l = +$('kk-lam').value; lam = snap && onH && Math.abs(l - lam) < MSNAP ? lam : l; }
      if (manS()) { var sg = +$('kk-sig').value; sig = snap && onG && Math.abs(sg - sig) < MSNAP ? sig : sg; }
      if (manL()) { base[0] += lam * gh[0]; base[1] += lam * gh[1]; } else if (onH) rest.push(Hc);
      if (manS()) { base[0] += sig * gg[0]; base[1] += sig * gg[1]; } else if (onG) rest.push(Gc);
      mu = fit(x, rest, base);
      if (!manL() && onH) lam = mu[0];
      if (!manS() && onG) sig = mu[0];
    }
    var S = { x: x, lam: lam, sig: sig, onG: onG, h: Hc.v(x), g: Gc.v(x), gf: gf,
              lgh: [lam * gh[0], lam * gh[1]], sgg: [sig * gg[0], sig * gg[1]] };
    S.sum = [gf[0] + S.lgh[0] + S.sgg[0], gf[1] + S.lgh[1] + S.sgg[1]];
    return S;
  }

  // ---- static traces -------------------------------------------------------
  var NG = 90, gx = [], gy = [], gz = [], i, j;
  for (i = 0; i < NG; i++) { gx.push(-8 + 20 * i / (NG - 1)); gy.push(-8 + 20 * i / (NG - 1)); }
  for (j = 0; j < NG; j++) { gz.push([]); for (i = 0; i < NG; i++) gz[j].push(F.v([gx[i], gy[j]])); }
  var hx = [], hy = [];
  for (i = 0; i <= 200; i++) { var t = -8 + 16 * i / 200; hx.push(t); hy.push(t * t / 4 - 0.5); }
  function circle() {
    var cx = [], cy = [], ox = [-30, 30, 30, -30, -30], oy = [-30, -30, 30, 30, -30], i;
    for (i = 0; i <= 180; i++) { var a = 2 * Math.PI * i / 180; cx.push(R * Math.cos(a)); cy.push(R * Math.sin(a)); }
    // outside of the disk: the big box, then around the circle the other way
    for (i = 180; i >= 0; i--) { ox.push(cx[i]); oy.push(cy[i]); }
    ox.push(-30); oy.push(-30);
    return { cx: cx, cy: cy, ox: ox, oy: oy };
  }

  function arrow(x, v, color, label, width) {
    var len = Math.hypot(v[0], v[1]) * SCALE, tip = [x[0] + SCALE * v[0], x[1] + SCALE * v[1]];
    if (len < 0.02) return [];
    var u = [v[0] * SCALE / len, v[1] * SCALE / len];
    return [
      { x: tip[0], y: tip[1], ax: x[0], ay: x[1], xref: 'x', yref: 'y', axref: 'x', ayref: 'y', text: '',
        showarrow: true, arrowhead: 2, arrowsize: width ? 0.6 : 1, arrowwidth: width || 2.8, arrowcolor: color },
    ].concat(label ? [
      { x: tip[0] + (0.14 + 0.07 * label.replace(/<[^>]*>/g, '').length) * u[0], y: tip[1] + 0.22 * u[1],
        xref: 'x', yref: 'y', text: label, showarrow: false,
        font: { color: color, size: 15 }, bgcolor: 'rgba(255,255,255,.7)', borderpad: 1 }
    ] : []);
  }

  function draw() {
    var S = state(pt);
    // a solved multiplier's (disabled) slider follows it, so switching to hand-set starts from there
    if (solveL) $('kk-lam').value = S.lam;
    if (solveS) $('kk-sig').value = S.sig;
    var x = S.x, ns = Math.hypot(S.sum[0], S.sum[1]), stat = ns < 1e-6;
    var traces = [
      { type: 'contour', x: gx, y: gy, z: gz, showscale: false, hoverinfo: 'skip', name: 'f',
        contours: { coloring: 'lines', start: 0.5, end: 120, size: 2.5 }, line: { width: 1 },
        colorscale: [[0, '#9db4c6'], [1, '#9db4c6']] }
    ];
    var C = circle();
    if (useG) traces.push(
      { type: 'scatter', mode: 'lines', x: C.ox, y: C.oy, fill: 'toself', fillcolor: 'rgba(176,106,0,.10)',
        line: { width: 0 }, hoverinfo: 'skip', showlegend: false },
      { type: 'scatter', mode: 'lines', x: C.cx, y: C.cy, line: { color: CG, width: 2.5 }, name: 'g = 0 (shaded: g > 0)',
        hoverinfo: 'skip' });
    if (useH) traces.push(
      { type: 'scatter', mode: 'lines', x: hx, y: hy, line: { color: CH, width: 2.5 }, name: 'h = 0', hoverinfo: 'skip' });
    traces.push(
      { type: 'scatter', mode: 'markers', x: [x[0]], y: [x[1]], showlegend: false, hoverinfo: 'skip',
        marker: { size: 13, color: stat ? CL : '#fff', line: { color: '#000', width: 2.5 } } });

    // with no active constraint, grad L is grad f: one wide arrow under the other, one shared label
    var same = Math.hypot(S.sum[0] - S.gf[0], S.sum[1] - S.gf[1]) < 1e-9;
    var ann = arrow(x, S.sum, CL, same ? '' : '∇ℒ', same ? 6 : 0)
      .concat(useG ? arrow(x, S.sgg, CG, 'σ∇g') : [])
      .concat(useH ? arrow(x, S.lgh, CH, 'λ∇h') : [])
      .concat(arrow(x, S.gf, CF, same ? '∇f = <span style="color:' + CL + '">∇ℒ</span>' : '∇f'));
    var layout = {
      margin: { l: 45, r: 10, t: 10, b: 40 }, paper_bgcolor: 'rgba(0,0,0,0)', dragmode: false,
      xaxis: { title: 'x₁', range: [X1L, X1R], zeroline: false, constrain: 'domain', fixedrange: true, dtick: 1 },
      yaxis: { title: 'x₂', range: [X2L, X2R], zeroline: false, constrain: 'domain', fixedrange: true, dtick: 1,
               scaleanchor: 'x', scaleratio: 1 },
      legend: { x: 0.01, y: 0.01, yanchor: 'bottom', bgcolor: 'rgba(255,255,255,.8)' },
      annotations: ann
    };
    Plotly.react('kk-plot', traces, layout, PLOTOPTS);

    function chip(ok, txt, on) {
      if (!on) return '<span class="kk-chip kk-off">' + txt + '</span>';
      return '<span class="kk-chip ' + (ok ? 'kk-ok' : 'kk-bad') + '">' + (ok ? '✓ ' : '✗ ') + txt + '</span>';
    }
    var hok = Math.abs(S.h) < TOL, gok = S.g < TOL, sok = S.sig > -1e-9, cok = Math.abs(S.sig * S.g) < 1e-6;
    var chips = chip(stat, 'stationarity', true) + chip(hok, 'h = 0', useH) +
      chip(gok, 'g ≤ 0', useG) + chip(sok, 'σ ≥ 0', useG) + chip(cok, 'σ g = 0', useG);
    var kkt = stat && (!useH || hok) && (!useG || (gok && sok && cok));
    $('kk-note').innerHTML =
      '<div>x = (' + fmt(x[0], 3) + ', ' + fmt(x[1], 3) + '),  f = ' + fmt(F.v(x), 3) +
        (useH ? ',  h = ' + fmt(S.h, 3) : '') + (useG ? ',  g = ' + fmt(S.g, 3) : '') + '</div>' +
      '<div><span class="kk-cl">∇ℒ</span> = <span class="kk-cf">∇f</span>' +
        (useH ? ' + <span class="kk-ch">λ∇h</span>' : '') + (useG ? ' + <span class="kk-cg">σ∇g</span>' : '') +
        ' = (' + fmt(S.sum[0], 3) + ', ' + fmt(S.sum[1], 3) + ')' +
        (useH ? ',  <span class="kk-ch">λ = ' + fmt(S.lam, 3) + '</span>' : '') +
        (useG ? ',  <span class="kk-cg">σ = ' + fmt(S.sig, 3) + '</span>' : '') + '</div>' +
      '<div>' + chips + (kkt ? '<span class="kk-kkt kk-chip kk-ok">KKT point</span>' : '') + '</div>';
  }

  // ---- dragging ------------------------------------------------------------
  var dragging = false;
  function clamp(v, a, b) { return Math.min(b, Math.max(a, v)); }
  function move(ev) {
    var gd = $('kk-plot'), fl = gd._fullLayout, bb = gd.getBoundingClientRect();
    var xr = fl.xaxis.range, yr = fl.yaxis.range;
    pt = place([clamp(fl.xaxis.p2d(ev.clientX - bb.left - fl.xaxis._offset), xr[0], xr[1]),
                clamp(fl.yaxis.p2d(ev.clientY - bb.top - fl.yaxis._offset), yr[0], yr[1])]);
    draw();
  }
  function inPlot(gd, ev) {
    var fl = gd._fullLayout, bb = gd.getBoundingClientRect();
    var px = ev.clientX - bb.left, py = ev.clientY - bb.top;
    return px >= fl.xaxis._offset && px <= fl.xaxis._offset + fl.xaxis._length &&
           py >= fl.yaxis._offset && py <= fl.yaxis._offset + fl.yaxis._length;
  }
  var gd = $('kk-plot');
  gd.addEventListener('pointerdown', function (ev) {
    if (!inPlot(gd, ev)) return;
    dragging = true; ev.preventDefault(); move(ev);
  });
  window.addEventListener('pointermove', function (ev) { if (!dragging) return; ev.preventDefault(); move(ev); });
  window.addEventListener('pointerup', function () { dragging = false; });
  window.addEventListener('pointercancel', function () { dragging = false; });

  function opts() {
    useH = $('kk-useh').checked; useG = $('kk-useg').checked; snap = $('kk-snap').checked;
    solveL = $('kk-solvel').checked; solveS = $('kk-solves').checked;
    $('kk-lrow').hidden = !useH; $('kk-srow').hidden = !useG; $('kk-rrow').hidden = !useG;
    $('kk-lam').disabled = solveL; $('kk-sig').disabled = solveS;
    pt = place(pt); draw();
  }
  ['kk-useh', 'kk-useg', 'kk-snap', 'kk-solvel', 'kk-solves'].forEach(function (id) { $(id).addEventListener('change', opts); });
  $('kk-r').addEventListener('input', function () {
    var on = useG && Math.abs(Gc.v(pt)) < TOL;
    R = +$('kk-r').value; $('kk-rv').textContent = R.toFixed(2);
    if (on) pt = project(Gc, pt) || pt;      // a point on the circle rides along with it
