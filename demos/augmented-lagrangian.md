  <p>The landing problem from last lecture: a rocket at
  \((x_0, y_0) = (-18, -4)\) picks the closest point on a \(20 \times 15\)
  pad centered at the origin, avoiding a circular obstacle of radius 5 at
  \((-10, 0)\).</p>
  \[ \begin{aligned}
     \underset{\vec{x}}{\text{minimize}} &\quad f(\vec{x}) = (x - x_0)^2 + (y - y_0)^2 \\
     \text{subject to} &\quad g_1 = -10 - x \le 0, \quad g_2 = x - 10 \le 0, \quad
                              g_3 = -7.5 - y \le 0, \quad g_4 = y - 7.5 \le 0, \\
                       &\quad g_5 = 5^2 - (x + 10)^2 - y^2 \le 0
     \end{aligned} \]
  <p>Without the obstacle the solution is the nearest pad point
  \(\vec{x}^* = (-10, -4)\) with one active constraint, \(\lambda_1^* = 16\).
  That point is inside the obstacle, so with it the solution moves to
  \(\vec{x}^* = (-10, -5)\) with two active constraints,
  \(\lambda_1^* = 16\) and \(\lambda_5^* = 0.2\); the other multipliers
  are zero.</p>

  <p>Augmented Lagrangian with the inequalities folded in through the
  modified constraints \(\bar g_j\)
  (<a href="https://mdobook.github.io/html/constrained/#augmented-lagrangian">MDO book</a>,
  Eqs. 5.53&ndash;5.54):</p>
  \[ \hat f(\vec{x};\, \vec{\lambda}, \mu) = f(\vec{x}) + \vec{\lambda}^\mathsf{T} \bar g(\vec{x})
       + \frac{\mu}{2}\, \|\bar g(\vec{x})\|_2^2, \qquad
     \bar g_j(\vec{x}) = \begin{cases} g_j(\vec{x}) & g_j(\vec{x}) \ge -\lambda_j/\mu \\
                                        -\lambda_j/\mu & \text{otherwise} \end{cases} \]
  <p>Each outer iteration \(k\) minimizes \(\hat f\) with Newton's method,
  started from the previous minimizer (the first from the pad center), then
  updates</p>
  \[ \vec{\lambda}_{k+1} = \vec{\lambda}_k + \mu_k\, \bar g(\vec{x}_k^*),
     \qquad \mu_{k+1} = \rho\, \mu_k . \]

(function () {
  function $(id) { return document.getElementById(id); }
  function fmt(v, d) { return isFinite(v) ? v.toFixed(d) : 'NaN'; }
  function fmtMu(mu) { return mu >= 100 ? fmt(mu, 0) : mu >= 10 ? fmt(mu, 1) : fmt(mu, 2); }
  var PLOTOPTS = { displayModeBar: false, responsive: true };

  // ---- problem (landing_point.m from the constrained optimization lecture) ----
  var W = 20, H = 15, ROCKET = [-18, -4], OBS = [-10, 0], ROBS = 5;
  function f(x) { return Math.pow(x[0] - ROCKET[0], 2) + Math.pow(x[1] - ROCKET[1], 2); }
  function gradf(x) { return [2 * (x[0] - ROCKET[0]), 2 * (x[1] - ROCKET[1])]; }
  var ALLCONS = [
    { name: 'x ≥ −10 (pad edge)', g: function (x) { return -W / 2 - x[0]; }, grad: function () { return [-1, 0]; }, hess: null },
    { name: 'x ≤ 10',             g: function (x) { return x[0] - W / 2; },  grad: function () { return [1, 0]; },  hess: null },
    { name: 'y ≥ −7.5',           g: function (x) { return -H / 2 - x[1]; }, grad: function () { return [0, -1]; }, hess: null },
    { name: 'y ≤ 7.5',            g: function (x) { return x[1] - H / 2; },  grad: function () { return [0, 1]; },  hess: null },
    { name: 'obstacle',           g: function (x) { return ROBS * ROBS - Math.pow(x[0] - OBS[0], 2) - Math.pow(x[1] - OBS[1], 2); },
      grad: function (x) { return [-2 * (x[0] - OBS[0]), -2 * (x[1] - OBS[1])]; }, hess: [[-2, 0], [0, -2]] }
  ];
  var CONS, NC, XSTAR, LSTAR, FSTAR, GMASK, HASOBS;
  var XINIT = [0, 0];                       // first inner solve starts at the pad center
  function setProblem(obstacle) {
    HASOBS = obstacle;
    CONS = obstacle ? ALLCONS : ALLCONS.slice(0, 4);
    NC = CONS.length;
    XSTAR = obstacle ? [-10, -5] : [-10, -4];
    LSTAR = obstacle ? [16, 0, 0, 0, 0.2] : [16, 0, 0, 0];
    FSTAR = f(XSTAR);
    GMASK = g2.map(function (y) { return g1.map(function (x) {
      return Math.max.apply(null, gvec([x, y])) > 0 ? 1 : 0; }); });
  }

  function gvec(x) { return CONS.map(function (c) { return c.g(x); }); }
  function gbar(x, lam, mu) { return CONS.map(function (c, j) { return Math.max(c.g(x), -lam[j] / mu); }); }
  function fhat(x, lam, mu) {
    var gb = gbar(x, lam, mu), v = f(x);
    for (var j = 0; j < NC; j++) v += lam[j] * gb[j] + 0.5 * mu * gb[j] * gb[j];
    return v;
  }
  // where ḡ_j = g_j the weight w_j = λ_j + μ g_j multiplies ∇g_j; where ḡ_j is clamped the constraint contributes nothing
  function gradfhat(x, lam, mu) {
    var gr = gradf(x);
    CONS.forEach(function (c, j) {
      var w = lam[j] + mu * c.g(x);
      if (w > 0) { var dg = c.grad(x); gr[0] += w * dg[0]; gr[1] += w * dg[1]; }
    });
    return gr;
  }
  function hessfhat(x, lam, mu) {
    var Hm = [[2, 0], [0, 2]];
    CONS.forEach(function (c, j) {
      var w = lam[j] + mu * c.g(x);
      if (w > 0) {
        var dg = c.grad(x);
        Hm[0][0] += mu * dg[0] * dg[0]; Hm[0][1] += mu * dg[0] * dg[1];
        Hm[1][0] += mu * dg[1] * dg[0]; Hm[1][1] += mu * dg[1] * dg[1];
        if (c.hess) { Hm[0][0] += w * c.hess[0][0]; Hm[1][1] += w * c.hess[1][1]; }
      }
    });
    return Hm;
  }

  // Newton with Armijo backtracking; steepest descent where the Hessian is not positive definite.
  function minimize(x0, lam, mu) {
    var x = x0.slice(), path = [x.slice()];
    for (var it = 0; it < 200; it++) {
      var gr = gradfhat(x, lam, mu), gn = Math.hypot(gr[0], gr[1]);
      if (gn < 1e-7) break;
      var Hm = hessfhat(x, lam, mu), det = Hm[0][0] * Hm[1][1] - Hm[0][1] * Hm[1][0], d, a = 1;
      if (Hm[0][0] > 0 && det > 1e-12) {
        d = [-(Hm[1][1] * gr[0] - Hm[0][1] * gr[1]) / det, -(-Hm[1][0] * gr[0] + Hm[0][0] * gr[1]) / det];
      } else { d = [-gr[0], -gr[1]]; a = 2 / gn; }
      var slope = gr[0] * d[0] + gr[1] * d[1], F0 = fhat(x, lam, mu), xn;
      for (var j = 0; j < 40; j++) {
        xn = [x[0] + a * d[0], x[1] + a * d[1]];
        if (fhat(xn, lam, mu) <= F0 + 1e-4 * a * slope) break;
        a *= 0.5;
      }
      if (Math.hypot(xn[0] - x[0], xn[1] - x[1]) < 1e-12) break;   // roundoff floor
      x = xn; path.push(x.slice());
    }
    return { x: x, path: path };
  }

  var KMAX = 12;
  function run(mu0, rho) {
    var it = [], lam = CONS.map(function () { return 0; }), mu = mu0, x = XINIT;
    for (var k = 0; k <= KMAX; k++) {
      var r = minimize(x, lam, mu);
      x = r.x;
      var gv = gvec(x), gb = gbar(x, lam, mu);
      it.push({ k: k, lam: lam.slice(), mu: mu, x: x, g: gv, gbar: gb, path: r.path,
                viol: Math.max.apply(null, gv.map(function (v) { return Math.max(0, v); })) });
      lam = lam.map(function (l, j) { return l + mu * gb[j]; });
      mu *= rho;
    }
    return it;
  }

  // ---- contour plot --------------------------------------------------------
  var X1L = -22, X1R = 12, X2L = -10, X2R = 10, NG = 171;
  var g1 = [], g2 = [];
  for (var i = 0; i < NG; i++) { g1.push(X1L + (X1R - X1L) * i / (NG - 1)); g2.push(X2L + (X2R - X2L) * i / (NG - 1)); }
  var circ1 = [], circ2 = [];
  for (var t = 0; t <= 100; t++) { var th = 2 * Math.PI * t / 100; circ1.push(OBS[0] + ROBS * Math.cos(th)); circ2.push(OBS[1] + ROBS * Math.sin(th)); }

  function drawMain(it, k, i, which) {
    var c = it[k], z = [], zt = [], cur = c.path[i], n = c.path.length - 1;
    var fun = which === 'f' ? f : function (x) { return fhat(x, c.lam, c.mu); };
    var zmin = Infinity, zmax = -Infinity;
    for (var j = 0; j < NG; j++) {
      var row = g1.map(function (x) { return fun([x, g2[j]]); });
      z.push(row); zmin = Math.min(zmin, Math.min.apply(null, row)); zmax = Math.max(zmax, Math.max.apply(null, row));
    }
    // contour levels bunch up near the minimum: plot log(1 + f̂ − f̂min) and show the real value on hover
    for (var j2 = 0; j2 < NG; j2++) zt.push(z[j2].map(function (v) { return Math.log10(1 + v - zmin); }));
    var col = which === 'f' ? '#123f63' : '#b06a00';
    var traces = [
      { type: 'contour', x: g1, y: g2, z: GMASK, showscale: false, hoverinfo: 'skip',
        contours: { coloring: 'fill', start: 0.5, end: 0.5, size: 1, showlines: false },
        colorscale: [[0, 'rgba(0,0,0,0)'], [1, 'rgba(192,42,42,.12)']] },

        contours: { coloring: 'lines', start: 0, end: Math.log10(1 + zmax - zmin), size: 0.1 },
        colorscale: [[0, col], [1, col]], line: { width: 1 },
        hovertemplate: 'x %{x:.2f}, y %{y:.2f}<br>' + (which === 'f' ? 'f' : 'f̂') + ' %{customdata:.3f}<extra></extra>' },
      { type: 'scatter', mode: 'lines', x: [-W / 2, W / 2, W / 2, -W / 2, -W / 2], y: [-H / 2, -H / 2, H / 2, H / 2, -H / 2],
        name: 'landing pad', hoverinfo: 'skip', line: { color: '#2a7a3b', width: 2 } },
      { type: 'scatter', mode: 'lines', x: HASOBS ? circ1 : [], y: HASOBS ? circ2 : [], name: 'obstacle', hoverinfo: 'skip',
        visible: HASOBS, fill: 'toself', fillcolor: 'rgba(192,42,42,.25)', line: { color: '#c02a2a', width: 1.5 } },
      { type: 'scatter', mode: 'markers', x: [ROCKET[0]], y: [ROCKET[1]], name: 'rocket (x₀, y₀)', hoverinfo: 'skip',
        marker: { size: 12, color: '#123f63', symbol: 'triangle-up' } }
    ];
    var ip = c.path.slice(0, i + 1);
    traces.push({ type: 'scatter', mode: 'lines+markers', x: ip.map(function (p) { return p[0]; }), y: ip.map(function (p) { return p[1]; }),

      hovertemplate: 'inner step %{pointNumber}<br>x %{x:.3f}, y %{y:.3f}<extra></extra>' });
    var outer = it.slice(0, i === n ? k + 1 : k);
    traces.push({ type: 'scatter', mode: 'lines+markers', x: outer.map(function (r) { return r.x[0]; }), y: outer.map(function (r) { return r.x[1]; }),

  }

  function drawConv(it, k) {
    var ks = it.map(function (r) { return r.k; });
    function pos(v) { return Math.max(Math.abs(v), 1e-12); }
    function lamerr(r) { return Math.sqrt(r.lam.reduce(function (s, l, j) { return s + Math.pow(l - LSTAR[j], 2); }, 0)); }

  function drawTable(it, k) {
    var html = '<tr><th>k</th><th>μ<sub>k</sub></th><th>λ<sub>1</sub> (pad edge)</th>' + (HASOBS ? '<th>λ<sub>5</sub> (obstacle)</th>' : '') +
