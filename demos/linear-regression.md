# Linear Regression as Optimization

  <p>The model is a line with parameters \(\theta = (a_0, a_1)\),
  \[ \hat{y} = m(x; \theta) = a_0 + a_1 x . \]
  Each data point \((x_i, y_i)\) contributes a loss \(\ell(\hat{y}_i, y_i)\),
  either the squared residual \((y_i - \hat{y}_i)^2\) or the absolute residual
  \(|y_i - \hat{y}_i|\), and the cost is the mean over the \(n\) points,
  \[ J(\theta) = \frac{1}{n} \sum_{i=1}^n \ell\big( a_0 + a_1 x_i,\; y_i \big) . \]
  Fitting the line means solving \(\theta^* = \operatorname{argmin}_\theta J(\theta)\).</p>

  <p><b>Drag the point</b> on the cost plot (or click anywhere on the contours)
  to move \(\theta\) and watch the line and residuals above change.</p>

  <p>With the squared loss the contours are ellipses because \(J\) is
  quadratic in \(\theta\): the residuals are linear in \(a_0\) and \(a_1\),
  so their squares are quadratic. A quadratic cost has a single minimum, and
  setting \(\nabla J = 0\) gives a linear system (the normal equations) that
  can be solved directly. The ellipses are tilted because \(a_0\) and \(a_1\)
  trade off: raising the slope can be partly compensated by lowering the
  intercept. With the absolute loss \(J\) is piecewise linear with creases
  wherever the line passes through a data point, so there is no closed-form
  solution and \(\nabla J\) jumps across the creases; the \(\theta^*\) shown is
  found numerically. The absolute loss is less sensitive to outliers because a
  far-away point pulls with a fixed force instead of one that grows with the
  residual: add the outlier and compare where \(\theta^*\) lands under each
  loss.</p>

<script>
(function () {
  function $(id) { return document.getElementById(id); }
  function fmt(v, d) { return isFinite(v) ? v.toFixed(d) : 'NaN'; }
  var PLOTOPTS = { displayModeBar: false, responsive: true };
  var BLUE = '#123f63', RED = '#c02a2a', AMB = '#b06a00', GRN = '#2a7a3b';

  // ---- data ------------------------------------------------------------------
  var n0 = 12, XMAX = 10, X0 = [], Y0 = [], X = [], Y = [], n = 0;
  var OUTLIER = [9.3, -1];                            // one far-off point, appended when the box is checked
  var A0BOX = [-5, 7], A1BOX = [-1, 2];            // cost plot axes
  var XBOX = [-0.5, XMAX + 0.5], YBOX = [-2, 10];   // data plot axes (fixed)
  var ABSW = 0.1;                                   // rectangle width (data units) for the absolute loss
  function randn() { var u = 1 - Math.random(), v = Math.random(); return Math.sqrt(-2 * Math.log(u)) * Math.cos(2 * Math.PI * v); }
  function newData() {
    X0 = []; Y0 = [];
    var a0 = 1 + 0.8 * (Math.random() - 0.5), a1 = 0.6 + 0.3 * (Math.random() - 0.5);
    for (var i = 0; i < n0; i++) {
      var x = XMAX * (i + 0.5 + 0.6 * (Math.random() - 0.5)) / n0;
      X0.push(x); Y0.push(a0 + a1 * x + 0.9 * randn());
    }
    assemble();
  }
  function assemble() {                                // base data plus the optional outlier
    X = X0.slice(); Y = Y0.slice();
    if ($('lr-out').checked) { X.push(OUTLIER[0]); Y.push(OUTLIER[1]); }
    n = X.length;
  }

  // ---- model, cost, and the exact solution ----------------------------------
  function m(x, a0, a1) { return a0 + a1 * x; }
  function lossMode() { return document.querySelector('input[name=lr-loss]:checked').value; }
  function loss(r) { return lossMode() === 'sq' ? r * r : Math.abs(r); }
  function dloss(r) { return lossMode() === 'sq' ? 2 * r : (r > 0 ? 1 : r < 0 ? -1 : 0); }   // subgradient for |r|
  function J(a0, a1) {
    var s = 0;
    for (var i = 0; i < n; i++) s += loss(Y[i] - m(X[i], a0, a1));
    return s / n;
  }
  function gradJ(a0, a1) {
    var g0 = 0, g1 = 0;
    for (var i = 0; i < n; i++) { var d = dloss(Y[i] - m(X[i], a0, a1)); g0 -= d; g1 -= d * X[i]; }
    return [g0 / n, g1 / n];
  }
  function wls(w) {                                    // weighted least squares, 2x2 normal equations
    var sw = 0, sx = 0, sy = 0, sxx = 0, sxy = 0;
    for (var i = 0; i < n; i++) { sw += w[i]; sx += w[i] * X[i]; sy += w[i] * Y[i]; sxx += w[i] * X[i] * X[i]; sxy += w[i] * X[i] * Y[i]; }
    var det = sw * sxx - sx * sx;
    return [(sxx * sy - sx * sxy) / det, (sw * sxy - sx * sy) / det];
  }
  function solve() {
    var w = X.map(function () { return 1; }), th = wls(w);
    if (lossMode() === 'sq') return th;
    for (var it = 0; it < 200; it++) {                 // iteratively reweighted least squares for the L1 fit
      for (var i = 0; i < n; i++) w[i] = 1 / Math.max(Math.abs(Y[i] - m(X[i], th[0], th[1])), 1e-6);
      var nt = wls(w), dx = Math.abs(nt[0] - th[0]) + Math.abs(nt[1] - th[1]); th = nt;
      if (dx < 1e-9) break;
    }
    return th;
  }

  // ---- drawing ---------------------------------------------------------------
  var GN = 101, cost = null, star = null, made = false, dragging = false;
  function clamp(v, lo, hi) { return Math.min(hi, Math.max(lo, v)); }
  function theta() { return [+$('lr-a0').value, +$('lr-a1').value]; }
  function setTheta(a0, a1) {
    $('lr-a0').value = clamp(a0, A0BOX[0], A0BOX[1]);
    $('lr-a1').value = clamp(a1, A1BOX[0], A1BOX[1]);
    draw();
  }

  function costGrid() {
    var as = [], bs = [];
    for (var i = 0; i < GN; i++) {
      as.push(A1BOX[0] + (A1BOX[1] - A1BOX[0]) * i / (GN - 1));   // a1 (slope) on x
      bs.push(A0BOX[0] + (A0BOX[1] - A0BOX[0]) * i / (GN - 1));   // a0 (intercept) on y
    }
    var sq = lossMode() === 'sq';   // sqrt of a quadratic gives evenly spaced ellipses; |r| cost is already cone-like
    var z = bs.map(function (a0) { return as.map(function (a1) { var v = J(a0, a1); return sq ? Math.sqrt(v) : v; }); });
    var zmax = -Infinity, zmin = Infinity;
    z.forEach(function (r) { r.forEach(function (v) { if (v > zmax) zmax = v; if (v < zmin) zmin = v; }); });
    cost = { as: as, bs: bs, z: z, zmin: zmin, zmax: zmax };
    star = solve();
  }

  function draw() {
    var th = theta(), a0 = th[0], a1 = th[1];
    $('lr-a0v').textContent = fmt(a0, 2); $('lr-a1v').textContent = fmt(a1, 2);
    var yhat = X.map(function (x) { return m(x, a0, a1); });
    var res = Y.map(function (y, i) { return y - yhat[i]; });
    var Jv = J(a0, a1), g = gradJ(a0, a1), Jstar = J(star[0], star[1]);

    // -- top pane: data, line, residual segments --
    var rx = [], ry = [];
    for (var i = 0; i < n; i++) { rx.push(X[i], X[i], null); ry.push(Y[i], yhat[i], null); }
    var dt = [
      { type: 'scatter', mode: 'lines', x: rx, y: ry, line: { color: RED, width: 2 }, hoverinfo: 'skip', showlegend: false },
      { type: 'scatter', mode: 'lines', x: [-1e3, 1e3], y: [m(-1e3, a0, a1), m(1e3, a0, a1)],
        line: { color: AMB, width: 3 }, hoverinfo: 'skip', showlegend: false },
      { type: 'scatter', mode: 'markers', x: X, y: Y, showlegend: false,
        marker: { size: 9, color: BLUE, line: { width: X.map(function (x, i) { return i >= n0 ? 2 : 0; }), color: RED } },
        customdata: res.map(function (r, i) { return [r, loss(r), yhat[i]]; }),
        hovertemplate: 'x = %{x:.2f}, y = %{y:.2f}<br>ŷ = %{customdata[2]:.2f}<br>r = %{customdata[0]:.2f}, ℓ = ' +
                       (lossMode() === 'sq' ? 'r²' : '|r|') + ' = %{customdata[1]:.3f}<extra></extra>' }
    ];
    var shapes = [];
    if ($('lr-sq').checked) {
      for (i = 0; i < n; i++) {
        var hw = Math.abs(res[i]) / 20;                          // squared: height : width = 10 : 1, area = r²/10
        if (lossMode() === 'abs') hw = ABSW / 2;                   // absolute: fixed width, area = ABSW·|r|
        shapes.push({ type: 'rect', x0: X[i] - hw, x1: X[i] + hw, y0: yhat[i], y1: Y[i],
                      line: { color: RED, width: 1 }, fillcolor: 'rgba(192,42,42,0.18)' });
      }
    }
    var dl = { margin: { l: 40, r: 10, t: 10, b: 40 }, paper_bgcolor: 'rgba(0,0,0,0)', shapes: shapes,
               xaxis: { title: 'x', range: XBOX, fixedrange: true },
               yaxis: { title: 'y', range: YBOX, fixedrange: true } };
    Plotly.react('lr-data', dt, dl, PLOTOPTS);
    drawJBox(Jv);

    // -- bottom pane: contours of J with the draggable theta (a1 on x, a0 on y) --
    var showC = $('lr-ct').checked;
    var ct = [
      { type: 'contour', visible: showC, x: cost.as, y: cost.bs, z: cost.z, zmin: cost.zmin, zmax: cost.zmax, showscale: false,
        colorscale: [[0, '#a9cbe6'], [1, '#f4f8fc']], line: { width: 1, color: '#5b86b0' },
        contours: { start: cost.zmin, end: cost.zmax, size: (cost.zmax - cost.zmin) / 24, coloring: 'fill' },
        hoverinfo: 'skip' },
      { type: 'scatter', mode: 'markers', x: [star[1]], y: [star[0]], hoverinfo: 'skip', showlegend: false, visible: showC,
        marker: { size: 12, symbol: 'star', color: GRN, line: { color: '#fff', width: 1 } } },
      { type: 'scatter', mode: 'markers', x: [a1], y: [a0], hoverinfo: 'skip', showlegend: false,
        marker: { size: 16, color: AMB, line: { color: '#fff', width: 2 } } }
    ];
    var cl = { margin: { l: 45, r: 10, t: 10, b: 40 }, paper_bgcolor: 'rgba(0,0,0,0)', dragmode: false,
               xaxis: { title: 'a₁ (slope)', range: A1BOX, fixedrange: true },
               yaxis: { title: 'a₀ (intercept)', range: A0BOX, fixedrange: true } };
    Plotly.react('lr-cost', ct, cl, PLOTOPTS);

    $('lr-note').innerHTML =
      '<div><span style="color:' + AMB + ';font-weight:600">θ</span> = (' + fmt(a0, 2) + ', ' + fmt(a1, 2) + '),  ' +
      'J(θ) = ' + fmt(Jv, 3) + ',  ∇J = (' + fmt(g[0], 2) + ', ' + fmt(g[1], 2) + ')</div>' +
      (showC ? '<div><span style="color:' + GRN + ';font-weight:600">θ*</span> = (' + fmt(star[0], 2) + ', ' + fmt(star[1], 2) + '),  ' +
      'J(θ*) = ' + fmt(Jstar, 3) + '</div>' : '<div>&nbsp;</div>');

    if (!made) { made = true; setupDrag(); }
  }

  // ---- rectangle with area J, drawn beside the data axes at the same scale ---
  function drawJBox(Jv) {
    var svg = $('lr-jbox'), gd = $('lr-data'), fl = gd._fullLayout;
    while (svg.firstChild) svg.removeChild(svg.firstChild);
    if (!$('lr-sq').checked || !fl) return;
    var ppx = fl.xaxis._length / (XBOX[1] - XBOX[0]), ppy = fl.yaxis._length / (YBOX[1] - YBOX[0]);   // pixels per data unit
    var jh, jw;
    if (lossMode() === 'sq') { jh = Math.sqrt(10 * Jv); jw = jh / 10; }   // 10 : 1 like the residuals, area = J/10
    else { jh = Jv; jw = ABSW; }                                          // same width as the residuals, area = ABSW·J
    var W = svg.clientWidth || 72, cx = W / 2, y0 = fl.yaxis._offset + fl.yaxis.d2p(0);
    var top = Math.max(fl.yaxis._offset, y0 - jh * ppy);
    function el(tag, attrs) { var e = document.createElementNS('http://www.w3.org/2000/svg', tag);
      for (var k in attrs) e.setAttribute(k, attrs[k]); svg.appendChild(e); return e; }
    el('rect', { x: cx - jw * ppx / 2, y: top, width: Math.max(jw * ppx, 1), height: y0 - top });
    el('text', { x: cx, y: y0 + 16 }).textContent = 'J(θ)';
    if (y0 - jh * ppy < fl.yaxis._offset) el('text', { x: cx, y: fl.yaxis._offset - 4 }).textContent = '⋮';
  }

  // ---- dragging on the cost plot ---------------------------------------------
  function pixToData(gd, ev) {
    var fl = gd._fullLayout, bb = gd.getBoundingClientRect();
    return [fl.xaxis.p2d(ev.clientX - bb.left - fl.xaxis._offset),
            fl.yaxis.p2d(ev.clientY - bb.top - fl.yaxis._offset)];
  }
  function inPlot(gd, ev) {
    var fl = gd._fullLayout, bb = gd.getBoundingClientRect();
    var px = ev.clientX - bb.left, py = ev.clientY - bb.top;
    return px >= fl.xaxis._offset && px <= fl.xaxis._offset + fl.xaxis._length &&
           py >= fl.yaxis._offset && py <= fl.yaxis._offset + fl.yaxis._length;
  }
  function setupDrag() {
    var gd = $('lr-cost');
    gd.addEventListener('pointerdown', function (ev) {
      if (!inPlot(gd, ev)) return;
      dragging = true; ev.preventDefault();
      var p = pixToData(gd, ev); setTheta(p[1], p[0]);
    });
    window.addEventListener('pointermove', function (ev) {
      if (!dragging) return; ev.preventDefault();
      var p = pixToData(gd, ev); setTheta(p[1], p[0]);
    });
    window.addEventListener('pointerup', function () { dragging = false; });
    window.addEventListener('pointercancel', function () { dragging = false; });
  }

  // ---- wiring ------------------------------------------------------------------
  ['lr-a0', 'lr-a1'].forEach(function (id) { $(id).addEventListener('input', draw); });
  ['lr-sq', 'lr-ct'].forEach(function (id) { $(id).addEventListener('change', draw); });
  $('lr-solve').addEventListener('click', function () { setTheta(star[0], star[1]); });
  $('lr-new').addEventListener('click', function () { newData(); costGrid(); draw(); });
  $('lr-out').addEventListener('change', function () { assemble(); costGrid(); draw(); });
  document.querySelectorAll('input[name=lr-loss]').forEach(function (el) { el.addEventListener('change', function () { costGrid(); draw(); }); });
  window.addEventListener('resize', function () { setTimeout(draw, 60); });   // keep the J box in step with the responsive plot
  newData(); costGrid(); draw();
})();
</script>
