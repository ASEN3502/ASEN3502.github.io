  <p class="mm-intro">Suppose we have written a solver for the initial value
  problem
  \[ y' = f(t, y), \qquad y(0) = y_0, \]
  with, say, \(f(t, y) = -y^2\) plus some forcing.  How do we check that the
  code is right when we cannot solve the problem by hand?  The <b>method of
  manufactured solutions</b> turns the question around: <i>pick the answer
  first</i>.  Choose any smooth function \(y_m(t)\), then add a source term
  to the right-hand side so that \(y_m\) is exactly a solution:
  \[ y' = f(t, y) + s(t), \qquad s(t) = y_m'(t) - f\big(t, y_m(t)\big). \]
  The solver never sees \(y_m\), only \(f + s\) and the initial condition
  \(y_m(0)\).  If the code is correct, the error should shrink at the
  method's theoretical rate \(O(h^p)\) as the step size \(h\) decreases.  If
  it does not, there is a bug.</p>

(function () {
  var T = 4;                                    // solve on [0, T]
  function f(t, y) { return -y * y; }           // the "real" right-hand side

  // Manufactured solutions and their exact derivatives.
  var SOLS = [
    { y: function (t) { return 1 + Math.sin(t) / 2; },
      dy: function (t) { return Math.cos(t) / 2; },
      ym: 'y<sub>m</sub>(t) = 1 + (sin t) / 2',
      dym: 'y<sub>m</sub>′(t) = (cos t) / 2',
      s: 's(t) = (cos t) / 2 + (1 + (sin t) / 2)<sup>2</sup>' },
    { y: function (t) { return 1 + Math.exp(-t) * Math.cos(3 * t) / 2; },
      dy: function (t) { return -Math.exp(-t) * (Math.cos(3 * t) + 3 * Math.sin(3 * t)) / 2; },
      ym: 'y<sub>m</sub>(t) = 1 + e<sup>−t</sup> (cos 3t) / 2',
      dym: 'y<sub>m</sub>′(t) = −e<sup>−t</sup> (cos 3t + 3 sin 3t) / 2',
      s: 's(t) = −e<sup>−t</sup> (cos 3t + 3 sin 3t) / 2 + (1 + e<sup>−t</sup> (cos 3t) / 2)<sup>2</sup>' },
    { y: function (t) { return 1 + Math.pow(t - 2, 3) / 16; },
      dy: function (t) { return 3 * (t - 2) * (t - 2) / 16; },
      ym: 'y<sub>m</sub>(t) = 1 + (t − 2)<sup>3</sup> / 16',
      dym: 'y<sub>m</sub>′(t) = 3 (t − 2)<sup>2</sup> / 16',
      s: 's(t) = 3 (t − 2)<sup>2</sup> / 16 + (1 + (t − 2)<sup>3</sup> / 16)<sup>2</sup>' }
  ];

  // One step of each method: (F, t, y, h) -> y_{n+1}.  The last two contain
  // deliberate mistakes of the kind that are easy to make and hard to spot:
  // the code runs, and the answer still looks roughly right.
  var METHODS = [
    { name: 'Euler', order: 1,
      step: function (F, t, y, h) { return y + h * F(t, y); },
      code: ['k1 = F(t, y)', 'y = y + h * k1'] },
    { name: 'Heun (RK2)', order: 2,
      step: function (F, t, y, h) {
        var k1 = F(t, y), k2 = F(t + h, y + h * k1);
        return y + h * (0.5 * k1 + 0.5 * k2); },
      code: ['k1 = F(t, y)', 'k2 = F(t + h, y + h * k1)', 'y = y + h * (k1 + k2) / 2'] },
    { name: 'classical RK4', order: 4,
      step: function (F, t, y, h) {
        var k1 = F(t, y), k2 = F(t + h / 2, y + h / 2 * k1),
            k3 = F(t + h / 2, y + h / 2 * k2), k4 = F(t + h, y + h * k3);
        return y + h / 6 * (k1 + 2 * k2 + 2 * k3 + k4); },
      code: ['k1 = F(t, y)', 'k2 = F(t + h/2, y + h/2 * k1)', 'k3 = F(t + h/2, y + h/2 * k2)',
             'k4 = F(t + h, y + h * k3)', 'y = y + h * (k1 + 2 k2 + 2 k3 + k4) / 6'] },
    { name: 'Heun with a typo', order: 2, bug: 2,
      step: function (F, t, y, h) {
        var k1 = F(t, y), k2 = F(t + h, y + h * k1);
        return y + h * (0.4 * k1 + 0.6 * k2); },
      code: ['k1 = F(t, y)', 'k2 = F(t + h, y + h * k1)', 'y = y + h * (0.4 k1 + 0.6 k2)'] },
    { name: 'RK4 with a typo', order: 4, bug: 2,
      step: function (F, t, y, h) {
        var k1 = F(t, y), k2 = F(t + h / 2, y + h / 2 * k1),
            k3 = F(t + h / 2, y + h * k2), k4 = F(t + h, y + h * k3);
        return y + h / 6 * (k1 + 2 * k2 + 2 * k3 + k4); },
      code: ['k1 = F(t, y)', 'k2 = F(t + h/2, y + h/2 * k1)', 'k3 = F(t + h/2, y + h * k2)',
             'k4 = F(t + h, y + h * k3)', 'y = y + h * (k1 + 2 k2 + 2 k3 + k4) / 6'] }
  ];

  var svg1 = document.getElementById('mm-svg1'),
      svg2 = document.getElementById('mm-svg2'),
      nIn  = document.getElementById('mm-n'),
      solIn = document.getElementById('mm-sol'),
      methIn = document.getElementById('mm-meth'),
      descOut = document.getElementById('mm-desc'),
      nOut = document.getElementById('mm-nval'),
      hOut = document.getElementById('mm-h'),
      sumOut = document.getElementById('mm-summary'),
      code = document.getElementById('mm-code');

  var W = 640, H = 340, ML = 62, MR = 16, MT = 14, MB = 44;
  var PW = W - ML - MR, PH = H - MT - MB;

  function el(name, attrs, text) {
    var e = document.createElementNS('http://www.w3.org/2000/svg', name);
    for (var a in attrs) e.setAttribute(a, attrs[a]);
    if (text !== undefined) e.textContent = text;
    return e;
  }
  function clear(svg) { while (svg.firstChild) svg.removeChild(svg.firstChild); }

  // Solve y' = F(t, y), y(0) = y0 with N steps; return the grid values.
  function solve(method, F, y0, N) {
    var h = T / N, ys = [y0], y = y0;
    for (var n = 0; n < N; n++) { y = method.step(F, n * h, y, h); ys.push(y); }
    return ys;
  }
  function maxError(sol, ys, N) {
    var h = T / N, e = 0;
    for (var n = 0; n <= N; n++) e = Math.max(e, Math.abs(ys[n] - sol.y(n * h)));
    return e;
  }

  function render() {
    var sol = SOLS[+solIn.value], method = METHODS[+methIn.value];
    var N = Math.pow(2, +nIn.value), h = T / N;
    nOut.textContent = N;
    hOut.textContent = h.toFixed(4);

    // The right-hand side the solver actually sees.
    var F = function (t, y) { return f(t, y) + sol.dy(t) - f(t, sol.y(t)); };

    descOut.innerHTML = 'Solving <code>y′ = −y<sup>2</sup> + s(t)</code>, ' +
      '<code>y(0) = ' + sol.y(0).toFixed(2) + '</code>, where the source term is chosen so that ' +
      '<code>' + sol.ym + '</code> is the exact solution:<br>' +
      '<code>' + sol.dym + '</code>, &nbsp; so &nbsp; ' +
      '<code>s(t) = y<sub>m</sub>′ − f(t, y<sub>m</sub>) = y<sub>m</sub>′ + y<sub>m</sub><sup>2</sup></code>, ' +
      '&nbsp; i.e. &nbsp; <code>' + sol.s + '</code>.';

    // ---- plot 1: exact vs numerical -----------------------------------
    var ys = solve(method, F, sol.y(0), N);
    var YLO = 0, YHI = 2;
    function px(t) { return ML + t / T * PW; }
    function py(y) { return MT + PH - (y - YLO) / (YHI - YLO) * PH; }
    clear(svg1);
    for (var yt = YLO; yt <= YHI + 1e-9; yt += 0.5) {
      svg1.appendChild(el('line', { x1: ML, y1: py(yt), x2: ML + PW, y2: py(yt), class: 'mm-tick' }));
      svg1.appendChild(el('text', { x: ML - 8, y: py(yt) + 4, 'text-anchor': 'end',
                                    class: 'mm-ticklbl' }, yt.toFixed(1)));
    }
    for (var xt = 0; xt <= T; xt += 1) {
      svg1.appendChild(el('line', { x1: px(xt), y1: MT, x2: px(xt), y2: MT + PH, class: 'mm-tick' }));
      svg1.appendChild(el('text', { x: px(xt), y: MT + PH + 18, 'text-anchor': 'middle',
                                    class: 'mm-ticklbl' }, xt.toFixed(0)));
    }
    svg1.appendChild(el('rect', { x: ML, y: MT, width: PW, height: PH, class: 'mm-axis' }));
    svg1.appendChild(el('text', { x: ML + PW / 2, y: H - 6, 'text-anchor': 'middle',
                                  class: 'mm-axlbl', 'font-style': 'italic' }, 't'));
    svg1.appendChild(el('text', { x: 16, y: MT + PH / 2 + 5, 'text-anchor': 'middle',
                                  class: 'mm-axlbl', 'font-style': 'italic' }, 'y'));
    var d = '', M = 400;
    for (var i = 0; i <= M; i++) {
      var t = T * i / M;
      d += (i ? ' L' : 'M') + px(t).toFixed(2) + ',' + py(sol.y(t)).toFixed(2);
    }
    svg1.appendChild(el('path', { d: d, class: 'mm-curve' }));
    var dn = '', shown = 0;
    for (var n = 0; n <= N; n++) {
      var yn = Math.min(Math.max(ys[n], YLO - 0.2), YHI + 0.2);
      dn += (n ? ' L' : 'M') + px(n * h).toFixed(2) + ',' + py(yn).toFixed(2);
    }
    svg1.appendChild(el('path', { d: dn, class: 'mm-numline' }));
    for (n = 0; n <= N; n++) {
      if (ys[n] >= YLO && ys[n] <= YHI && (N <= 128 || n % (N / 128) === 0)) {
        svg1.appendChild(el('circle', { cx: px(n * h), cy: py(ys[n]),
                                        r: N <= 32 ? 4.5 : 3, class: 'mm-pt' }));
        shown++;
      }
    }

    // ---- plot 2: convergence study ------------------------------------
    var pts = [];
    for (var k = 3; k <= 10; k++) {
      var Nk = Math.pow(2, k), e = maxError(sol, solve(method, F, sol.y(0), Nk), Nk);
      pts.push({ N: Nk, h: T / Nk, e: e });
    }
    // least-squares slope of log(e) vs log(h), ignoring round-off-dominated points
    var fitPts = pts.filter(function (p) { return isFinite(p.e) && p.e > 1e-12 && p.e < 10; });
    var sx = 0, sy = 0, sxx = 0, sxy = 0, m = fitPts.length;
    fitPts.forEach(function (p) {
      var lx = Math.log10(p.h), ly = Math.log10(p.e);
      sx += lx; sy += ly; sxx += lx * lx; sxy += lx * ly;
    });
    var slope = (m * sxy - sx * sy) / (m * sxx - sx * sx),
        icpt = (sy - slope * sx) / m;

    var LXLO = Math.log10(T / 1024) - 0.15, LXHI = Math.log10(1) + 0.15;
    var les = pts.filter(function (p) { return isFinite(p.e) && p.e > 0; })
                 .map(function (p) { return Math.log10(p.e); });
    var LYLO = Math.floor(Math.max(Math.min.apply(null, les), -15) - 0.3),
        LYHI = Math.ceil(Math.min(Math.max.apply(null, les), 6) + 0.3);
    if (LYHI - LYLO < 3) LYHI = LYLO + 3;
    function lx(h) { return ML + (Math.log10(h) - LXLO) / (LXHI - LXLO) * PW; }
    function ly(e) { return MT + PH - (Math.log10(e) - LYLO) / (LYHI - LYLO) * PH; }
    clear(svg2);
    for (var dec = LYLO; dec <= LYHI; dec++) {
      svg2.appendChild(el('line', { x1: ML, y1: ly(Math.pow(10, dec)), x2: ML + PW,
                                    y2: ly(Math.pow(10, dec)), class: 'mm-tick' }));
      if ((LYHI - LYLO <= 8) || dec % 2 === 0) {
        var lbl = el('text', { x: ML - 8, y: ly(Math.pow(10, dec)) + 4, 'text-anchor': 'end',
                               class: 'mm-ticklbl' }, '10');
        var sup = el('tspan', { dy: -6, 'font-size': '9px' }, dec);
        lbl.appendChild(sup);
        svg2.appendChild(lbl);
      }
    }
    pts.forEach(function (p) {
      svg2.appendChild(el('line', { x1: lx(p.h), y1: MT, x2: lx(p.h), y2: MT + PH, class: 'mm-tick' }));
      svg2.appendChild(el('text', { x: lx(p.h), y: MT + PH + 18, 'text-anchor': 'middle',
                                    class: 'mm-ticklbl' }, p.h < 0.01 ? p.h.toFixed(4) : p.h < 0.1 ? p.h.toFixed(3) : p.h.toFixed(2)));
    });
    svg2.appendChild(el('rect', { x: ML, y: MT, width: PW, height: PH, class: 'mm-axis' }));
    svg2.appendChild(el('text', { x: ML + PW / 2, y: H - 6, 'text-anchor': 'middle',
                                  class: 'mm-axlbl' }, 'step size h  (log scale)'));
    var ylab = el('text', { x: 16, y: MT + PH / 2, 'text-anchor': 'middle', class: 'mm-axlbl',
                            transform: 'rotate(-90 16 ' + (MT + PH / 2) + ')' }, 'max |yₙ − yₘ(tₙ)|');
    svg2.appendChild(ylab);
    if (m >= 2 && isFinite(slope)) {
      var h0 = Math.pow(10, LXLO + 0.05), h1 = Math.pow(10, LXHI - 0.05);
      var e0 = Math.pow(10, icpt + slope * Math.log10(h0)),
          e1 = Math.pow(10, icpt + slope * Math.log10(h1));
      svg2.appendChild(el('path', { d: 'M' + lx(h0) + ',' + Math.min(Math.max(ly(e0), MT), MT + PH) +
                                       ' L' + lx(h1) + ',' + Math.min(Math.max(ly(e1), MT), MT + PH),
                                    class: 'mm-fitline' }));
      svg2.appendChild(el('text', { x: ML + 12, y: MT + 20, class: 'mm-fitlbl' },
                          'slope ≈ ' + slope.toFixed(2) + '   (expected ' + method.order + ')'));
    }
    pts.forEach(function (p) {
      if (!(isFinite(p.e) && p.e > 0)) return;
      var cy = Math.min(Math.max(ly(p.e), MT), MT + PH);
      svg2.appendChild(el('circle', { cx: lx(p.h), cy: cy, r: p.N === N ? 7 : 4.5,
                                      class: p.N === N ? 'mm-curpt' : 'mm-errpt' }));
    });

    // ---- summary and code ---------------------------------------------
    var cur = pts.filter(function (p) { return p.N === N; })[0];
    var ok = Math.abs(slope - method.order) < 0.3;
    var s = 'With N = ' + N + ' the maximum error is <b>' + cur.e.toExponential(2) +
            '</b>.  Over the whole range of step sizes the error scales like ' +
            'h<sup>' + slope.toFixed(2) + '</sup>; ' + method.name + ' should be order ' +
            method.order + '.  ';
    if (ok) {
      s += '<span class="mm-pass">Observed order matches the theory</span>: no evidence of a bug.';
    } else if (method.bug) {
      s += '<span class="mm-fail">Observed order does not match</span>: the manufactured ' +
           'solution has caught the bug highlighted below.  Notice that the numerical ' +
           'solution in the top plot still looks perfectly reasonable, which is exactly why ' +
           'eyeballing a plot is not verification.';
    } else {
      s += '<span class="mm-fail">Observed order does not match</span>.  Try a different ' +
           'manufactured solution; a polynomial of low degree can make a method look better than it is.';
    }
    sumOut.innerHTML = s;

    var lines = method.code.map(function (l) {
      var bug = method.bug && l === method.code[method.bug];
